1//! Retention for conversation transcripts.
2//!
3//! Transcripts past the retention window are removed from `~/.claude/projects`
4//! and the sync repository together: removing them on one side only means the
5//! next sync restores them from the other.
6//!
7//! The window is never shorter than `cleanupPeriodDays`, which is when Claude
8//! Code deletes the transcript anyway, nor shorter than six months.
9
10use anyhow::{bail, Context, Result};
11use chrono::{DateTime, Duration, Utc};
12use rayon::prelude::*;
13use serde_json::Value;
14use std::path::{Path, PathBuf};

16/// Shortest retention window applied by default: six months.
17pub const MINIMUM_RETENTION_DAYS: u32 = 180;
18
19/// Claude Code's documented default for `cleanupPeriodDays`.
20pub const CLAUDE_CODE_DEFAULT_RETENTION_DAYS: u32 = 30;
21
22/// One session that has aged out, with every file belonging to it.
23#[derive(Debug, Clone)]
24pub struct PurgeTarget {
25    pub session_id: String,
26    /// Timestamp of the last message, which the age is measured from.
27    pub last_activity: DateTime<Utc>,
28    pub age_days: i64,
29    /// Files and directories to remove, across both trees.
30    pub paths: Vec<PathBuf>,
31    pub bytes: u64,
32}
33
34/// What a purge would remove, computed before anything is deleted.
35#[derive(Debug, Default)]
36pub struct PurgePlan {
37    pub retention_days: u32,
38    pub cutoff: Option<DateTime<Utc>>,
39    pub targets: Vec<PurgeTarget>,
40    /// Sessions kept because at least one copy has no readable age.
41    pub undated: usize,
42    /// Transcript files that failed to parse, such as one being written right
43    /// now. Their session is kept.
44    pub unreadable: usize,
45}
46
47impl PurgePlan {
48    pub fn is_empty(&self) -> bool {
49        self.targets.is_empty()
50    }
51
52    pub fn total_bytes(&self) -> u64 {
53        self.targets.iter().map(|t| t.bytes).sum()
54    }
55
56    /// Age of the oldest session in the plan, in days.
57    pub fn oldest_age_days(&self) -> i64 {
58        self.targets.iter().map(|t| t.age_days).max().unwrap_or(0)
59    }
60}

62/// What a purge removed.
63#[derive(Debug, Default)]
64pub struct PurgeReport {
65    pub sessions: usize,
66    pub files: usize,
67    pub bytes: u64,
68    /// Paths that could not be removed, with the reason.
69    pub failures: Vec<String>,
70}

72/// Retention window in days: the longer of [`MINIMUM_RETENTION_DAYS`] and this
73/// machine's `cleanupPeriodDays`, lengthened (never shortened) by a configured
74/// override. A purge can run unattended after every sync, so a short window
75/// set once, or left in an older config, must not quietly delete recent work.
76pub fn retention_days(claude_dir: &Path, configured: Option<u32>) -> u32 {
77    let floor = MINIMUM_RETENTION_DAYS.max(claude_code_retention_days(claude_dir));
78    match configured {
79        Some(days) if days < floor => {
80            log::warn!(
81                "Purge window of {days} days is below the {floor}-day minimum; using {floor}"
82            );
83            floor
84        }
85        Some(days) => days,
86        None => floor,
87    }
88}
89
90/// `cleanupPeriodDays` from the user-level settings files, or
91/// [`CLAUDE_CODE_DEFAULT_RETENTION_DAYS`].
92///
93/// The longest of them wins rather than the one Claude Code's precedence picks:
94/// the value only ever holds a deletion back. Project-level settings are not
95/// consulted; `--older-than` covers that case.
96fn claude_code_retention_days(claude_dir: &Path) -> u32 {
97    let configured = ["settings.json", "settings.local.json"]
98        .into_iter()
99        .filter_map(|name| std::fs::read_to_string(claude_dir.join(name)).ok())
100        .filter_map(|text| serde_json::from_str::<Value>(&text).ok())
101        .filter_map(|settings| {
102            settings
103                .get("cleanupPeriodDays")
104                .and_then(Value::as_u64)
105                .filter(|days| *days > 0)
106        })
107        .map(|days| u32::try_from(days).unwrap_or(u32::MAX))
108        .max();
109
110    configured.unwrap_or(CLAUDE_CODE_DEFAULT_RETENTION_DAYS)
111}

113/// Every copy of one session found across the trees, before deciding its fate.
114#[derive(Default)]
115struct SessionCopies {
116    last_activity: Option<DateTime<Utc>>,
117    /// Set when any copy has no readable age, which keeps the session.
118    undatable: bool,
119    paths: Vec<PathBuf>,
120    bytes: u64,
121}
122
123/// Classify what a purge would remove from both project trees, without
124/// deleting anything.
125///
126/// A session is decided as a whole — every copy of it, with its sidecars — and
127/// its age is that of the most recent copy.
128pub fn plan(
129    claude_projects_dir: &Path,
130    repo_projects_dir: &Path,
131    retention_days: u32,
132) -> Result<PurgePlan> {
133    let mut plan = PurgePlan {
134        retention_days,
135        ..Default::default()
136    };
137    // Checked: a window large enough to overflow has nothing to purge anyway.
138    let Some(cutoff) = Utc::now().checked_sub_signed(Duration::days(i64::from(retention_days)))
139    else {
140        return Ok(plan);
141    };
142    plan.cutoff = Some(cutoff);
143
144    let mut by_session: std::collections::BTreeMap<String, SessionCopies> =
145        std::collections::BTreeMap::new();
146
147    for transcripts_root in [claude_projects_dir, repo_projects_dir] {
148        // Summarizing is the whole cost of planning a purge, and every file is
149        // independent: read them across the cores, then fold the results in the
150        // order the walk found them.
151        let summarized: Vec<(PathBuf, Result<crate::parser::ConversationSession>)> =
152            transcripts(transcripts_root)
153                .into_par_iter()
154                .map(|transcript| {
155                    let summary = crate::parser::ConversationSession::from_file(&transcript);
156                    (transcript, summary)
157                })
158                .collect();
159
160        for (transcript, summary) in summarized {
161            let session = match summary {
162                Ok(session) => session,
163                Err(error) => {
164                    // The session is kept whole: a copy that cannot be read
165                    // says nothing about the age of the one that can.
166                    log::warn!("Skipping unreadable {}: {error}", transcript.display());
167                    plan.unreadable += 1;
168                    if let Some(stem) = transcript.file_stem().and_then(|s| s.to_str()) {
169                        by_session.entry(stem.to_string()).or_default().undatable = true;
170                    }
171                    continue;
172                }
173            };
174
175            let paths: Vec<PathBuf> = std::iter::once(transcript.clone())
176                .chain(sidecars(&transcript))
177                .collect();
178            let bytes: u64 = paths.iter().map(|path| path_size(path)).sum();
179            let last_activity = last_activity(&session);
180
181            let copies = by_session.entry(session.session_id.clone()).or_default();
182            copies.paths.extend(paths);
183            copies.bytes += bytes;
184            copies.undatable |= last_activity.is_none();
185            copies.last_activity = copies.last_activity.max(last_activity);
186        }
187    }
188
189    let now = Utc::now();
190    for (session_id, copies) in by_session {
191        if copies.undatable {
192            plan.undated += 1;
193            continue;
194        }
195        let Some(last_activity) = copies.last_activity else {
196            continue;
197        };
198        if last_activity > cutoff {
199            continue;
200        }
201        plan.targets.push(PurgeTarget {
202            session_id,
203            last_activity,
204            age_days: (now - last_activity).num_days(),
205            paths: copies.paths,
206            bytes: copies.bytes,
207        });
208    }
209
210    plan.targets.sort_by_key(|target| target.last_activity);
211    Ok(plan)
212}

214/// Delete everything the plan names. A path that is already gone is skipped,
215/// not an error.
216pub fn apply(plan: &PurgePlan) -> Result<PurgeReport> {
217    let mut report = PurgeReport::default();
218
219    for target in &plan.targets {
220        report.sessions += 1;
221        for path in &target.paths {
222            let bytes = path_size(path);
223            let removed = if path.is_dir() {
224                std::fs::remove_dir_all(path)
225            } else if path.is_file() {
226                std::fs::remove_file(path)
227            } else {
228                continue;
229            };
230            match removed {
231                Ok(()) => {
232                    report.files += 1;
233                    report.bytes += bytes;
234                }
235                Err(error) => report.failures.push(format!("{}: {error}", path.display())),
236            }
237        }
238    }
239
240    Ok(report)
241}

243/// Open the sync repository, before anything is deleted: a repository that
244/// cannot record the removal must stop the purge while the files still exist.
245pub fn open_sync_repo(repo_root: &Path) -> Result<Box<dyn crate::scm::Scm>> {
246    let repo = crate::scm::open(repo_root)
247        .with_context(|| format!("Failed to open repository at {}", repo_root.display()))?;
248    repo.has_changes()
249        .with_context(|| format!("Cannot read the repository at {}", repo_root.display()))?;
250    Ok(repo)
251}
252
253/// Refuse to purge while the repository holds uncommitted work.
254///
255/// A repo-side transcript that is not committed yet — a push that was declined,
256/// a path an ignore rule covers — would be deleted with no copy in history,
257/// which is the one thing a purge promises not to do.
258pub fn ensure_nothing_uncommitted(repo: &dyn crate::scm::Scm) -> Result<()> {
259    if repo.has_changes()? {
260        bail!(
261            "the sync repository has uncommitted changes; run `claude-code-sync push` \
262             first, so a purge cannot delete a copy that was never committed"
263        );
264    }
265    Ok(())
266}

268/// Stage and commit the removals. Returns whether a commit was made.
269pub fn commit_removals(repo: &dyn crate::scm::Scm, plan: &PurgePlan) -> Result<bool> {
270    repo.stage_all()?;
271    if !repo.has_changes()? {
272        return Ok(false);
273    }
274    repo.commit(&format!(
275        "Purge {} transcripts older than {} days",
276        plan.targets.len(),
277        plan.retention_days
278    ))?;
279    Ok(true)
280}

282/// Session transcripts of a project tree: `<project>/<session>.jsonl` only.
283///
284/// Subagent transcripts live deeper, under `<session>/subagents/`, and are
285/// removed with their parent session rather than on their own age.
286fn transcripts(projects_dir: &Path) -> Vec<PathBuf> {
287    if !projects_dir.is_dir() {
288        return Vec::new();
289    }
290    walkdir::WalkDir::new(projects_dir)
291        .follow_links(false)
292        .min_depth(2)
293        .max_depth(2)
294        .into_iter()
295        .filter_map(|entry| entry.ok())
296        .filter(|entry| entry.file_type().is_file())
297        .map(|entry| entry.path().to_path_buf())
298        .filter(|path| path.extension().and_then(|e| e.to_str()) == Some("jsonl"))
299        .collect()
300}

302/// Files Claude Code keeps beside a transcript and deletes with it: the
303/// session directory (`subagents/`, `tool-results/`) and any set-aside copies.
304fn sidecars(transcript: &Path) -> Vec<PathBuf> {
305    let Some(directory) = transcript.parent() else {
306        return Vec::new();
307    };
308    let Some(stem) = transcript.file_stem().and_then(|s| s.to_str()) else {
309        return Vec::new();
310    };
311
312    let mut found = Vec::new();
313    let session_directory = directory.join(stem);
314    if session_directory.is_dir() {
315        found.push(session_directory);
316    }
317
318    // Every sibling named after this session: `.orphaned-…`, `.superseded-…`,
319    // `.stopoffset`, and whatever Claude Code adds next.
320    let companion = format!("{stem}.");
321    let transcript_name = transcript.file_name();
322    let Ok(entries) = std::fs::read_dir(directory) else {
323        return found;
324    };
325    for entry in entries.filter_map(|entry| entry.ok()) {
326        if Some(entry.file_name().as_os_str()) == transcript_name {
327            continue;
328        }
329        let Some(name) = entry.file_name().to_str().map(str::to_string) else {
330            continue;
331        };
332        if name.starts_with(&companion) {
333            found.push(entry.path());
334        }
335    }
336    found
337}

339/// Timestamp of the session's last message, in UTC.
340fn last_activity(session: &crate::parser::ConversationSession) -> Option<DateTime<Utc>> {
341    let stamp = session.latest_timestamp()?;
342    DateTime::parse_from_rfc3339(stamp)
343        .ok()
344        .map(|dated| dated.with_timezone(&Utc))
345}
346
347fn path_size(path: &Path) -> u64 {
348    if path.is_dir() {
349        return walkdir::WalkDir::new(path)
350            .follow_links(false)
351            .into_iter()
352            .filter_map(|entry| entry.ok())
353            .filter_map(|entry| entry.metadata().ok())
354            .filter(|metadata| metadata.is_file())
355            .map(|metadata| metadata.len())
356            .sum();
357    }
358    std::fs::metadata(path).map(|m| m.len()).unwrap_or(0)
359}

365    fn claude_dir_with(settings: Option<&str>) -> tempfile::TempDir {
366        let dir = tempfile::tempdir().unwrap();
