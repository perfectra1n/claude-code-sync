1//! `.gitattributes` in the sync repository.
2//!
3//! Every machine only ever appends to a transcript, to the prompt history and
4//! to a memory index. Git cannot know that: when two machines both appended
5//! between syncs, it reports a content conflict at the end of the file and the
6//! pull stops with nothing merged. Git's own `union` merge driver keeps both
7//! sides' lines instead, in no meaningful order. claude-code-sync puts them
8//! back in order on the way into `~/.claude`: the prompt history and memory
9//! indexes through `artifacts::union_jsonl` / `artifacts::memory_index`, and
10//! session transcripts through the smart merge (`merge.rs`), which drops
11//! duplicate UUIDs and sorts by timestamp. The next push writes that ordered
12//! copy back to the repository.
13
14use anyhow::{Context, Result};
15use std::fs;
16use std::path::Path;
17
18const SYNC_ATTRIBUTE_RULES: &[&str] = &[
19    "* text=auto eol=lf",
20    "*.pdf binary",
21    "*.bat text eol=crlf",
22    "*.cmd text eol=crlf",
23    "*.jsonl merge=union",
24    "MEMORY.md merge=union",
25];
26
27/// Write the sync rules into the repository's `.gitattributes`.
28///
29/// Returns whether anything was added, so a caller can commit only a real
30/// change.
31pub fn ensure_sync_attributes(repo_path: &Path) -> Result<bool> {
32    let rules: Vec<String> = SYNC_ATTRIBUTE_RULES.iter().map(|s| s.to_string()).collect();
33    ensure_lines(&repo_path.join(".gitattributes"), &rules)
34}
35
36/// Append every line that the file does not already contain, creating it if
37/// needed. Returns whether the file changed.
38pub fn ensure_lines(path: &Path, lines: &[String]) -> Result<bool> {
39    let mut content = if path.exists() {
40        fs::read_to_string(path).with_context(|| format!("Failed to read {}", path.display()))?
41    } else {
42        String::new()
43    };
44
45    let mut added = false;
46    for line in lines {
47        if content.lines().any(|existing| existing == line) {
48            continue;
49        }
50        if !content.is_empty() && !content.ends_with('\n') {
51            content.push('\n');
52        }
53        content.push_str(line);
54        content.push('\n');
55        added = true;
56    }
57
58    if added {
59        fs::write(path, content).with_context(|| format!("Failed to write {}", path.display()))?;
60    }
61
62    Ok(added)
63}

65#[cfg(test)]
66mod tests {
67    use super::*;
68    use tempfile::TempDir;
69
70    #[test]
71    fn the_sync_rules_are_written_once() {
72        let repo = TempDir::new().unwrap();
73
74        assert!(ensure_sync_attributes(repo.path()).unwrap());
75        let written = fs::read_to_string(repo.path().join(".gitattributes")).unwrap();
76        assert!(written.contains("* text=auto eol=lf"));
77        assert!(written.contains("*.jsonl merge=union"));
78        assert!(written.contains("MEMORY.md merge=union"));
79
80        assert!(
81            !ensure_sync_attributes(repo.path()).unwrap(),
82            "a second run has nothing to add"
83        );
84        assert_eq!(
85            fs::read_to_string(repo.path().join(".gitattributes")).unwrap(),
86            written
87        );
88    }
89
90    #[test]
91    fn rules_are_added_beside_what_is_already_there() {
92        let repo = TempDir::new().unwrap();
93        let attributes = repo.path().join(".gitattributes");
94        fs::write(&attributes, "*.png filter=lfs diff=lfs merge=lfs -text").unwrap();
95
96        assert!(ensure_sync_attributes(repo.path()).unwrap());
97
98        let written = fs::read_to_string(&attributes).unwrap();
99        assert!(written.starts_with("*.png filter=lfs"));
100        assert!(written.contains("*.jsonl merge=union"));
101    }
102}
