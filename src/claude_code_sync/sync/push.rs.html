22/// One session's planned copy into the sync repository.
23#[derive(Debug, Clone)]
24pub struct PlannedSessionPush {

56/// Compute a session's destination path relative to the projects directory: a
57/// mapped project goes under its canonical id, otherwise `use_project_name_only`
58/// and the plain encoded path decide, exactly as before. Returns None when the
59/// session lacks the `cwd` needed for project-name mapping.
60fn compute_relative_path(
61    session: &crate::parser::ConversationSession,
62    claude_dir: &Path,
63    filter: &FilterConfig,
64) -> Option<PathBuf> {
65    let full_relative = session
66        .file_path
67        .strip_prefix(claude_dir)
68        .unwrap_or(&session.file_path);
69
70    let mut parts = full_relative.components();
71    let encoded_dir = parts.next()?.as_os_str().to_str()?;
72    if let Some(id) = crate::project_map::canonical_id(&filter.project_map, encoded_dir) {
73        return Some(Path::new(&id).join(parts.as_path()));
74    }
75
76    if filter.use_project_name_only {
77        // The session's own cwd, not the encoded directory: its last segment
78        // is the project's folder name, which a `-` inside that name would
79        // otherwise truncate ("shop-web" -> "web"). The rest of the path is
80        // kept, so a session's subagent transcripts stay under it.
81        let project_name = session.project_name()?;
82        return Some(PathBuf::from(project_name).join(parts.as_path()));
83    }
84
85    Some(full_relative.to_path_buf())
86}

88/// Whether applying this plan entry still has to write the transcript.
89///
90/// An unchanged session already sits in the repository, and copying it again
91/// would rewrite bytes git then has to store for no change: a transcript an
92/// older version wrote is normalized JSON, byte for byte different from the
93/// verbatim copy made today while holding the very same conversation.
94pub fn needs_copy(entry: &PlannedSessionPush, dest_path: &Path) -> bool {
95    if entry.operation != SyncOperation::Unchanged {
96        return true;
97    }
98    !dest_path.exists()
99}

101/// Classify every discovered session against the sync repository's current
102/// contents without writing anything. Sessions are keyed by their identity
103/// (filename stem), so sibling files sharing an interior sessionId — subagent
104/// sidechains, resumed sessions — classify independently (issue #68).
105pub fn plan_push(
106    sessions: &[crate::parser::ConversationSession],
107    claude_dir: &Path,
108    projects_dir: &Path,
109    filter: &FilterConfig,
110) -> Result<PushPlan> {
111    let existing_sessions = if projects_dir.exists() {
112        discover_sessions(projects_dir, filter)?
113    } else {
114        Vec::new()
115    };
116    let existing_map: HashMap<_, _> = existing_sessions
117        .iter()
118        .map(|s| (s.session_id.clone(), s))
119        .collect();
120
121    let mut plan = PushPlan::default();
122
123    for (session_index, session) in sessions.iter().enumerate() {
124        let relative_path = match compute_relative_path(session, claude_dir, filter) {
125            Some(path) => path,
126            None => {
127                plan.skipped_no_cwd += 1;
128                log::debug!("Skipping session {} (no cwd)", session.session_id);
129                continue;
130            }
131        };
132
133        let operation = if let Some(existing) = existing_map.get(&session.session_id) {
134            if existing.content_hash() == session.content_hash() {
135                plan.unchanged += 1;
136                SyncOperation::Unchanged
137            } else {
138                plan.modified += 1;
139                SyncOperation::Modified
140            }
141        } else {
142            plan.added += 1;
143            SyncOperation::Added
144        };
145
146        plan.entries.push(PlannedSessionPush {
147            session_index,
148            relative_path,
149            operation,
150        });
151    }
152
153    Ok(plan)
154}

156/// Push local Claude Code history to sync repository
157#[allow(clippy::too_many_arguments)]
158pub fn push_history(
159    commit_message: Option<&str>,
160    push_remote: bool,
161    branch: Option<&str>,
162    exclude_attachments: bool,
163    interactive: bool,
164    verbosity: crate::VerbosityLevel,
165    skip_artifact_paths: &HashSet<(CategoryId, PathBuf)>,
166    resurrect: &[String],
167) -> Result<PushReport> {
168    use crate::VerbosityLevel;
169
170    if verbosity != VerbosityLevel::Quiet {
171        println!("{}", "Pushing Claude Code history...".cyan().bold());
172    }
173
174    let state = SyncState::load()?;
175    let repo = scm::open(&state.sync_repo_path)?;
176    let merge_is_unfinished = repo.has_unfinished_merge();
177    if merge_is_unfinished {
178        anyhow::bail!(
179            "The sync repository has a merge an interrupted pull left unfinished. \
180             Run `claude-code-sync pull` to settle or undo it first."
181        );
182    }
183    let mut filter = FilterConfig::load()?;
184
185    // 5ff1d62 escape hatch: drop the listed paths from the local tracked
186    // record BEFORE the push runs, so the push's guard at engine.rs:711-735
187    // sees no tracked entry and publishes the file as a fresh `Added`.
188    // `engine::prepare_resurrection` returns Err if any path is not in the
189    // current record — defense in depth against a typo in the flag value
190    // or a path that was never held back. The user-facing discovery is
191    // `claude-code-sync status --held-back`.
192    if !resurrect.is_empty() {
193        let claude_dir = claude_projects_dir()?;
194        crate::artifacts::engine::prepare_resurrection(
195            &claude_dir,
196            &state.sync_repo_path,
197            resurrect,
198        )?;
199    }
200
201    // Override exclude_attachments if specified in command
202    if exclude_attachments {
203        filter.exclude_attachments = true;
204    }
205
206    // A `kept_local` (or `kept_local_delete`) file from the previous
207    // pull is held back per-file: the push's `push_artifacts` consults
208    // the `skip_artifact_paths` set and does not touch the repo copy
209    // for any entry. Other files in the same category (and other
210    // categories entirely) are unaffected — the held file is one
211    // surgical skip, not a category-wide hold.
212
213    // Set up LFS if enabled
214    if filter.enable_lfs {
215        if verbosity != VerbosityLevel::Quiet {
216            println!("  {} Git LFS...", "Configuring".cyan());
217        }
218        scm::lfs::setup(&state.sync_repo_path, &filter.lfs_patterns)
219            .context("Failed to set up Git LFS")?;
220    }
221
222    crate::scm::attributes::ensure_sync_attributes(&state.sync_repo_path)?;
223
224    let claude_dir = claude_projects_dir()?;
225
226    // Get the current branch name for operation record
227    let branch_name = branch
228        .map(|s| s.to_string())
229        .or_else(|| repo.current_branch().ok())
230        .unwrap_or_else(|| "main".to_string());
231
232    // Discover all sessions
233    println!("  {} conversation sessions...", "Discovering".cyan());
234    let sessions = discover_sessions(&claude_dir, &filter)?;
235    println!("  {} {} sessions", "Found".green(), sessions.len());
236
237    // Check for project name collisions when using project-name-only mode
238    if filter.use_project_name_only {
239        let collisions = find_colliding_projects(&claude_dir);
240        if !collisions.is_empty() {
241            println!();
242            println!(
243                "{}",
244                "Warning: Multiple projects map to the same name:"
245                    .yellow()
246                    .bold()
247            );
248            for (name, paths) in &collisions {
249                println!("  {} -> {} locations:", name.cyan(), paths.len());
250                for path in paths.iter().take(3) {
251                    let display_path = path
252                        .file_name()
253                        .and_then(|n| n.to_str())
254                        .unwrap_or("unknown");
255                    println!("    - {}", display_path);
256                }
257                if paths.len() > 3 {
258                    println!("    ... and {} more", paths.len() - 3);
259                }
260            }
261            println!();
262            println!(
263                "{}",
264                "Sessions from colliding projects will be merged into the same directory.".yellow()
265            );
266            println!();
267        }
268    }
269
270    // ============================================================================
271    // COPY SESSIONS AND TRACK CHANGES
272    // ============================================================================
273    let projects_dir = state.sync_repo_path.join(&filter.sync_subdirectory);
274    fs::create_dir_all(&projects_dir)?;
275
276    // Classify every session against the sync repo, then apply the plan
277    println!("  {} sessions to sync repository...", "Copying".cyan());
278    let plan = plan_push(&sessions, &claude_dir, &projects_dir, &filter)?;
279    let added_count = plan.added;
280    let modified_count = plan.modified;
281    let unchanged_count = plan.unchanged;
282    let skipped_no_cwd = plan.skipped_no_cwd;
283
284    // Track pushed conversations for operation record
285    let mut pushed_conversations: Vec<ConversationSummary> = Vec::new();
286
287    for entry in &plan.entries {
288        let session = &sessions[entry.session_index];
289        let dest_path = projects_dir.join(&entry.relative_path);
290
291        if needs_copy(entry, &dest_path) {
292            session.copy_to(&dest_path)?;
293        }
294
295        // Track this session in pushed conversations
296        let relative_path_str = entry.relative_path.to_string_lossy().to_string();
297        match ConversationSummary::new(
298            session.session_id.clone(),
299            relative_path_str.clone(),
300            session.latest_timestamp().map(str::to_string),
301            session.message_count(),
302            entry.operation,
303        ) {
304            Ok(summary) => pushed_conversations.push(summary),
305            Err(e) => log::warn!("Failed to create summary for {}: {}", relative_path_str, e),
306        }
307    }
308
309    // ============================================================================
310    // COPY ARTIFACTS (settings, skills, agents, ...) AND WRITE IGNORE GUARD
311    // ============================================================================
312    let artifact_report = crate::artifacts::engine::push_artifacts(
313        &claude_home_dir()?,
314        &state.sync_repo_path,
315        &filter,
316        skip_artifact_paths,
317    )?;
318    crate::artifacts::engine::ensure_ignore_files(&state.sync_repo_path, filter.backend()?)?;
319
320    // ============================================================================
321    // SHOW SUMMARY AND INTERACTIVE CONFIRMATION
322    // ============================================================================
323    if verbosity != VerbosityLevel::Quiet {
324        println!();
325        println!("{}", "Push Summary:".bold().cyan());
326        println!("  {} Added: {}", "•".green(), added_count);
327        println!("  {} Modified: {}", "•".yellow(), modified_count);
328        println!("  {} Unchanged: {}", "•".dimmed(), unchanged_count);
329        let total_with_cwd = sessions.len().saturating_sub(skipped_no_cwd);
330        println!("  {} Skipped (no cwd): {}", "•".dimmed(), skipped_no_cwd);
331        println!(
332            "  {} Sessions (with project context): {}",
333            "•".cyan(),
334            total_with_cwd
335        );
336        if !artifact_report.counts.is_empty() {
337            // "skipped" (unreadable / over-size / refused), NOT "held
338            // back" — the pull's banner owns that word for the
339            // per-file skip set, and files the skip set held count as
340            // unchanged here. Two labels, two sets.
341            let held_back_remote_lost = artifact_report.total_held_back_remote_lost();
342            if held_back_remote_lost > 0 {
343                println!(
344                    "  {} Artifacts: {} added, {} modified, {} deleted, {} unchanged, {} skipped, {} held back (remote lost)",
345                    "•".cyan(),
346                    artifact_report.total_added(),
347                    artifact_report.total_modified(),
348                    artifact_report.total_deleted(),
349                    artifact_report.total_unchanged(),
350                    artifact_report.total_skipped(),
351                    held_back_remote_lost,
352                );
353                // List the actual file paths so the user can target them
354                // with `push --resurrect <path>`. Capped to 20 with a
355                // trailing "... and N more" to match the verbose list
356                // style used elsewhere in the push summary.
357                let held_back_list: Vec<&PathBuf> = artifact_report
358                    .held_back_paths()
359                    .map(|(_, path, _)| path)
360                    .collect();
361                let shown = held_back_list.len().min(20);
362                if shown > 0 {
363                    println!(
364                        "    {} Files the remote lost (use `claude-code-sync push --resurrect <path>` to override):",
365                        "→".cyan()
366                    );
367                    for path in &held_back_list[..shown] {
368                        println!("      {}", path.display());
369                    }
370                    if held_back_list.len() > shown {
371                        println!("      ... and {} more", held_back_list.len() - shown);
372                    }
373                }
374            } else {
375                println!(
376                    "  {} Artifacts: {} added, {} modified, {} deleted, {} unchanged, {} skipped",
377                    "•".cyan(),
378                    artifact_report.total_added(),
379                    artifact_report.total_modified(),
380                    artifact_report.total_deleted(),
381                    artifact_report.total_unchanged(),
382                    artifact_report.total_skipped(),
383                );
384            }
385        }
386        println!();
387    }
388
389    // Show detailed file list in verbose mode
390    if verbosity == VerbosityLevel::Verbose {
391        println!("{}", "Files to be pushed:".bold());
392        for (idx, entry) in plan.entries.iter().enumerate().take(20) {
393            let status = match entry.operation {
394                SyncOperation::Unchanged => "unchanged".dimmed(),
395                SyncOperation::Modified => "modified".yellow(),
396                _ => "new".green(),
397            };
398
399            println!(
400                "  {}. {} [{}]",
401                idx + 1,
402                entry.relative_path.display(),
403                status
404            );
405        }
406        if plan.entries.len() > 20 {
407            println!("  ... and {} more", plan.entries.len() - 20);
408        }
409        println!();
410    }
411
412    // Interactive confirmation
413    if interactive && interactive_conflict::is_interactive() {
414        let confirm = Confirm::new("Push?")
415            .with_default(true)
416            .prompt()
417            .context("Failed to get confirmation")?;
418
419        if !confirm {
420            println!("\n{}", "Push cancelled.".yellow());
421            return Ok(PushReport::default());
422        }
423    }
424
425    // ============================================================================
426    // COMMIT AND PUSH CHANGES
427    // ============================================================================
428    repo.stage_all()?;
429
430    let has_changes = repo.has_changes()?;
431    if has_changes {
432        // Get the current commit hash before making any changes
433        // This allows us to undo the push later by resetting to this commit
434        // Note: We don't create file snapshots for push - git already has history!
435        // Undo push simply does `git reset` to this commit.
436        // On a brand new repo with no commits, this will be None (no undo available for first push)
437        let commit_before_push = repo.current_commit_hash().ok();
438
439        if let Some(ref hash) = commit_before_push {
440            if verbosity != VerbosityLevel::Quiet {
441                println!("  {} Recorded commit {} for undo", "✓".green(), &hash[..8]);
442            }
443        } else if verbosity != VerbosityLevel::Quiet {
444            println!(
445                "  {} First push - no previous commit to undo to",
446                "ℹ".cyan()
447            );
448        }
449
450        let default_message = format!(
451            "Sync {} sessions at {}",
452            sessions.len(),
453            chrono::Utc::now().format("%Y-%m-%d %H:%M:%S UTC")
454        );
455        let message = commit_message.unwrap_or(&default_message);
456
457        println!("  {} changes...", "Committing".cyan());
458        repo.commit(message)?;
459        println!("  {} Committed: {}", "✓".green(), message);
460
461        // Push to remote if configured
462        if push_remote && state.has_remote {
463            println!("  {} to remote...", "Pushing".cyan());
464
465            match repo.push("origin", &branch_name) {
466                Ok(_) => println!("  {} Pushed to origin/{}", "✓".green(), branch_name),
467                Err(e) => log::warn!("Failed to push: {}", e),
468            }
469        }
470
471        // ============================================================================
472        // CREATE AND SAVE OPERATION RECORD
473        // ============================================================================
474        let mut operation_record = OperationRecord::new(
475            OperationType::Push,
476            Some(branch_name.clone()),
477            pushed_conversations.clone(),
478        );
479
480        // Store commit hash for undo (no file snapshot needed - git has history)
481        // On first push (no prior commits), this will be None
482        operation_record.commit_hash = commit_before_push;
483        operation_record.artifact_counts = artifact_report.counts.clone();
484
485        // Load operation history and add this operation
486        let mut history = match OperationHistory::load() {
487            Ok(h) => h,
488            Err(e) => {
489                log::warn!("Failed to load operation history: {}", e);
490                log::info!("Creating new history...");
491                OperationHistory::default()
492            }
493        };
494
495        if let Err(e) = history.add_operation(operation_record) {
496            log::warn!("Failed to save operation to history: {}", e);
497            log::info!("Push completed successfully, but history was not updated.");
498        }
499    } else {
500        println!("  {} No changes to commit", "Note:".yellow());
501    }
502
503    // ============================================================================
504    // DISPLAY SUMMARY TO USER
505    // ============================================================================
506    println!("\n{}", "=== Push Summary ===".bold().cyan());
507
508    // Show operation statistics
509    let stats_msg = format!(
510        "  {} Added    {} Modified    {} Unchanged",
511        format!("{added_count}").green(),
512        format!("{modified_count}").cyan(),
513        format!("{unchanged_count}").dimmed(),
514    );
515    println!("{stats_msg}");
516    println!();
517
518    // Group conversations by project (top-level directory)
519    let mut by_project: HashMap<String, Vec<&ConversationSummary>> = HashMap::new();
520    for conv in &pushed_conversations {
521        // Skip unchanged conversations in detailed output
522        if conv.operation == SyncOperation::Unchanged {
523            continue;
524        }
525
526        let project = conv
527            .project_path
528            .split('/')
529            .next()
530            .unwrap_or("unknown")
531            .to_string();
532        by_project.entry(project).or_default().push(conv);
533    }
534
535    // Display conversations grouped by project
536    if !by_project.is_empty() {
537        println!("{}", "Pushed Conversations:".bold());
538
539        let mut projects: Vec<_> = by_project.keys().collect();
540        projects.sort();
541
542        for project in projects {
543            let conversations = &by_project[project];
544            println!("\n  {} {}/", "Project:".bold(), project.cyan());
545
546            for conv in conversations.iter().take(MAX_CONVERSATIONS_TO_DISPLAY) {
547                let operation_str = match conv.operation {
548                    SyncOperation::Added => "ADD".green(),
549                    SyncOperation::Modified => "MOD".cyan(),
550                    SyncOperation::Conflict => "CONFLICT".yellow(),
551                    SyncOperation::Unchanged => "---".dimmed(),
552                };
553
554                let timestamp_str = conv
555                    .timestamp
556                    .as_ref()
557                    .and_then(|t| {
558                        // Extract just the date portion for compact display
559                        t.split('T').next()
560                    })
561                    .unwrap_or("unknown");
562
563                println!(
564                    "    {} {} ({}msg, {})",
565                    operation_str,
566                    conv.project_path,
567                    conv.message_count,
568                    timestamp_str.dimmed()
569                );
570            }
571
572            if conversations.len() > MAX_CONVERSATIONS_TO_DISPLAY {
573                println!(
574                    "    {} ... and {} more conversations",
575                    "...".dimmed(),
576                    conversations.len() - MAX_CONVERSATIONS_TO_DISPLAY
577                );
578            }
579        }
580    }
581
582    super::print_artifact_changes(&artifact_report);
583
584    if verbosity == VerbosityLevel::Quiet {
585        println!("Push complete");
586    } else {
587        println!("\n{}", "Push complete!".green().bold());
588    }
589
590    // Clean up old snapshots automatically
591    if let Err(e) = crate::undo::cleanup_old_snapshots(None, false) {
592        log::warn!("Failed to cleanup old snapshots: {}", e);
593    }
594
595    Ok(PushReport {
596        added: added_count,
597        modified: modified_count,
598        unchanged: unchanged_count,
599        skipped_no_cwd,
600        artifacts: artifact_report,
601    })
