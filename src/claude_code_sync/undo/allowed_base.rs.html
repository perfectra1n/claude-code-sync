1//! The one allowed-base computation every snapshot-driven write is
2//! sandboxed to (files AND artifact records): the given directory when
3//! present, the home directory otherwise — canonicalized, so symlinked
4//! spellings cannot slip a path past the boundary.
5
6use anyhow::{Context, Result};
7use std::path::{Path, PathBuf};
8
9/// Resolve and canonicalize the base directory every restored path must
10/// live under.
11pub(crate) fn allowed_base(allowed: Option<&Path>) -> Result<PathBuf> {
12    match allowed {
13        Some(base) => base
14            .canonicalize()
15            .with_context(|| format!("Failed to canonicalize base directory: {}", base.display())),
16        None => dirs::home_dir()
17            .context("Failed to get home directory")?
18            .canonicalize()
19            .context("Failed to canonicalize home directory"),
20    }
21}
