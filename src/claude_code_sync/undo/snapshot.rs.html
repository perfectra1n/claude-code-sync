11/// Represents a snapshot of conversation files at a point in time
12///
13/// Snapshots are created before each sync operation to enable undo functionality.
14/// They capture the complete state of all conversation files that might be affected.
15///
16/// ## Differential Snapshots
17///
18/// To save disk space, snapshots can be differential - only storing files that changed
19/// since the previous snapshot. This is controlled by the `base_snapshot_id` field:
20/// - `None`: Full snapshot containing all files
21/// - `Some(id)`: Differential snapshot containing only changes since base snapshot
22///
23/// When restoring a differential snapshot, we recursively load the chain of base
24/// snapshots to reconstruct the full state.
25#[derive(Debug, Clone, Serialize, Deserialize)]
26pub struct Snapshot {
27    /// Unique identifier for this snapshot
28    pub snapshot_id: String,
29
30    /// When this snapshot was created
31    pub timestamp: chrono::DateTime<chrono::Utc>,
32
33    /// Type of operation this snapshot was created for
34    pub operation_type: OperationType,
35
36    /// Git commit hash before the operation (for push operations)
37    #[serde(skip_serializing_if = "Option::is_none")]
38    pub git_commit_hash: Option<String>,
39
40    /// Mapping of file paths (relative to Claude projects dir) to their content
41    ///
42    /// We store the raw bytes to preserve exact file state including encoding.
43    /// The HashMap key is a string path for JSON serialization compatibility.
44    ///
45    /// For differential snapshots, only contains files that changed/were added.
46    #[serde(with = "base64_map")]
47    pub files: HashMap<String, Vec<u8>>,
48
49    /// Git branch name at the time of snapshot
50    #[serde(skip_serializing_if = "Option::is_none")]
51    pub branch: Option<String>,
52
53    /// Base snapshot ID for differential snapshots
54    ///
55    /// If present, this snapshot only contains changes relative to the base.
56    /// The full state can be reconstructed by loading the chain of snapshots.
57    #[serde(skip_serializing_if = "Option::is_none")]
58    pub base_snapshot_id: Option<String>,
59
60    /// Files that were deleted since the base snapshot
61    ///
62    /// Only populated for differential snapshots. Lists file paths that existed
63    /// in the base snapshot but should be removed when restoring this snapshot.
64    #[serde(default, skip_serializing_if = "Vec::is_empty")]
65    pub deleted_files: Vec<String>,
66
67    /// Artifact record files (bases, tracked) this snapshot carries, at
68    /// their pull-time spellings. They hold one entry per sync repository
69    /// and are restored surgically — this repository's entry only, never
70    /// the whole file — so undo needs to know exactly which keys are
71    /// records; matching by file name alone could be hijacked by a synced
72    /// artifact that happens to share the name. Absent on snapshots
73    /// written before the field existed.
74    #[serde(default, skip_serializing_if = "Vec::is_empty")]
75    pub record_files: Vec<String>,
76
77    /// Artifact record files the apply of this snapshot's pull may create.
78    /// Undo forgets this repository's entry instead of deleting the shared
79    /// file.
80    #[serde(default, skip_serializing_if = "Vec::is_empty")]
81    pub created_record_files: Vec<String>,
82
83    /// Repo-relative keys the pull's apply writes or prunes in the BASES
84    /// record, as the plan computed them. Undo restores the PRE-PULL value
85    /// of exactly these keys — never the whole entry, which would erase
86    /// entries a push recorded between the pull and its undo. `None` on
87    /// snapshots written before the field existed (undo then falls back
88    /// to the whole-entry restore those snapshots were written with);
89    /// `Some(empty)` is a CURRENT snapshot whose pull owned no key — the
90    /// restore is then a no-op, never a rewind.
91    #[serde(default, skip_serializing_if = "Option::is_none")]
92    pub record_touched_bases: Option<Vec<String>>,
93
94    /// The TRACKED record's counterpart of `record_touched_bases`.
95    #[serde(default, skip_serializing_if = "Option::is_none")]
96    pub record_touched_tracked: Option<Vec<String>>,
97
98    /// Files the pull intended to modify whose pre-pull backup FAILED
99    /// (unreadable at snapshot time). The apply reads this and skips
100    /// them: the pre-PR contract — no file is modified without a backup
101    /// — must survive the never-fatal snapshot, or a transient read
102    /// error at snapshot time that recovers before the apply turns into
103    /// a modification undo cannot restore.
104    #[serde(default, skip_serializing_if = "Vec::is_empty")]
105    pub unreadable_files: Vec<String>,
106
107    /// Record-shaped blobs `promote_legacy_records` stripped from
108    /// `files` (unverifiable placement). NOT serialized: the on-disk
109    /// snapshot keeps them inside `files` untouched — this stash exists
110    /// only so the undo's pinned re-save can put them BACK before
111    /// overwriting the sole copy (the warning tells the user to recover
112    /// them by hand from that file; persisting the stripped map would
113    /// destroy what it points at).
114    #[serde(skip, default)]
115    pub stripped_blobs: Vec<(String, Vec<u8>)>,
116
117    /// Set when an undo KEPT this snapshot (record-surgery warnings): it
118    /// holds the sole copy of the unrestored record entries, and the
119    /// regular snapshot cleanup must never delete it from under the
120    /// user. Cleared by hand once the entries are recovered.
121    #[serde(default, skip_serializing_if = "is_false")]
122    pub pinned: bool,
123}

127    !*value

130/// Custom serialization for `HashMap<String, Vec<u8>>` using base64 encoding
131///
132/// This is necessary because JSON doesn't natively support binary data,
133/// so we encode file contents as base64 strings for storage.
134mod base64_map {
135    use base64::engine::general_purpose::STANDARD;
136    use base64::Engine;
137    use serde::{Deserialize, Deserializer, Serialize, Serializer};
138    use std::collections::HashMap;
139
140    pub fn serialize<S>(map: &HashMap<String, Vec<u8>>, serializer: S) -> Result<S::Ok, S::Error>
141    where
142        S: Serializer,
143    {
144        let base64_map: HashMap<String, String> = map
145            .iter()
146            .map(|(k, v)| (k.clone(), STANDARD.encode(v)))
147            .collect();
148        base64_map.serialize(serializer)
149    }
150
151    pub fn deserialize<'de, D>(deserializer: D) -> Result<HashMap<String, Vec<u8>>, D::Error>
152    where
153        D: Deserializer<'de>,
154    {
155        let base64_map: HashMap<String, String> = HashMap::deserialize(deserializer)?;
156        base64_map
157            .into_iter()
158            .map(|(k, v)| {
159                STANDARD
160                    .decode(&v)
161                    .map(|bytes| (k, bytes))
162                    .map_err(serde::de::Error::custom)
163            })
164            .collect()
165    }
166}

168impl Snapshot {
169    /// Declare the shared record files a LEGACY snapshot (written before
170    /// the declaration fields, e.g. by a previous release still inside
171    /// the retention window) carries among `files` without declaring
172    /// them — so both undo and its preview treat them surgically instead
173    /// of the whole-file generic restore, which would clobber every
174    /// other repository's entries.
175    ///
176    /// Recognition must not be by file name alone: a synced artifact can
177    /// share the name (the decoy test pins that contract). A real record
178    /// sits at the CLAUDE ROOT with every other snapshotted file under
179    /// that same root, and it PARSES — a JSON object keyed by repository
180    /// roots. A name-matching key counts as a legacy record only when
181    /// its parent directory is an ancestor of every other snapshot key
182    /// AND its bytes parse as a record. A decoy under skills/docs/ fails
183    /// the first (nothing else lives under it), a text decoy at the root
184    /// fails the second, and a snapshot of records alone (no artifacts
185    /// to hijack) passes trivially.
186    /// Returns whether anything was STRIPPED (a record-looking blob that
187    /// could not be safely classified): the caller must warn and pin —
188    /// the stripped bytes are the sole copy and are recoverable by hand.
189    pub fn promote_legacy_records(&mut self) -> bool {
190        if !self.record_files.is_empty() || !self.created_record_files.is_empty() {
191            return false;
192        }
193        let is_record_name = |key: &str| {
194            let path = Path::new(key);
195            crate::artifacts::bases::is_record_path(path)
196                || crate::artifacts::tracked::is_record_path(path)
197        };
198        let parses_as_record = |key: &str| {
199            use crate::artifacts::repo_record::RepoRecord;
200            let bytes = &self.files[key];
201            RepoRecord::<crate::artifacts::tracked::TrackedPaths>::from_bytes_strict(bytes)
202                .is_some()
203                || RepoRecord::<crate::artifacts::bases::BaseHashes>::from_bytes_strict(bytes)
204                    .is_some()
205        };
206        let others: Vec<&String> = self
207            .files
208            .keys()
209            .filter(|key| !is_record_name(key))
210            .collect();
211        let mut legacy_records = Vec::new();
212        let mut stripped = Vec::new();
213        for key in self.files.keys() {
214            if !is_record_name(key) || !parses_as_record(key) {
215                // Not a record at all (a same-named artifact decoy stays
216                // an ordinary file), or one whose shape disqualifies it.
217                continue;
218            }
219            let ancestor_of_all = Path::new(key).parent().is_some_and(|parent| {
220                others
221                    .iter()
222                    .all(|other| Path::new(other.as_str()).starts_with(parent))
223            });
224            if ancestor_of_all {
225                legacy_records.push(key.clone());
226            } else {
227                // Record-shaped but unverifiable placement: restoring it
228                // WHOLESALE would clobber every other repository's entries
229                // (the exact loss the surgery exists to prevent), and
230                // surgically is impossible without a trusted directory.
231                // Strip it from the generic restore entirely — the bytes
232                // are recoverable by hand from the kept file, and stashed
233                // here so the pinned re-save puts them back.
234                if let Some(bytes) = self.files.get(key) {
235                    self.stripped_blobs.push((key.clone(), bytes.clone()));
236                }
237                stripped.push(key.clone());
238            }
239        }
240        if !legacy_records.is_empty() {
241            log::warn!(
242                "Legacy snapshot carries {} undeclared shared record file(s); \
243                 restoring their entries surgically",
244                legacy_records.len()
245            );
246            self.record_files.extend(legacy_records);
247        }
248        let stripped_anything = !stripped.is_empty();
249        for key in stripped {
250            self.files.remove(&key);
251        }
252        stripped_anything
253    }

255    /// Declare the artifact record files a pull snapshot carries (or may
256    /// create), so undo can restore this repository's entries surgically
257    /// instead of the whole shared files. The rules live on the plan
258    /// (`carries_*_record` / `may_create_*_record`) — the same ones
259    /// `paths_to_snapshot` uses, so carrier and declaration cannot drift.
260    pub fn attach_record_bookkeeping(
261        &mut self,
262        plan: &crate::artifacts::engine::PullPlan,
263        claude_dir: &Path,
264        interactive: bool,
265    ) {
266        let bases_record = crate::artifacts::bases::record_path(claude_dir);
267        if plan.carries_bases_record(interactive) {
268            self.record_files
269                .push(bases_record.to_string_lossy().to_string());
270        } else if plan.may_create_bases_record(interactive) {
271            self.created_record_files
272                .push(bases_record.to_string_lossy().to_string());
273        }
274        let tracked_record = crate::artifacts::tracked::record_path(claude_dir);
275        if plan.carries_tracked_record() {
276            self.record_files
277                .push(tracked_record.to_string_lossy().to_string());
278        } else if plan.may_create_tracked_record() {
279            self.created_record_files
280                .push(tracked_record.to_string_lossy().to_string());
281        }
282        // Which keys the apply owns in each record: the undo restores the
283        // pre-pull value of exactly these, so entries a later push recorded
284        // survive it (see `record_touched_bases`). Always set here — even
285        // empty — so the undo can tell a CURRENT snapshot (per-key restore,
286        // possibly of zero keys) from a pre-field one (whole-entry restore).
287        self.record_touched_bases = Some(plan.touched_base_keys());
288        self.record_touched_tracked = Some(plan.touched_tracked_keys.clone());
289    }

291    /// Create a new snapshot from a set of file paths
292    ///
293    /// # Arguments
294    /// * `operation_type` - Type of operation this snapshot is for
295    /// * `file_paths` - Iterator of file paths to include in snapshot
296    /// * `commit_hash` - Optional git commit hash to store in the snapshot
297    ///
298    /// # Returns
299    /// A new Snapshot instance with all file contents captured
300    pub fn create<P, I>(
301        operation_type: OperationType,
302        file_paths: I,
303        commit_hash: Option<&str>,
304    ) -> Result<Self>
305    where
306        P: AsRef<Path>,
307        I: IntoIterator<Item = P>,
308    {
309        let snapshot_id = Uuid::new_v4().to_string();
310        let timestamp = chrono::Utc::now();
311        let mut files = HashMap::new();
312        let mut unreadable_files = Vec::new();
313
314        // Capture current state of all specified files
315        for path in file_paths {
316            let path = path.as_ref();
317
318            // Use direct read instead of checking existence first to avoid TOCTOU
319            match fs::read(path) {
320                Ok(content) => {
321                    // Store with path as string for JSON compatibility
322                    files.insert(path.to_string_lossy().to_string(), content);
323                }
324                Err(e) if e.kind() == std::io::ErrorKind::NotFound => {
325                    // File doesn't exist - this is expected in some cases (e.g., deleted files)
326                    // Just skip it without error
327                    continue;
328                }
329                Err(e) => {
330                    // Never fatal — the same rule every pull arm follows:
331                    // one unreadable file must not block the whole sync.
332                    // But the file is RECORDED: the apply consults this
333                    // list and refuses to modify a file it could not
334                    // back up (the pre-PR guarantee — nothing is modified
335                    // without a pre-pull copy — survives the never-fatal
336                    // snapshot; a transient permission error that recovers
337                    // before the apply must not become an unbackupable
338                    // modification).
339                    log::warn!(
340                        "Not snapshotting {} (unreadable: {e}); the pull will not modify it",
341                        path.display()
342                    );
343                    unreadable_files.push(path.to_string_lossy().to_string());
344                    continue;
345                }
346            }
347        }
348
349        Ok(Snapshot {
350            snapshot_id,
351            timestamp,
352            operation_type,
353            git_commit_hash: commit_hash.map(|s| s.to_string()),
354            files,
355            branch: None,
356            base_snapshot_id: None,
357            deleted_files: Vec::new(),
358            record_files: Vec::new(),
359            created_record_files: Vec::new(),
360            record_touched_bases: None,
361            record_touched_tracked: None,
362            unreadable_files,
363            stripped_blobs: Vec::new(),
364            pinned: false,
365        })
366    }

368    /// Save this snapshot to disk
369    ///
370    /// Snapshots are saved to `~/.claude-code-sync/snapshots/{snapshot_id}.json`
371    ///
372    /// # Arguments
373    /// * `custom_path` - Optional custom directory to save snapshot (for testing)
374    pub fn save_to_disk(&self, custom_path: Option<&Path>) -> Result<PathBuf> {
375        let snapshot_dir = if let Some(path) = custom_path {
376            path.to_path_buf()
377        } else {
378            Self::snapshots_dir()?
379        };
380
381        // Ensure snapshots directory exists
382        fs::create_dir_all(&snapshot_dir).with_context(|| {
383            format!(
384                "Failed to create snapshots directory: {}",
385                snapshot_dir.display()
386            )
387        })?;
388
389        let snapshot_path = snapshot_dir.join(format!("{}.json", self.snapshot_id));
390
391        let json =
392            serde_json::to_string_pretty(self).context("Failed to serialize snapshot to JSON")?;
393
394        // Same-directory temp + atomic rename (see the artifacts engine's
395        // `write_atomic`): a re-save can overwrite the ONLY pre-pull backup
396        // long after the files it describes are already gone (the undo-delete
397        // narrowing after an apply, a pin during an undo). A truncate-in-place
398        // write that fails midway would corrupt that sole copy in place, and a
399        // crash mid-write leaves the same corruption — the temp file keeps
400        // the original intact on every failure path.
401        let tmp = tempfile::NamedTempFile::new_in(&snapshot_dir).with_context(|| {
402            format!(
403                "Failed to create temp file in snapshots directory: {}",
404                snapshot_dir.display()
405            )
406        })?;
407        fs::write(tmp.path(), &json).with_context(|| {
408            format!(
409                "Failed to write snapshot to disk: {}",
410                snapshot_path.display()
411            )
412        })?;
413        // Flush before the rename (see RepoRecord::write): a pinned
414        // snapshot is the SOLE copy of record entries a warned undo could
415        // not restore — a rename landing before the data blocks would
416        // turn a power loss into an empty sole copy.
417        tmp.as_file().sync_all().with_context(|| {
418            format!(
419                "Failed to flush snapshot to disk: {}",
420                snapshot_path.display()
421            )
422        })?;
423        tmp.persist(&snapshot_path).with_context(|| {
424            format!(
425                "Failed to persist snapshot to disk: {}",
426                snapshot_path.display()
427            )
428        })?;
429
430        // Log snapshot size information (to file only, UI output is handled by caller)
431        let size_mb = json.len() as f64 / (1024.0 * 1024.0);
432        let snapshot_type = if self.base_snapshot_id.is_some() {
433            "differential"
434        } else {
435            "full"
436        };
437
438        log::debug!(
439            "Created {} snapshot: {} ({:.1} MB, {} files)",
440            snapshot_type,
441            self.snapshot_id,
442            size_mb,
443            self.files.len()
444        );
445
446        if size_mb > 100.0 {
447            log::debug!("Large snapshot size - consider cleaning up old conversation files");
448        }
449
450        Ok(snapshot_path)
451    }

453    /// Load a snapshot from disk
454    ///
455    /// # Arguments
456    /// * `snapshot_path` - Path to the snapshot JSON file
457    ///
458    /// # Behavior
459    /// - Logs snapshot size information
460    /// - Warns if snapshot is unusually large (>100MB) but still loads it
461    /// - Snapshots are critical for undo functionality and should never be skipped
462    ///
463    /// # Errors
464    /// Returns error only if file cannot be read or parsed
465    pub fn load_from_disk<P: AsRef<Path>>(snapshot_path: P) -> Result<Self> {
466        let snapshot_path = snapshot_path.as_ref();
467
468        // Get file size for logging
469        let metadata = fs::metadata(snapshot_path).with_context(|| {
470            format!(
471                "Failed to read snapshot file metadata: {}",
472                snapshot_path.display()
473            )
474        })?;
475
476        let size_mb = metadata.len() as f64 / (1024.0 * 1024.0);
477
478        // Log size information - always show this for visibility
479        if size_mb > 100.0 {
480            println!(
481                "    {} Loading large snapshot: {} ({:.1} MB) - This may take a moment...",
482                "⚠".yellow(),
483                snapshot_path.file_name().unwrap().to_string_lossy().cyan(),
484                size_mb
485            );
486        } else {
487            println!(
488                "    {} snapshot: {} ({:.1} MB)",
489                "Loading".dimmed(),
490                snapshot_path.file_name().unwrap().to_string_lossy().cyan(),
491                size_mb
492            );
493        }
494
495        let content = fs::read_to_string(snapshot_path).with_context(|| {
496            format!("Failed to read snapshot file: {}", snapshot_path.display())
497        })?;
498
499        let snapshot: Snapshot = serde_json::from_str(&content).with_context(|| {
500            format!("Failed to parse snapshot JSON: {}", snapshot_path.display())
501        })?;
502
503        // Log file count information
504        println!("    {} {} files", "Contains".dimmed(), snapshot.files.len());
505
506        // Warn if unusually large number of files
507        if snapshot.files.len() > 1000 {
508            println!(
509                "    {} Large number of files - this is a full (non-differential) snapshot",
510                "Note:".yellow()
511            );
512        }
513
514        Ok(snapshot)
515    }
