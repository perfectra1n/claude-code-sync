8/// Represents a single line/entry in the JSONL conversation file
9#[derive(Debug, Clone, Serialize, Deserialize)]
10pub struct ConversationEntry {
11    /// The type of this entry (e.g., "user", "assistant", "file-history-snapshot")
12    ///
13    /// This field identifies what kind of entry this is in the conversation.
14    /// Common types include user messages, assistant responses, and system events.
15    #[serde(rename = "type")]
16    pub entry_type: String,
17
18    /// Unique identifier for this conversation entry
19    ///
20    /// Each entry may have its own UUID to uniquely identify it within the conversation.
21    /// Not all entry types require a UUID, hence this is optional.
22    #[serde(skip_serializing_if = "Option::is_none")]
23    pub uuid: Option<String>,
24
25    /// UUID of the parent entry in the conversation thread
26    ///
27    /// This links entries together in a conversation tree, allowing for branching
28    /// and threading of messages. If present, it references the UUID of the entry
29    /// that this entry is responding to or following from.
30    #[serde(rename = "parentUuid", skip_serializing_if = "Option::is_none")]
31    pub parent_uuid: Option<String>,
32
33    /// Session identifier grouping related conversation entries together
34    ///
35    /// All entries within a single conversation session share the same session ID.
36    /// This is used to associate entries across multiple files or to reconstruct
37    /// conversation context. If not present in the entry, the filename may be used.
38    #[serde(rename = "sessionId", skip_serializing_if = "Option::is_none")]
39    pub session_id: Option<String>,
40
41    /// ISO 8601 timestamp indicating when this entry was created
42    ///
43    /// Format is typically "YYYY-MM-DDTHH:MM:SS.sssZ" (e.g., "2025-01-01T00:00:00.000Z").
44    /// Used for sorting entries chronologically and determining the latest activity.
45    #[serde(skip_serializing_if = "Option::is_none")]
46    pub timestamp: Option<String>,
47
48    /// The actual message content as a JSON value
49    ///
50    /// Contains the text and structured data of the user or assistant message.
51    /// Stored as a generic JSON value to accommodate different message formats
52    /// and structures without strict schema requirements.
53    #[serde(skip_serializing_if = "Option::is_none")]
54    pub message: Option<Value>,
55
56    /// Current working directory at the time this entry was created
57    ///
58    /// Stores the filesystem path of the working directory, providing context
59    /// about where the conversation or command was executed. Useful for
60    /// reproducing environments and understanding file references.
61    #[serde(skip_serializing_if = "Option::is_none")]
62    pub cwd: Option<String>,
63
64    /// Version string of the Claude Code CLI that created this entry
65    ///
66    /// Records which version of the tool generated this conversation entry,
67    /// helpful for debugging compatibility issues and tracking feature support.
68    #[serde(skip_serializing_if = "Option::is_none")]
69    pub version: Option<String>,
70
71    /// Git branch name active when this entry was created
72    ///
73    /// Captures the current git branch context, allowing conversation entries
74    /// to be associated with specific branches in version control. Useful for
75    /// tracking which branch work was performed on.
76    #[serde(rename = "gitBranch", skip_serializing_if = "Option::is_none")]
77    pub git_branch: Option<String>,
78
79    /// Catch-all field for additional JSON properties not explicitly defined
80    ///
81    /// Preserves any extra fields in the JSON that aren't part of the explicit schema.
82    /// This allows forward compatibility - newer versions can add fields without breaking
83    /// older parsers. The flattened serde attribute merges these fields at the same level
84    /// as the named fields when serializing/deserializing.
85    #[serde(flatten)]
86    pub extra: Value,
87}

89/// One conversation session, summarized.
90///
91/// The entries themselves are deliberately NOT held. Discovery reads every
92/// transcript under a tree, and a machine with a few gigabytes of history
93/// would need tens of gigabytes of memory to keep every parsed message of
94/// every session alive at once — while sync only ever compares identity,
95/// recency, size and content hash. The messages of the one session being
96/// merged are read back from the file with [`ConversationSession::load_entries`].
97#[derive(Debug, Clone)]
98pub struct ConversationSession {
99    /// Unique identifier for this conversation session
100    ///
101    /// Either extracted from the first entry that contains a sessionId field,
102    /// or derived from the filename (without extension) if no entries contain
103    /// a session ID. Used to group related conversation entries together.
104    pub session_id: String,
105
106    /// Path to the JSONL file this session was loaded from
107    ///
108    /// Stores the filesystem path of the source file, used for tracking the
109    /// origin of the conversation data and for reading the messages back. Kept
110    /// as a path, not a string: a project directory whose name is not valid
111    /// UTF-8 would otherwise come back with replacement characters and no
112    /// longer name a file that exists.
113    pub file_path: PathBuf,
114
115    latest_timestamp: Option<String>,
116    message_count: usize,
117    content_hash: String,
118    project_name: Option<String>,
119
120    /// How much of the file the summary covers: the offset just past the last
121    /// message read. Anything after it arrived while the file was being read.
122    summarized_bytes: u64,
123}
124
125impl ConversationSession {
126    /// Summarize a JSONL transcript in a single streaming pass, keeping the
127    /// per-session facts and dropping every message as it goes.
128    pub fn from_file<P: AsRef<Path>>(path: P) -> Result<Self> {
129        let path = path.as_ref();
130        let mut entry_session_id = None;
131        let mut latest_timestamp: Option<String> = None;
132        let mut message_count = 0;
133        let mut first_cwd: Option<String> = None;
134        let mut hasher = ContentHasher::new();
135
136        let summarized_bytes = for_each_entry(path, |entry| {
137            // Remember the first interior sessionId, used only as a fallback below.
138            if entry_session_id.is_none() {
139                if let Some(ref sid) = entry.session_id {
140                    entry_session_id = Some(sid.clone());
141                }
142            }
143
144            if let Some(ref timestamp) = entry.timestamp {
145                let is_later = latest_timestamp
146                    .as_ref()
147                    .is_none_or(|latest| timestamp > latest);
148                if is_later {
149                    latest_timestamp = Some(timestamp.clone());
150                }
151            }
152
153            if is_message(&entry) {
154                message_count += 1;
155            }
156
157            if first_cwd.is_none() {
158                first_cwd.clone_from(&entry.cwd);
159            }
160
161            hash_entry(&entry, &mut hasher);
162
163            Ok(())
164        })?;
165
166        // Identity is the filename stem, which Claude Code guarantees is unique per
167        // transcript: `<uuid>.jsonl` for a session and `agent-<hash>.jsonl` for each
168        // of its subagents. The interior `sessionId` field is NOT unique — subagent
169        // sidechain transcripts carry their *parent* session's id, so keying identity
170        // on it collapses every subagent onto the parent and makes sync drop the
171        // parent's main transcript. Fall back to the interior id only if the path has
172        // no usable stem.
173        let session_id = path
174            .file_stem()
175            .and_then(|s| s.to_str())
176            .map(|s| s.to_string())
177            .or(entry_session_id)
178            .with_context(|| {
179                format!(
180                    "No session ID found in file or filename: {}",
181                    path.display()
182                )
183            })?;
184
185        // The project is the working directory of the first entry that names
186        // one, whatever the later entries say.
187        let project_name = first_cwd
188            .as_deref()
189            .and_then(|cwd| Path::new(cwd).file_name())
190            .and_then(|name| name.to_str())
191            .map(|name| name.to_string());
192
193        Ok(ConversationSession {
194            session_id,
195            file_path: path.to_path_buf(),
196            latest_timestamp,
197            message_count,
198            content_hash: hasher.finish(),
199            project_name,
200            summarized_bytes,
201        })
202    }
203
204    /// Read the session's messages back from its file. Only a session being
205    /// merged needs them, and one transcript at a time is what keeps sync's
206    /// memory flat however much history a machine has.
207    pub fn load_entries(&self) -> Result<Vec<ConversationEntry>> {
208        let mut entries = Vec::new();
209        for_each_entry(&self.file_path, |entry| {
210            entries.push(entry);
211            Ok(())
212        })?;
213        Ok(entries)
214    }
215
216    /// Copy the transcript verbatim to another path, creating its directory.
217    /// Byte-for-byte: a re-serialized transcript is a rewritten transcript.
218    ///
219    /// The copy holds exactly the messages this summary was taken from.
220    /// Claude Code appends to a live session while sync runs, and a message
221    /// that arrived since — possibly half-written — would make the copy parse
222    /// differently, or not at all. It travels with the next sync instead.
223    pub fn copy_to<P: AsRef<Path>>(&self, path: P) -> Result<()> {
224        let path = path.as_ref();
225
226        // Copying a file onto itself truncates it: the destination is opened
227        // for writing first, and that is the same inode.
228        if is_same_file(&self.file_path, path) {
229            return Ok(());
230        }
231
232        if let Some(parent) = path.parent() {
233            std::fs::create_dir_all(parent)
234                .with_context(|| format!("Failed to create directory: {}", parent.display()))?;
235        }
236
237        std::fs::copy(&self.file_path, path).with_context(|| {
238            format!(
239                "Failed to copy {} to {}",
240                self.file_path.display(),
241                path.display()
242            )
243        })?;
244
245        trim_to_summarized_length(path, self.summarized_bytes)
246            .with_context(|| format!("Failed to finish copying to {}", path.display()))?;
247
248        Ok(())
249    }
250
251    /// The latest timestamp seen in the conversation.
252    pub fn latest_timestamp(&self) -> Option<&str> {
253        self.latest_timestamp.as_deref()
254    }
255
256    /// The number of messages (user + assistant) in the conversation.
257    pub fn message_count(&self) -> usize {
258        self.message_count
259    }
260
261    /// The project name from the first entry's `cwd` path.
262    pub fn project_name(&self) -> Option<&str> {
263        self.project_name.as_deref()
264    }
265
266    /// A hash of the conversation content, for telling two copies apart.
267    pub fn content_hash(&self) -> &str {
268        &self.content_hash
269    }
270}

272/// The largest line buffer a parse keeps between lines, 1 MB.
273const MAX_RETAINED_LINE_BUFFER: usize = 1024 * 1024;

275/// Whether an entry is a message, as opposed to a snapshot, a summary or
276/// whatever else Claude Code records in a transcript.
277pub fn is_message(entry: &ConversationEntry) -> bool {
278    entry.entry_type == "user" || entry.entry_type == "assistant"
279}

281/// Whether two paths name the same file on disk.
282fn is_same_file(source: &Path, destination: &Path) -> bool {
283    let (Ok(source), Ok(destination)) = (source.canonicalize(), destination.canonicalize()) else {
284        return false;
285    };
286    source == destination
287}

289/// Cut a freshly copied transcript back to the messages that were summarized,
290/// dropping whatever was appended to the original while it was being read.
291fn trim_to_summarized_length(path: &Path, summarized_bytes: u64) -> Result<()> {
292    let copied_bytes = std::fs::metadata(path)?.len();
293    if copied_bytes <= summarized_bytes {
294        return Ok(());
295    }
296
297    // The copy carries the original's permissions, and a read-only transcript
298    // would leave a destination neither this run nor the next one can write.
299    let permissions = std::fs::metadata(path)?.permissions();
300    if permissions.readonly() {
301        allow_writing(path, permissions)?;
302    }
303
304    std::fs::OpenOptions::new()
305        .write(true)
306        .open(path)?
307        .set_len(summarized_bytes)?;
308
309    Ok(())
310}

312/// Give the file's owner permission to write it, leaving the rest alone.
313#[cfg(unix)]
314fn allow_writing(path: &Path, permissions: std::fs::Permissions) -> Result<()> {
315    use std::os::unix::fs::PermissionsExt;
316
317    let mut permissions = permissions;
318    permissions.set_mode(permissions.mode() | 0o200);
319    std::fs::set_permissions(path, permissions)?;
320
321    Ok(())
322}

324/// Clear the file's read-only flag.
325#[cfg(not(unix))]
326// The lint warns that this makes a file world-writable on Unix; this is the
327// non-Unix branch, where it only clears the read-only attribute.
328#[allow(clippy::permissions_set_readonly_false)]
329fn allow_writing(path: &Path, permissions: std::fs::Permissions) -> Result<()> {
330    let mut permissions = permissions;
331    permissions.set_readonly(false);
332    std::fs::set_permissions(path, permissions)?;
333
334    Ok(())
335}

337/// The content hash of a transcript: FNV-1a over the serialized messages.
338///
339/// Written out here rather than taken from `std::collections::hash_map::
340/// DefaultHasher`, whose value std explicitly does not keep stable across Rust
341/// releases: a hashing rule nobody can state is one nobody can test, and this
342/// one decides whether a conversation counts as changed. FNV-1a is nine lines,
343/// needs no dependency, is measurably faster on the many small writes
344/// serialization produces, and is pinned by `the_content_hash_never_changes`.
345///
346/// The hash is always 16 hex digits, which is what the conflict screen prints.
347///
348/// Bytes are fed in as they are serialized, so hashing a message never builds
349/// its JSON text in memory first: a message can be megabytes, and every core
350/// is summarizing a different transcript at the same time.
351struct ContentHasher {
352    state: u64,
353}
354
355const FNV_OFFSET_BASIS: u64 = 0xcbf2_9ce4_8422_2325;
356const FNV_PRIME: u64 = 0x0000_0100_0000_01b3;
357
358/// Separates one message from the next, so two transcripts cannot hash alike
359/// by moving a byte across a message boundary. Never appears in the input:
360/// serialized JSON is UTF-8, and 0xff is not.
361const ENTRY_TERMINATOR: u8 = 0xff;
362
363impl ContentHasher {
364    fn new() -> Self {
365        ContentHasher {
366            state: FNV_OFFSET_BASIS,
367        }
368    }
369
370    fn write_bytes(&mut self, bytes: &[u8]) {
371        for byte in bytes {
372            self.state ^= u64::from(*byte);
373            self.state = self.state.wrapping_mul(FNV_PRIME);
374        }
375    }
376
377    /// The hash so far, as the fixed 16 hex digits callers slice and print.
378    fn finish(&self) -> String {
379        format!("{:016x}", self.state)
380    }
381}
382
383impl std::io::Write for ContentHasher {
384    fn write(&mut self, buf: &[u8]) -> std::io::Result<usize> {
385        self.write_bytes(buf);
386        Ok(buf.len())
387    }
388
389    fn flush(&mut self) -> std::io::Result<()> {
390        Ok(())
391    }
392}

394/// Hash one entry's JSON, then the terminator.
395fn hash_entry(entry: &ConversationEntry, hasher: &mut ContentHasher) {
396    let serialized = serde_json::to_writer(&mut *hasher, entry);
397    hasher.write_bytes(&[ENTRY_TERMINATOR]);
398
399    if let Err(error) = serialized {
400        log::warn!("Hashing an entry that would not serialize: {error}");
401    }
402}

404/// Parse a JSONL transcript line by line, handing each entry to `visit` and
405/// dropping it again unless the caller keeps it. The entry is handed over by
406/// value, so a caller that wants to keep it never copies it.
407///
408/// Returns the offset just past the last message read, which is how much of
409/// the file the caller has actually seen.
410fn for_each_entry<F>(path: &Path, mut visit: F) -> Result<u64>
411where
412    F: FnMut(ConversationEntry) -> Result<()>,
413{
414    let file =
415        File::open(path).with_context(|| format!("Failed to open file: {}", path.display()))?;
416    let mut reader = BufReader::new(file);
417
418    // One buffer for the whole file, reused line by line: a transcript is read
419    // in a few kilobytes at a time however large it is, and the thread that
420    // reads it allocates once instead of once per message.
421    let mut line = String::new();
422    let mut line_num = 0;
423    let mut read_bytes = 0u64;
424    let mut parsed_bytes = 0u64;
425
426    loop {
427        line.clear();
428        // Clearing keeps the capacity, which is the point — except after a
429        // multi-megabyte tool result, where holding that buffer for the rest of
430        // the file on every core is exactly the memory this avoids elsewhere.
431        if line.capacity() > MAX_RETAINED_LINE_BUFFER {
432            line.shrink_to(MAX_RETAINED_LINE_BUFFER);
433        }
434        let bytes_read = reader.read_line(&mut line).with_context(|| {
435            format!("Failed to read line {} in {}", line_num + 1, path.display())
436        })?;
437        if bytes_read == 0 {
438            break;
439        }
440        line_num += 1;
441        read_bytes += bytes_read as u64;
442
443        if line.trim().is_empty() {
444            continue;
445        }
446
447        let entry: ConversationEntry = serde_json::from_str(&line).with_context(|| {
448            format!(
449                "Failed to parse JSON at line {} in {}",
450                line_num,
451                path.display()
452            )
453        })?;
454
455        visit(entry)?;
456        parsed_bytes = read_bytes;
457    }
458
459    Ok(parsed_bytes)
460}

462/// Write conversation entries to a JSONL file, creating its directory. Used
463/// for the one transcript a merge rewrites; everything else is copied.
464pub fn write_entries_to_file<P: AsRef<Path>>(path: P, entries: &[ConversationEntry]) -> Result<()> {
465    let path = path.as_ref();
466
467    if let Some(parent) = path.parent() {
468        std::fs::create_dir_all(parent)
469            .with_context(|| format!("Failed to create directory: {}", parent.display()))?;
470    }
471
472    let mut file =
473        File::create(path).with_context(|| format!("Failed to create file: {}", path.display()))?;
474
475    for entry in entries {
476        let json =
477            serde_json::to_string(entry).context("Failed to serialize conversation entry")?;
478        writeln!(file, "{json}")
479            .with_context(|| format!("Failed to write to file: {}", path.display()))?;
480    }
481
482    Ok(())
483}

826        }

837        assert_eq!(session.message_count(), 500);

853    /// Runs `work` and reports how many bytes it allocated and did not free.
854    ///
855    /// Nothing else can tell a summary apart from a struct that keeps every
856    /// message: both are the same handful of bytes wide, and only the heap
857    /// behind them differs.
858    fn bytes_retained_by<T>(work: impl FnOnce() -> T) -> (T, isize) {
859        let before = LIVE_BYTES.with(|live| live.get());
860        let produced = work();
861        let after = LIVE_BYTES.with(|live| live.get());
862        (produced, after - before)
863    }

865    #[global_allocator]
866    static COUNTING_ALLOCATOR: CountingAllocator = CountingAllocator;
867
868    /// The system allocator, counting what the calling thread holds.
869    struct CountingAllocator;
870
871    thread_local! {
872        static LIVE_BYTES: std::cell::Cell<isize> = const { std::cell::Cell::new(0) };
873    }
874
875    unsafe impl std::alloc::GlobalAlloc for CountingAllocator {
876        unsafe fn alloc(&self, layout: std::alloc::Layout) -> *mut u8 {
877            let _ = LIVE_BYTES.try_with(|live| live.set(live.get() + layout.size() as isize));
878            std::alloc::System.alloc(layout)
879        }
880
881        unsafe fn dealloc(&self, pointer: *mut u8, layout: std::alloc::Layout) {
882            let _ = LIVE_BYTES.try_with(|live| live.set(live.get() - layout.size() as isize));
883            std::alloc::System.dealloc(pointer, layout)
884        }
885    }
