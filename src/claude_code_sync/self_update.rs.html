19/// The release asset built for the platform this binary was compiled for,
20/// mirroring the matrix in `.github/workflows/build.yml`.
21///
22/// Keyed on the compile target rather than the running OS, so a musl build
23/// updates to musl and an x86_64 build running under emulation stays x86_64.
24fn release_asset() -> Option<&'static str> {
25    let asset = match (
26        std::env::consts::OS,
27        std::env::consts::ARCH,
28        cfg!(target_env = "musl"),
29    ) {
30        ("linux", "x86_64", false) => "claude-code-sync-linux-x86_64.tar.gz",
31        ("linux", "x86_64", true) => "claude-code-sync-linux-x86_64-musl.tar.gz",
32        ("linux", "aarch64", false) => "claude-code-sync-linux-aarch64.tar.gz",
33        ("linux", "aarch64", true) => "claude-code-sync-linux-aarch64-musl.tar.gz",
34        ("macos", "x86_64", _) => "claude-code-sync-macos-x86_64.tar.gz",
35        ("macos", "aarch64", _) => "claude-code-sync-macos-aarch64.tar.gz",
36        ("windows", "x86_64", _) => "claude-code-sync-windows-x86_64.exe.zip",
37        ("windows", "aarch64", _) => "claude-code-sync-windows-aarch64.exe.zip",
38        _ => return None,
39    };
40    Some(asset)
41}

43/// A package manager that owns the installed binary. Replacing its file in
44/// place would desync the manager's records (or fail on a read-only store),
45/// so `self-update` defers to it.
46#[derive(Debug, PartialEq, Eq)]
47enum Manager {
48    Nix,
49    Homebrew,
50    Scoop,
51    Cargo,
52}
53
54impl Manager {
55    fn detect(exe: &Path) -> Option<Self> {
56        // Normalise separators so one set of patterns covers Windows paths.
57        let path = exe.to_string_lossy().replace('\\', "/").to_lowercase();
58        if path.starts_with("/nix/store/") {
59            Some(Manager::Nix)
60        } else if path.contains("/cellar/") || path.contains("/homebrew/") {
61            Some(Manager::Homebrew)
62        } else if path.contains("/scoop/apps/") {
63            Some(Manager::Scoop)
64        } else if path.contains("/.cargo/bin/") {
65            Some(Manager::Cargo)
66        } else {
67            None
68        }
69    }
70
71    fn name(&self) -> &'static str {
72        match self {
73            Manager::Nix => "Nix",
74            Manager::Homebrew => "Homebrew",
75            Manager::Scoop => "Scoop",
76            Manager::Cargo => "cargo",
77        }
78    }
79
80    fn upgrade_hint(&self) -> &'static str {
81        match self {
82            Manager::Nix => "nix profile upgrade claude-code-sync  (or bump your flake input)",
83            Manager::Homebrew => "brew upgrade claude-code-sync",
84            Manager::Scoop => "scoop update claude-code-sync",
85            Manager::Cargo => {
86                "cargo binstall claude-code-sync  (or: cargo install claude-code-sync)"
87            }
88        }
89    }
90}

92/// Normalise a user-supplied version to a release tag: `0.3.3` -> `v0.3.3`.
93fn normalize_tag(version: &str) -> String {
94    let version = version.trim();
95    if version.starts_with('v') {
96        version.to_string()
97    } else {
98        format!("v{version}")
99    }
100}
101
102/// Parse the version out of a release tag, accepting a leading `v`.
103fn parse_tag(tag: &str) -> Result<Version> {
104    Version::parse(tag.trim_start_matches('v'))
105        .with_context(|| format!("'{tag}' is not a valid release version"))
106}

108/// Extract the tag from the `Location` of a `/releases/latest` redirect,
109/// e.g. `.../releases/tag/v0.3.3` -> `v0.3.3`.
110fn tag_from_location(location: &str) -> Option<&str> {
111    let (_, tag) = location.rsplit_once("/releases/tag/")?;
112    let tag = tag.trim_end_matches('/');
113    (!tag.is_empty() && !tag.contains('/')).then_some(tag)
114}

116/// The expected hash from a `<asset>.sha256` file (`<hex>  <filename>`).
117fn parse_checksum(contents: &str) -> Option<String> {
118    let hash = contents.split_whitespace().next()?.to_ascii_lowercase();
119    (hash.len() == 64 && hash.bytes().all(|b| b.is_ascii_hexdigit())).then_some(hash)
120}
121
122fn sha256_hex(bytes: &[u8]) -> String {
123    Sha256::digest(bytes)
124        .iter()
125        .map(|b| format!("{b:02x}"))
126        .collect()
127}

129fn agent(follow_redirects: bool) -> ureq::Agent {
130    let config = ureq::Agent::config_builder()
131        .user_agent(concat!("claude-code-sync/", env!("CARGO_PKG_VERSION")))
132        .timeout_global(Some(std::time::Duration::from_secs(120)))
133        .max_redirects(if follow_redirects { 10 } else { 0 });
134    #[cfg(windows)]
135    let config = config.tls_config(
136        ureq::tls::TlsConfig::builder()
137            .provider(ureq::tls::TlsProvider::NativeTls)
138            .build(),
139    );
140    config.build().new_agent()
141}

143/// Resolve the newest release tag.
144///
145/// Reads the redirect GitHub serves for `/releases/latest` instead of calling
146/// the REST API, whose unauthenticated limit (60/hour, shared per IP) is easy
147/// to exhaust behind a NAT or on CI.
148fn latest_tag() -> Result<String> {
149    let url = format!("{REPO_URL}/releases/latest");
150    let response = agent(false)
151        .get(&url)
152        .call()
153        .with_context(|| format!("failed to query {url}"))?;
154    let location = response
155        .headers()
156        .get("location")
157        .and_then(|v| v.to_str().ok())
158        .ok_or_else(|| {
159            anyhow!(
160                "{url} did not redirect to a release (HTTP {})",
161                response.status()
162            )
163        })?;
164    tag_from_location(location)
165        .map(str::to_string)
166        .ok_or_else(|| anyhow!("unexpected redirect target for latest release: {location}"))
167}
168
169fn download(url: &str) -> Result<Vec<u8>> {
170    agent(true)
171        .get(url)
172        .call()
173        .with_context(|| format!("failed to download {url}"))?
174        .body_mut()
175        .with_config()
176        .limit(MAX_DOWNLOAD_BYTES)
177        .read_to_vec()
178        .with_context(|| format!("failed to read {url}"))
179}

181/// Pull the binary out of a release archive.
182#[cfg(not(windows))]
183fn extract_binary(archive: &[u8]) -> Result<Vec<u8>> {
184    let mut tar = tar::Archive::new(flate2::read::GzDecoder::new(archive));
185    for entry in tar
186        .entries()
187        .context("release archive is not a valid .tar.gz")?
188    {
189        let mut entry = entry?;
190        if entry.path()?.file_name() == Some(std::ffi::OsStr::new(BIN_NAME)) {
191            let mut bytes = Vec::new();
192            entry.read_to_end(&mut bytes)?;
193            return Ok(bytes);
194        }
195    }
196    bail!("release archive does not contain {BIN_NAME}")
197}

199/// Pull the binary out of a release archive.
200#[cfg(windows)]
201fn extract_binary(archive: &[u8]) -> Result<Vec<u8>> {
202    let mut zip = zip::ZipArchive::new(std::io::Cursor::new(archive))
203        .context("release archive is not a valid .zip")?;
204    let mut entry = zip
205        .by_name(BIN_NAME)
206        .with_context(|| format!("release archive does not contain {BIN_NAME}"))?;
207    let mut bytes = Vec::new();
208    entry.read_to_end(&mut bytes)?;
209    Ok(bytes)
210}

212/// Swap the running executable for `binary`.
213fn install_binary(binary: &[u8]) -> Result<()> {
214    let mut staged = tempfile::NamedTempFile::new().context("failed to create a temp file")?;
215    std::io::Write::write_all(&mut staged, binary)?;
216    #[cfg(unix)]
217    {
218        use std::os::unix::fs::PermissionsExt;
219        std::fs::set_permissions(staged.path(), std::fs::Permissions::from_mode(0o755))?;
220    }
221    // self_replace handles the platform quirks: an atomic rename on Unix, and
222    // on Windows moving the locked, running .exe aside before writing the new one.
223    self_replace::self_replace(staged.path()).context(
224        "failed to replace the running binary (is its directory writable? a system \
225         path may need sudo)",
226    )
227}

229/// Update this binary to the latest release, or to `target` if given.
230///
231/// With `check_only`, report whether an update is available and change
232/// nothing. `force` reinstalls even when already current, and overrides the
233/// refusal to touch a binary owned by a package manager.
234pub fn self_update(check_only: bool, target: Option<&str>, force: bool) -> Result<()> {
235    let current = parse_tag(env!("CARGO_PKG_VERSION"))?;
236    let tag = match target {
237        Some(v) => normalize_tag(v),
238        None => latest_tag()?,
239    };
240    let wanted = parse_tag(&tag)?;
241
242    let up_to_date = if target.is_some() {
243        wanted == current
244    } else {
245        wanted <= current
246    };
247
248    if check_only {
249        if up_to_date {
250            println!("{} claude-code-sync {} is up to date", "✓".green(), current);
251        } else {
252            println!(
253                "{} {} → {} (run {})",
254                "Update available:".yellow().bold(),
255                current,
256                wanted.to_string().green(),
257                "claude-code-sync self-update".cyan()
258            );
259        }
260        return Ok(());
261    }
262
263    if up_to_date && !force {
264        println!(
265            "{} claude-code-sync {} is already installed",
266            "✓".green(),
267            current
268        );
269        return Ok(());
270    }
271
272    let exe = std::env::current_exe().context("cannot locate the running executable")?;
273    let exe = exe.canonicalize().unwrap_or(exe);
274    if let Some(manager) = Manager::detect(&exe) {
275        if !force {
276            bail!(
277                "{} is managed by {}; update it with:\n  {}\n(pass --force to replace it anyway)",
278                exe.display(),
279                manager.name(),
280                manager.upgrade_hint()
281            );
282        }
283    }
284
285    let asset = release_asset().ok_or_else(|| {
286        anyhow!(
287            "no prebuilt release for {}-{}; update with: cargo install claude-code-sync",
288            std::env::consts::OS,
289            std::env::consts::ARCH
290        )
291    })?;
292    let url = format!("{REPO_URL}/releases/download/{tag}/{asset}");
293
294    println!("{} {} ({})...", "Downloading".cyan(), asset, tag);
295    let archive = download(&url)?;
296    let checksum_file = String::from_utf8(download(&format!("{url}.sha256"))?)
297        .context("checksum file is not valid UTF-8")?;
298    let expected = parse_checksum(&checksum_file)
299        .ok_or_else(|| anyhow!("malformed checksum file for {asset}"))?;
300    let actual = sha256_hex(&archive);
301    if expected != actual {
302        bail!("checksum mismatch for {asset}: expected {expected}, got {actual}");
303    }
304
305    let binary = extract_binary(&archive)?;
306    install_binary(&binary)?;
307
308    println!(
309        "{} claude-code-sync {} → {} ({})",
310        "Updated".green().bold(),
311        current,
312        wanted,
313        exe.display()
314    );
315    Ok(())
316}

409    #[test]
410    fn extract_binary_finds_the_executable() {
411        let mut builder = tar::Builder::new(flate2::write::GzEncoder::new(
412            Vec::new(),
413            flate2::Compression::default(),
414        ));
415        for (name, body) in [("README.md", &b"docs"[..]), (BIN_NAME, &b"\x7fELF"[..])] {
416            let mut header = tar::Header::new_gnu();
417            header.set_size(body.len() as u64);
418            header.set_mode(0o755);
419            header.set_cksum();
420            builder.append_data(&mut header, name, body).unwrap();
421        }
422        let archive = builder.into_inner().unwrap().finish().unwrap();
423
424        assert_eq!(extract_binary(&archive).unwrap(), b"\x7fELF");
425        assert!(extract_binary(b"not an archive").is_err());
426    }
