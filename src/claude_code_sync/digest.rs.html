1//! SHA-256 digests, hex-encoded: one helper for every caller (artifact base
2//! records, release self-update verification) so they cannot drift apart.
3
4use sha2::{Digest, Sha256};
5use std::fmt::Write as _;
6
7/// The lowercase hex SHA-256 of `bytes`.
8pub fn sha256_hex(bytes: &[u8]) -> String {
9    // One preallocated string, written into: this runs over every
10    // artifact byte on every sync, and a format! per digest byte meant
11    // millions of throwaway Strings for a single large file.
12    let digest = Sha256::digest(bytes);
13    let mut hex = String::with_capacity(digest.len() * 2);
14    for byte in digest {
15        let _ = write!(hex, "{byte:02x}");
16    }
17    hex
18}
