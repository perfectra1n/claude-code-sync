1//! Settling a file both machines changed only in its date-times.
2//!
3//! Claude Code rewrites timestamps such as `lastUpdated` in
4//! `plugins/known_marketplaces.json` and skill manifests on every machine, so
5//! those files differ between machines without anyone editing them. Such a
6//! difference settles on the later date-time, with no prompt.
7//!
8//! The settle silently picks one side, so it must never fire on real content:
9//! a value only counts as a timestamp when it is the whole value of a key
10//! whose name says it is a time (`lastUpdated`, `modified`, `createdAt`,
11//! `expires_at`, ...). An ISBN, an account number or a date written in prose
12//! is content, and a difference in it is a real conflict.
13
14use chrono::{DateTime, NaiveDateTime};
15
16const ISO_DATE_TIME_SHAPE: &[u8] = b"dddd-dd-ddTdd:dd:dd";
17const OFFSET_SHAPE: &[u8] = b"dd:dd";
18const EPOCH_MILLISECONDS_DIGITS: usize = 13;
19
20struct Timestamp<'a> {
21    text: &'a str,
22    milliseconds: i64,
23}
24
25/// The two versions with every date-time that differs set to the later one,
26/// or `None` when they differ in anything else.
27pub fn keep_later_timestamps(local: &[u8], remote: &[u8]) -> Option<Vec<u8>> {
28    let local = std::str::from_utf8(local).ok()?;
29    let remote = std::str::from_utf8(remote).ok()?;
30    let mut merged = String::with_capacity(local.len());
31    let mut local_position = 0;
32    let mut remote_position = 0;
33
34    while local_position < local.len() || remote_position < remote.len() {
35        let local_timestamp = read_timestamp(local, local_position);
36        let remote_timestamp = read_timestamp(remote, remote_position);
37        if let (Some(local_timestamp), Some(remote_timestamp)) = (local_timestamp, remote_timestamp)
38        {
39            let remote_is_later = remote_timestamp.milliseconds > local_timestamp.milliseconds;
40            let later = if remote_is_later {
41                remote_timestamp.text
42            } else {
43                local_timestamp.text
44            };
45            merged.push_str(later);
46            local_position += local_timestamp.text.len();
47            remote_position += remote_timestamp.text.len();
48            continue;
49        }
50
51        let local_character = local[local_position..].chars().next()?;
52        let remote_character = remote[remote_position..].chars().next()?;
53        if local_character != remote_character {
54            return None;
55        }
56        merged.push(local_character);
57        local_position += local_character.len_utf8();
58        remote_position += remote_character.len_utf8();
59    }
60
61    Some(merged.into_bytes())
62}

64fn read_timestamp(text: &str, position: usize) -> Option<Timestamp<'_>> {
65    if !is_the_value_of_a_time_key(&text[..position]) {
66        return None;
67    }
68    let rest = &text[position..];
69    let timestamp = read_iso_date_time(rest).or_else(|| read_epoch_milliseconds(rest))?;
70    // The whole value, not the start of a longer word or number.
71    let next_character = rest[timestamp.text.len()..].chars().next();
72    if next_character.is_some_and(|character| character.is_alphanumeric()) {
73        return None;
74    }
75    Some(timestamp)
76}

78/// Whether the text just before a value is `key: ` (YAML) or `"key": `
79/// (JSON), optionally followed by the value's opening quote, with a key name
80/// that says it holds a time.
81fn is_the_value_of_a_time_key(before: &str) -> bool {
82    let before = before
83        .strip_suffix('"')
84        .or_else(|| before.strip_suffix('\''))
85        .unwrap_or(before);
86    let Some(before) = before.trim_end_matches([' ', '\t']).strip_suffix(':') else {
87        return false;
88    };
89    let before = before
90        .strip_suffix('"')
91        .or_else(|| before.strip_suffix('\''))
92        .unwrap_or(before);
93    let key_start = before
94        .rfind(|character: char| {
95            !(character.is_ascii_alphanumeric() || character == '_' || character == '-')
96        })
97        .map_or(0, |index| index + 1);
98    names_a_time(&before[key_start..])
99}

101fn names_a_time(key: &str) -> bool {
102    const TIME_WORDS: &[&str] = &[
103        "time",
104        "date",
105        "updated",
106        "modified",
107        "created",
108        "seen",
109        "stamp",
110        "epoch",
111        "expires",
112        "synced",
113        "fetched",
114        "installed",
115    ];
116    let lowercase = key.to_ascii_lowercase();
117    !key.is_empty()
118        && (TIME_WORDS.iter().any(|word| lowercase.contains(word))
119            || key.ends_with("At")
120            || lowercase.ends_with("_at")
121            || lowercase.ends_with("-at")
122            || lowercase == "at")
123}

125fn read_iso_date_time(text: &str) -> Option<Timestamp<'_>> {
126    let bytes = text.as_bytes();
127    if !matches_shape(bytes, ISO_DATE_TIME_SHAPE) {
128        return None;
129    }
130
131    let mut end = ISO_DATE_TIME_SHAPE.len();
132    let has_fraction = bytes.get(end) == Some(&b'.');
133    if has_fraction {
134        let fraction_digits = count_digits(&bytes[end + 1..]);
135        if fraction_digits == 0 {
136            return None;
137        }
138        end += 1 + fraction_digits;
139    }
140
141    let offset_sign = bytes.get(end).copied();
142    let has_utc_designator = offset_sign == Some(b'Z');
143    let has_numeric_offset = matches!(offset_sign, Some(b'+') | Some(b'-'))
144        && matches_shape(&bytes[end + 1..], OFFSET_SHAPE);
145    if has_utc_designator {
146        end += 1;
147    } else if has_numeric_offset {
148        end += 1 + OFFSET_SHAPE.len();
149    }
150
151    let token = &text[..end];
152    let milliseconds = if has_utc_designator || has_numeric_offset {
153        DateTime::parse_from_rfc3339(token).ok()?.timestamp_millis()
154    } else {
155        NaiveDateTime::parse_from_str(token, "%Y-%m-%dT%H:%M:%S%.f")
156            .ok()?
157            .and_utc()
158            .timestamp_millis()
159    };
160
161    Some(Timestamp {
162        text: token,
163        milliseconds,
164    })
165}

167fn read_epoch_milliseconds(text: &str) -> Option<Timestamp<'_>> {
168    let digits = count_digits(text.as_bytes());
169    if digits != EPOCH_MILLISECONDS_DIGITS {
170        return None;
171    }
172
173    let token = &text[..digits];
174    Some(Timestamp {
175        text: token,
176        milliseconds: token.parse().ok()?,
177    })
178}
179
180fn matches_shape(bytes: &[u8], shape: &[u8]) -> bool {
181    if bytes.len() < shape.len() {
182        return false;
183    }
184    shape.iter().zip(bytes).all(|(expected, actual)| {
185        if *expected == b'd' {
186            actual.is_ascii_digit()
187        } else {
188            expected == actual
189        }
190    })
191}
192
193fn count_digits(bytes: &[u8]) -> usize {
194    bytes
195        .iter()
196        .take_while(|byte| byte.is_ascii_digit())
197        .count()
198}

204    #[test]
205    fn versions_that_differ_only_in_dates_settle_on_the_later_ones() {
206        let cases: &[(&str, &str, &str, Option<&str>)] = &[
207            (
208                "iso timestamp, remote later",
209                "{\"lastUpdated\": \"2026-10-01T06:00:01.741Z\"}",
210                "{\"lastUpdated\": \"2026-10-01T06:50:50.567Z\"}",
211                Some("{\"lastUpdated\": \"2026-10-01T06:50:50.567Z\"}"),
212            ),
213            (
214                "iso timestamp, local later",
215                "  modified: 2026-10-01T12:24:26.923Z\n",
216                "  modified: 2026-09-21T07:13:05.230Z\n",
217                Some("  modified: 2026-10-01T12:24:26.923Z\n"),
218            ),
219            (
220                "each date settles on its own",
221                "{\"createdAt\": \"2026-01-02T00:00:00Z\", \"lastSeen\": 1790137212675}",
222                "{\"createdAt\": \"2026-01-01T00:00:00Z\", \"lastSeen\": 1790837225831}",
223                Some("{\"createdAt\": \"2026-01-02T00:00:00Z\", \"lastSeen\": 1790837225831}"),
224            ),
225            (
226                "numeric offset compared as an instant",
227                "updated_at: 2026-10-01T08:00:00+02:00",
228                "updated_at: 2026-10-01T07:00:00Z",
229                Some("updated_at: 2026-10-01T07:00:00Z"),
230            ),
231            (
232                "a date-time in prose is content",
233                "met at 2026-10-01T08:00:00Z",
234                "met at 2026-10-02T08:00:00Z",
235                None,
236            ),
237            (
238                "an ISBN is content, not epoch milliseconds",
239                "\"isbn\": \"9780134685991\"",
240                "\"isbn\": \"9780201633610\"",
241                None,
242            ),
243            (
244                "an id that looks like epoch milliseconds is content",
245                "{\"accountId\": 1790137212675}",
246                "{\"accountId\": 1790837225831}",
247                None,
248            ),
249            (
250                "a 13-digit number after a letter is content",
251                "token: a1790137212675",
252                "token: a1790837225831",
253                None,
254            ),
255            (
256                "a time key followed by more digits is content",
257                "\"lastUpdated\": 17901372126751",
258                "\"lastUpdated\": 17908372258312",
259                None,
260            ),
261            (
262                "text differs besides the dates",
263                "modified: 2026-10-01T12:24:26Z\nnew paragraph\n",
264                "modified: 2026-09-21T07:13:05Z\n",
265                None,
266            ),
267            (
268                "a plain date in prose is content, not a timestamp",
269                "since 2026-09-26",
270                "since 2026-09-25",
271                None,
272            ),
273            (
274                "a number of another length is content",
275                "count: 12345",
276                "count: 12346",
277                None,
278            ),
279            (
280                "a date on one side only",
281                "at 2026-10-01T07:00:00Z",
282                "at never",
283                None,
284            ),
285        ];
286
287        for (name, local, remote, expected) in cases {
288            let merged = keep_later_timestamps(local.as_bytes(), remote.as_bytes());
289            let merged_text = merged.map(|bytes| String::from_utf8(bytes).unwrap());
290            assert_eq!(
291                merged_text.as_deref(),
292                *expected,
293                "{name}: {local:?} vs {remote:?}"
294            );
295        }
296    }

298    #[test]

301    }
