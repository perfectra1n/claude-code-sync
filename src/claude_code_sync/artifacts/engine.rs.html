23/// Whether one category participates for this configuration: toggles for the
24/// regular categories, the (inverted) attachments flag for ProjectAttachments.
25pub fn is_category_enabled(desc: &CategoryDescriptor, filter: &FilterConfig) -> bool {
26    match desc.id {
27        CategoryId::ProjectAttachments => !filter.exclude_attachments,
28        _ => filter.sync_artifacts.is_enabled(desc.id),
29    }
30}

32/// All registry rows active under this configuration.
33fn active_categories(
34    filter: &FilterConfig,

36    REGISTRY.iter().filter(|d| is_category_enabled(d, filter))

39/// The sync-repo root directory for one category.
40fn category_repo_root(
41    desc: &CategoryDescriptor,
42    repo_root: &Path,
43    filter: &FilterConfig,
44) -> PathBuf {
45    match desc.dest {
46        DestRoot::Artifacts => repo_root.join(ARTIFACTS_SUBDIR).join(desc.repo_subdir),
47        DestRoot::SessionTree => repo_root.join(&filter.sync_subdirectory),

51/// True when a file extension is excluded for this category.
52fn extension_excluded(desc: &CategoryDescriptor, path: &Path) -> bool {
53    path.extension()
54        .and_then(|e| e.to_str())
55        .map(|ext| {
56            desc.exclude_extensions
57                .iter()
58                .any(|x| ext.eq_ignore_ascii_case(x))
59        })
60        .unwrap_or(false)
61}

63/// Per-category outcome counts for one push or pull.
64#[derive(Debug, Clone, Serialize, Deserialize, PartialEq, Eq)]
65pub struct CategoryCounts {
66    pub category: CategoryId,
67    #[serde(default)]
68    pub added: usize,
69    #[serde(default)]
70    pub modified: usize,
71    #[serde(default)]

116/// Outcome of one artifact push or pull across all enabled categories.
117#[derive(Debug, Clone, Default)]
118pub struct ArtifactReport {
119    pub counts: Vec<CategoryCounts>,
120    /// Every file written or removed, in the order it happened.
121    pub changes: Vec<ArtifactChange>,
122}

124impl ArtifactReport {
125    pub fn total_added(&self) -> usize {
126        self.counts.iter().map(|c| c.added).sum()
127    }
128    pub fn total_modified(&self) -> usize {
129        self.counts.iter().map(|c| c.modified).sum()
130    }
131    pub fn total_unchanged(&self) -> usize {
132        self.counts.iter().map(|c| c.unchanged).sum()
133    }
134    pub fn total_deleted(&self) -> usize {
135        self.counts.iter().map(|c| c.deleted).sum()
136    }
137
138    fn record(&mut self, category: CategoryId, kind: ArtifactChangeKind, path: &Path) {
139        self.changes.push(ArtifactChange {
140            category,
141            kind,
142            path: path.to_path_buf(),
143        });
144    }
145
146    /// True when nothing was copied, merged, or even inspected.
147    #[allow(dead_code)] // used via the library target; the bin compiles this module separately
148    pub fn is_empty(&self) -> bool {
149        self.counts.is_empty()
150            || self
151                .counts
152                .iter()
153                .all(|c| c.added + c.modified + c.unchanged + c.skipped == 0)
154    }
155}

157/// One collected artifact file: its absolute source under `~/.claude` and its
158/// destination path relative to the category's repo subdirectory.
159struct CollectedFile {
160    abs: PathBuf,
161    rel: PathBuf,
162}
163
164/// Enumerate a category's files on disk. Missing sources yield an empty list;
165/// denied paths and oversized files are skipped (the latter counted).
166///
167/// An oversized file still exists on this machine, so its category-relative
168/// path goes into `held_back`: deletion mirroring must not read "not pushed
169/// this time" as "deleted here" and remove every machine's copy.
170fn collect(
171    desc: &CategoryDescriptor,
172    claude_dir: &Path,
173    filter: &FilterConfig,
174    skipped: &mut usize,
175    held_back: &mut Vec<PathBuf>,
176) -> Result<Vec<CollectedFile>> {
177    let max_file_size = filter.max_file_size_bytes;
178    let mut files = Vec::new();
179
180    match desc.source {
181        SourceSpec::Files(list) => {
182            for entry in list {
183                let claude_rel = Path::new(entry);
184                if is_denied(claude_rel) {
185                    *skipped += 1;
186                    continue;
187                }
188                let abs = claude_dir.join(entry);
189                if !abs.is_file() {
190                    continue;
191                }
192                let rel = claude_rel
193                    .file_name()
194                    .map(PathBuf::from)
195                    .unwrap_or_else(|| claude_rel.to_path_buf());
196                files.push(CollectedFile { abs, rel });
197            }
198        }
199        SourceSpec::Dir(dir) => {
200            let base = claude_dir.join(dir);
201            if !base.is_dir() {
202                return Ok(files);
203            }
204            // Resolved once per project: in name-only mode it reads a transcript.
205            let mut project_names: std::collections::HashMap<String, String> =
206                std::collections::HashMap::new();
207            for entry in walkdir::WalkDir::new(&base)
208                .follow_links(false)
209                .into_iter()
210                .filter_map(|e| e.ok())
211            {
212                if !entry.file_type().is_file() {
213                    continue;
214                }
215                let abs = entry.path();
216                // Deny rules match against the ~/.claude-relative path so a
217                // component like `cache/` is caught wherever it appears.
218                let claude_rel = abs.strip_prefix(claude_dir).unwrap_or(abs);
219                if is_denied(claude_rel) {
220                    *skipped += 1;
221                    continue;
222                }
223                if extension_excluded(desc, abs) {
224                    continue;
225                }
226                let mut rel = abs.strip_prefix(&base).unwrap_or(abs).to_path_buf();
227                // Attachments take the project's repo-side directory name,
228                // mirroring session layout.
229                if desc.dest == DestRoot::SessionTree {
230                    let mut parts = rel.components();
231                    let Some(encoded) = parts.next().and_then(|c| c.as_os_str().to_str()) else {
232                        *skipped += 1;
233                        continue;
234                    };
235                    let project = project_names
236                        .entry(encoded.to_string())
237                        .or_insert_with(|| {
238                            crate::project_map::repo_dir_name_in(filter, &base, encoded)
239                        })
240                        .clone();
241                    rel = Path::new(&project).join(parts.as_path());
242                }
243                if entry.metadata().map(|m| m.len()).unwrap_or(0) > max_file_size {
244                    log::warn!("Skipping {} (exceeds max_file_size_bytes)", abs.display());
245                    *skipped += 1;
246                    held_back.push(rel);
247                    continue;
248                }
249                files.push(CollectedFile {
250                    abs: abs.to_path_buf(),
251                    rel,
252                });
253            }
254        }
255    }
256
257    Ok(files)
258}

260/// Write `content` to `path` via a same-directory temp file + atomic rename,
261/// so a reader (or a crash) never sees a half-written file. `mode_source` is
262/// the file whose executable bit the result adopts.
263#[cfg_attr(not(unix), allow(unused_variables))]
264fn write_atomic(path: &Path, content: &[u8], mode_source: &Path) -> Result<()> {
265    let parent = path
266        .parent()
267        .with_context(|| format!("No parent directory for {}", path.display()))?;
268    fs::create_dir_all(parent)?;
269    let tmp = tempfile::NamedTempFile::new_in(parent)?;
270    fs::write(tmp.path(), content)?;
271    #[cfg(unix)]
272    {
273        if let Some(mode) = get_mode_for_copy(path, mode_source) {
274            set_mode(tmp.path(), mode);
275        }
276    }
277    tmp.persist(path)
278        .with_context(|| format!("Failed to persist {}", path.display()))?;
279    Ok(())
280}

282/// The mode a copy of `mode_source` should have at `destination`: what the
283/// destination already has, plus the owner's executable bit when the source is
284/// executable. Granting only: a repository written before this bit was synced
285/// holds every file non-executable, and a pull from it must not disarm the
286/// scripts on this machine.
287#[cfg(unix)]
288fn get_mode_for_copy(destination: &Path, mode_source: &Path) -> Option<u32> {
289    use std::os::unix::fs::PermissionsExt;
290
291    let source_mode = fs::metadata(mode_source).ok()?.permissions().mode();
292    let base = match fs::metadata(destination) {
293        Ok(existing) => existing.permissions().mode() & 0o777,
294        Err(_) => 0o600,
295    };
296    if source_mode & 0o111 == 0 {
297        return Some(base);
298    }
299    Some(base | 0o100)
300}

302/// Apply `mode` and report whether the file actually carries it afterwards. A
303/// filesystem without permission bits (exFAT, some CIFS mounts) either refuses
304/// the call or ignores it; either way the sync continues and the file counts as
305/// unchanged, instead of being offered again on every run.
306#[cfg(unix)]
307fn set_mode(path: &Path, mode: u32) -> bool {
308    use std::os::unix::fs::PermissionsExt;
309
310    if let Err(error) = fs::set_permissions(path, fs::Permissions::from_mode(mode)) {
311        log::warn!("Could not set permissions on {}: {error}", path.display());
312        return false;
313    }
314    let applied = fs::metadata(path).map(|m| m.permissions().mode() & 0o777);
315    applied.is_ok_and(|applied| applied == mode)
316}

318/// Whether `path` is missing an executable bit that `mode_source` has. Content
319/// comparison alone never notices a `chmod +x`, which would leave the bit stuck
320/// at whatever it was when the file was first copied.
321#[cfg(unix)]
322fn executable_bit_differs(path: &Path, mode_source: &Path) -> bool {
323    use std::os::unix::fs::PermissionsExt;
324
325    if is_symlink(path) {
326        return false;
327    }
328    let Some(wanted) = get_mode_for_copy(path, mode_source) else {
329        return false;
330    };
331    let Ok(metadata) = fs::metadata(path) else {
332        return false;
333    };
334    metadata.permissions().mode() & 0o777 != wanted
335}
336
337#[cfg(not(unix))]
338fn executable_bit_differs(_path: &Path, _mode_source: &Path) -> bool {
339    false
340}
341
342/// A chmod follows symlinks, so it would reach a file outside `~/.claude` that
343/// a write never touches: `write_atomic` replaces the link itself.
344#[cfg(unix)]
345fn is_symlink(path: &Path) -> bool {
346    fs::symlink_metadata(path).is_ok_and(|metadata| metadata.file_type().is_symlink())
347}
348
349/// Bring `path`'s executable bit in line with `mode_source`, reporting whether
350/// the file changed.
351#[cfg(unix)]
352fn align_executable_bit(path: &Path, mode_source: &Path) -> bool {
353    if !executable_bit_differs(path, mode_source) {
354        return false;
355    }
356    let Some(mode) = get_mode_for_copy(path, mode_source) else {
357        return false;
358    };
359    set_mode(path, mode)
360}
361
362#[cfg(not(unix))]
363fn align_executable_bit(_path: &Path, _mode_source: &Path) -> bool {
364    false
365}

367/// Copy every enabled artifact category into `<repo_root>/artifacts/`,
368/// classifying each file Added/Modified/Unchanged by byte comparison.
369/// Prompt history and memory indexes are union-merged into the repo copy
370/// instead of overwritten, config files are path-tokenized, and a file this
371/// machine previously synced and has since deleted is removed from the repo.
372pub fn push_artifacts(
373    claude_dir: &Path,
374    repo_root: &Path,
375    filter: &FilterConfig,
376) -> Result<ArtifactReport> {
377    let mut report = ArtifactReport::default();
378    let tokens = PathTokens::for_claude_dir(claude_dir);
379    let tracked_before = tracked::load(claude_dir, repo_root);
380    let mut tracked_now = TrackedPaths::new();
381
382    for desc in active_categories(filter) {
383        let mut counts = CategoryCounts::new(desc.id);
384
385        let mut held_back = Vec::new();
386        let files = collect(
387            desc,
388            claude_dir,
389            filter,
390            &mut counts.skipped,
391            &mut held_back,
392        )?;
393        let category_root = category_repo_root(desc, repo_root, filter);
394        let mut pushed: TrackedPaths = TrackedPaths::new();
395        // Files kept back for their size are still here: count them as
396        // present so the repo copy (if any) is neither deleted nor forgotten.
397        for rel in held_back {
398            if let Some(rel) = repo_relative(repo_root, &category_root.join(rel)) {
399                pushed.insert(rel);
400            }
401        }
402
403        for file in files {
404            let dest = category_root.join(&file.rel);
405
406            match desc.merge {
407                MergeStrategy::UnionJsonl => {
408                    let local_text = fs::read_to_string(&file.abs).unwrap_or_default();
409                    let existed = dest.is_file();
410                    let repo_text = if existed {
411                        fs::read_to_string(&dest).unwrap_or_default()
412                    } else {
413                        String::new()
414                    };
415                    let (merged, new_lines) = merge_history_lines(&repo_text, &local_text);
416                    counts.merged_entries += new_lines;
417                    if !existed {
418                        write_atomic(&dest, merged.as_bytes(), &file.abs)?;
419                        counts.added += 1;
420                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
421                    } else if merged != repo_text {
422                        write_atomic(&dest, merged.as_bytes(), &file.abs)?;
423                        counts.modified += 1;
424                        report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
425                    } else {
426                        counts.unchanged += 1;
427                    }
428                }
429                MergeStrategy::UnionMemoryIndex | MergeStrategy::RawOverwrite => {
430                    let read_bytes = fs::read(&file.abs).with_context(|| {
431                        format!("Failed to read artifact {}", file.abs.display())
432                    })?;
433                    let mut src_bytes = if desc.tokenize_paths {
434                        tokens.to_repo(&read_bytes)
435                    } else {
436                        read_bytes
437                    };
438                    let existed = dest.is_file();
439                    let unions_index =
440                        desc.merge == MergeStrategy::UnionMemoryIndex && is_memory_index(&file.rel);
441                    if existed && unions_index {
442                        let (merged, new_entries) =
443                            merge_memory_index(&fs::read(&dest)?, &src_bytes);
444                        counts.merged_entries += new_entries;
445                        src_bytes = merged;
446                    }
447
448                    if !existed {
449                        write_atomic(&dest, &src_bytes, &file.abs)?;
450                        counts.added += 1;
451                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
452                    } else if fs::read(&dest)? != src_bytes {
453                        write_atomic(&dest, &src_bytes, &file.abs)?;
454                        counts.modified += 1;
455                        report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
456                    } else {
457                        let realigned = align_executable_bit(&dest, &file.abs);
458                        if realigned {
459                            counts.modified += 1;
460                            report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
461                        } else {
462                            counts.unchanged += 1;
463                        }
464                    }
465                }
466            }
467
468            if let Some(rel) = repo_relative(repo_root, &dest) {
469                pushed.insert(rel);
470            }
471        }
472
473        if desc.mirror_deletes {
474            if source_is_present(desc, claude_dir) {
475                mark_category_synced(&category_root)?;
476                let removed =
477                    remove_from_repo(&tracked_before, &pushed, &category_root, repo_root)?;
478                counts.deleted += removed.len();
479                for path in &removed {
480                    report.record(desc.id, ArtifactChangeKind::Deleted, path);
481                }
482                tracked_now.extend(pushed);
483            } else {
484                // A category this machine does not have says nothing about
485                // what the others hold: leave the repo copy and the record.
486                log::info!(
487                    "Category {} is not present under {}; its repo copy is left untouched",
488                    desc.name,
489                    claude_dir.display()
490                );
491                tracked_now.extend(tracked_under(&tracked_before, &category_root, repo_root));
492            }
493        }
494
495        report.counts.push(counts);
496    }
497
498    if active_categories(filter).any(|desc| desc.mirror_deletes) {
499        tracked::save(claude_dir, repo_root, tracked_now)?;
500    }
501    Ok(report)
502}

504/// Marker file that keeps a category's repo directory present once its last
505/// real file is deleted.
506///
507/// Git does not track directories, so an emptied category would disappear and
508/// be read as "this repo has no such category", which pull must not delete
509/// for. The marker distinguishes an empty category from an absent one.
510pub const CATEGORY_MARKER: &str = ".synced";
511
512/// Keep the category's repo directory alive across an emptying push.
513fn mark_category_synced(category_root: &Path) -> Result<()> {
514    let marker = category_root.join(CATEGORY_MARKER);
515    if marker.is_file() {
516        return Ok(());
517    }
518    fs::create_dir_all(category_root)?;
519    fs::write(&marker, b"").with_context(|| format!("Failed to write {}", marker.display()))?;
520    Ok(())
521}

523/// A repo path as a `/`-separated string relative to the repository root.
524fn repo_relative(repo_root: &Path, path: &Path) -> Option<String> {
525    let rel = path.strip_prefix(repo_root).ok()?;
526    Some(rel.to_string_lossy().replace('\\', "/"))
527}

529/// Whether this machine has the source a category copies from. A `Files`
530/// category is always "present": its individual files are optional.
531fn source_is_present(desc: &CategoryDescriptor, claude_dir: &Path) -> bool {
532    match desc.source {
533        SourceSpec::Files(_) => true,
534        SourceSpec::Dir(dir) => claude_dir.join(dir).is_dir(),
535    }
536}

538/// The tracked paths that belong to one category's repo directory.
539fn tracked_under(tracked: &TrackedPaths, category_root: &Path, repo_root: &Path) -> Vec<String> {
540    let Some(prefix) = repo_relative(repo_root, category_root) else {
541        return Vec::new();
542    };
543    let prefix = format!("{prefix}/");
544    tracked
545        .iter()
546        .filter(|path| path.starts_with(&prefix))
547        .cloned()
548        .collect()
549}

551/// Delete the repo copies of files this machine synced before and no longer
552/// has. Returns the removed paths, relative to the category root.
553fn remove_from_repo(
554    tracked_before: &TrackedPaths,
555    pushed: &TrackedPaths,
556    category_root: &Path,
557    repo_root: &Path,
558) -> Result<Vec<PathBuf>> {
559    let mut removed = Vec::new();
560    let category_prefix = repo_relative(repo_root, category_root).unwrap_or_default();
561    for gone in tracked_under(tracked_before, category_root, repo_root) {
562        if pushed.contains(&gone) {
563            continue;
564        }
565        let path = repo_root.join(&gone);
566        if path.is_file() {
567            fs::remove_file(&path)
568                .with_context(|| format!("Failed to remove {}", path.display()))?;
569            let category_relative = Path::new(&gone)
570                .strip_prefix(&category_prefix)
571                .map(Path::to_path_buf)
572                .unwrap_or_default();
573            removed.push(category_relative);
574        }
575    }
576    Ok(removed)
577}

591/// Read-only classification of an artifact pull, computed BEFORE any write so
592/// the caller can snapshot the exact set of files that will change.
593#[derive(Debug, Default)]
594pub struct PullPlan {
595    /// Local file exists and repo bytes differ: remote wins after snapshot.
596    pub overwrites: Vec<PlannedWrite>,
597    /// No local file yet: created, and recorded for deletion on undo.
598    pub creates: Vec<PlannedWrite>,
599    /// Union-merge targets whose local file would gain lines.
600    pub unions: Vec<PlannedWrite>,
601    /// Local files whose content already matches but whose executable bit does
602    /// not: a `chmod +x` elsewhere, with nothing to rewrite.
603    pub mode_fixes: Vec<PlannedWrite>,
604    /// Local files this machine synced before that the repo no longer has.
605    pub deletes: Vec<PlannedDelete>,
606    pub unchanged: usize,
607    /// Repo files refused (denied names, unsafe paths).
608    pub skipped: usize,
609    /// Repo files whose project this machine has no destination for, grouped
610    /// by the project directory they came from, so the caller can warn once
611    /// per project instead of once per file.
612    pub unmapped_projects: crate::project_map::SkippedByProject,
613    /// This machine's path tokens, so applying renders repo bytes the same way
614    /// planning compared them.
615    pub tokens: PathTokens,
616    /// Where to record what this machine holds once the plan is applied.
617    pub claude_dir: PathBuf,
618    pub repo_root: PathBuf,
619    /// The configured external merge tool, offered when a file differs.
620    pub merge_tool: String,
621    /// The repo paths this machine will hold afterwards, for the next pull to
622    /// tell a deletion from a file it never had.
623    pub tracked_after: TrackedPaths,
624    /// Whether any active category mirrors deletions. When none does, the
625    /// record is left untouched rather than emptied.
626    pub tracks_deletions: bool,
627}

639impl PullPlan {
640    /// True when applying the plan would write nothing.
641    pub fn is_empty(&self) -> bool {
642        self.overwrites.is_empty()
643            && self.creates.is_empty()
644            && self.unions.is_empty()
645            && self.mode_fixes.is_empty()
646            && self.deletes.is_empty()
647    }
648
649    /// Existing local files the caller must snapshot before applying
650    /// (overwritten raw files, union-merged files and deletions — a snapshot is
651    /// what makes `undo pull` able to bring a deleted file back).
652    ///
653    /// The record of what this machine last synced is included whenever the
654    /// pull would rewrite it, so undoing a pull restores the files *and* the
655    /// record that describes them.
656    pub fn paths_to_snapshot(&self) -> Vec<PathBuf> {
657        let record = tracked::record_path(&self.claude_dir);
658        self.overwrites
659            .iter()
660            .chain(self.unions.iter())
661            .map(|w| w.local_path.clone())
662            .chain(self.deletes.iter().map(|d| d.local_path.clone()))
663            .chain((self.tracks_deletions && record.is_file()).then_some(record))
664            .collect()
665    }
666
667    /// Local paths this pull will create; recording them as a snapshot's
668    /// `deleted_files` makes undo remove them again.
669    pub fn created_paths(&self) -> Vec<String> {
670        self.creates
671            .iter()
672            .map(|w| w.local_path.to_string_lossy().to_string())
673            .collect()
674    }
675}

677/// Enumerate one category's files as stored in the sync repository, returning
678/// (absolute repo path, path relative to the category subdir). Denied and
679/// unsafe paths are refused here, so nothing below ever sees them.
680fn collect_repo_files(
681    desc: &CategoryDescriptor,
682    repo_root: &Path,
683    filter: &FilterConfig,
684    skipped: &mut usize,
685) -> Vec<(PathBuf, PathBuf)> {
686    let category_root = category_repo_root(desc, repo_root, filter);
687    if !category_root.is_dir() {
688        return Vec::new();
689    }
690
691    let mut files = Vec::new();
692    for entry in walkdir::WalkDir::new(&category_root)
693        .follow_links(false)
694        .into_iter()
695        .filter_map(|e| e.ok())
696    {
697        if !entry.file_type().is_file() {
698            continue;
699        }
700        let abs = entry.path();
701        let rel = abs
702            .strip_prefix(&category_root)
703            .unwrap_or(abs)
704            .to_path_buf();
705        if extension_excluded(desc, &rel) {
706            continue;
707        }
708        // The marker exists to keep the directory, and belongs to no machine.
709        if rel == Path::new(CATEGORY_MARKER) {
710            continue;
711        }
712        if is_unsafe_rel_path(&rel) || is_denied(&rel) {
713            log::warn!(
714                "Refusing denied/unsafe artifact from sync repo: {}",
715                abs.display()
716            );
717            *skipped += 1;
718            continue;
719        }
720        files.push((abs.to_path_buf(), rel));
721    }
722    files
723}

725/// Why a repo file has no local destination. The caller reports the two cases
726/// differently: an unmapped project is one warning for all of its files, an
727/// unrestorable path is a refusal worth its own line.
728enum SkipReason {
729    /// This machine has no directory for the named sync-repo project.
730    UnmappedProject(String),
731    /// Nothing in the category can receive this path.
732    NotRestorable,
733}
734
735/// Map a category-relative repo path back to its absolute local destination
736/// under `~/.claude`, or the reason it has none.
737fn local_destination(
738    desc: &CategoryDescriptor,
739    claude_dir: &Path,
740    rel: &Path,
741    filter: &FilterConfig,
742) -> Result<PathBuf, SkipReason> {
743    match desc.source {
744        // File lists are stored flat in the repo; restore to the listed
745        // location whose file name matches. An unlisted name has NO valid
746        // destination — the allowlist must hold on pull as well as push, or a
747        // poisoned repo could plant arbitrary files at the top of ~/.claude.
748        SourceSpec::Files(list) => list
749            .iter()
750            .find(|entry| Path::new(entry).file_name() == rel.file_name())
751            .map(|entry| claude_dir.join(entry))
752            .ok_or(SkipReason::NotRestorable),
753        SourceSpec::Dir(dir) => {
754            if desc.dest == DestRoot::SessionTree {
755                // Repo path is <project-dir>/<rest>; resolve the leading
756                // component the way session pull does.
757                let split = crate::project_map::split_project_path(rel);
758                let (project, inside_project) = split.ok_or(SkipReason::NotRestorable)?;
759                let projects_dir = claude_dir.join(dir);
760                let local_project =
761                    crate::project_map::local_project_dir(filter, &projects_dir, project)
762                        .ok_or_else(|| SkipReason::UnmappedProject(project.to_string()))?;
763                return Ok(local_project.join(inside_project));
764            }
765            Ok(claude_dir.join(dir).join(rel))
766        }
767    }
768}

770/// The bytes a repo file becomes on this machine before any merge: rendered
771/// back from tokens for config categories, verbatim otherwise.
772fn machine_bytes(
773    desc: &CategoryDescriptor,
774    tokens: &PathTokens,
775    repo_path: &Path,
776) -> Result<Vec<u8>> {
777    let bytes = fs::read(repo_path)
778        .with_context(|| format!("Failed to read artifact {}", repo_path.display()))?;
779    if desc.tokenize_paths {
780        return Ok(tokens.to_machine(&bytes));
781    }
782    Ok(bytes)
783}

788        .iter()
789        .find(|d| d.id == id)
790        .expect("every CategoryId has a registry row")

793/// Classify what a pull would write, without writing. Remote (repo) bytes win
794/// for raw categories; union targets are compared against local ∪ remote; a
795/// file this machine synced before and the repo no longer has is a deletion.
796pub fn plan_pull(claude_dir: &Path, repo_root: &Path, filter: &FilterConfig) -> Result<PullPlan> {
797    let mut plan = PullPlan {
798        tokens: PathTokens::for_claude_dir(claude_dir),
799        claude_dir: claude_dir.to_path_buf(),
800        repo_root: repo_root.to_path_buf(),
801        merge_tool: filter.merge_tool.clone(),
802        tracks_deletions: active_categories(filter).any(|desc| desc.mirror_deletes),
803        ..Default::default()
804    };
805    let tracked_before = tracked::load(claude_dir, repo_root);
806
807    for desc in active_categories(filter) {
808        let category_root = category_repo_root(desc, repo_root, filter);
809        let mut present: TrackedPaths = TrackedPaths::new();
810
811        for (repo_path, rel) in collect_repo_files(desc, repo_root, filter, &mut plan.skipped) {
812            let local_path = match local_destination(desc, claude_dir, &rel, filter) {
813                Ok(local_path) => local_path,
814                Err(SkipReason::UnmappedProject(project)) => {
815                    plan.skipped += 1;
816                    plan.unmapped_projects
817                        .entry(project)
818                        .or_default()
819                        .push(repo_path);
820                    continue;
821                }
822                Err(SkipReason::NotRestorable) => {
823                    plan.skipped += 1;
824                    log::warn!(
825                        "Skipping {} (not a file the {} category restores)",
826                        repo_path.display(),
827                        desc.name
828                    );
829                    continue;
830                }
831            };
832            if desc.mirror_deletes {
833                if let Some(tracked_path) = repo_relative(repo_root, &repo_path) {
834                    present.insert(tracked_path);
835                }
836            }
837            let write = PlannedWrite {
838                category: desc.id,
839                local_path: local_path.clone(),
840                repo_path: repo_path.clone(),
841                category_path: rel.clone(),
842            };
843
844            match desc.merge {
845                MergeStrategy::UnionJsonl => {
846                    if !local_path.is_file() {
847                        plan.creates.push(write);
848                        continue;
849                    }
850                    let local_text = fs::read_to_string(&local_path).unwrap_or_default();
851                    let repo_text = fs::read_to_string(&repo_path).unwrap_or_default();
852                    let (merged, _) = merge_history_lines(&local_text, &repo_text);
853                    if merged != local_text {
854                        plan.unions.push(write);
855                    } else {
856                        plan.unchanged += 1;
857                    }
858                }
859                MergeStrategy::UnionMemoryIndex if is_memory_index(&rel) => {
860                    if !local_path.is_file() {
861                        plan.creates.push(write);
862                        continue;
863                    }
864                    let local_bytes = fs::read(&local_path)?;
865                    let repo_bytes = machine_bytes(desc, &plan.tokens, &repo_path)?;
866                    let (merged, _) = merge_memory_index(&local_bytes, &repo_bytes);
867                    if merged != local_bytes {
868                        plan.unions.push(write);
869                    } else {
870                        plan.unchanged += 1;
871                    }
872                }
873                MergeStrategy::UnionMemoryIndex | MergeStrategy::RawOverwrite => {
874                    if !local_path.is_file() {
875                        plan.creates.push(write);
876                    } else if fs::read(&local_path)?
877                        != machine_bytes(desc, &plan.tokens, &repo_path)?
878                    {
879                        plan.overwrites.push(write);
880                    } else {
881                        let missing_executable_bit =
882                            executable_bit_differs(&local_path, &repo_path);
883                        if missing_executable_bit {
884                            plan.mode_fixes.push(write);
885                        } else {
886                            plan.unchanged += 1;
887                        }
888                    }
889                }
890            }
891        }
892
893        if !desc.mirror_deletes {
894            continue;
895        }
896
897        // Mirror of the push-side guard: a category the repo does not have —
898        // an older or rewound branch — is not "everything was deleted".
899        if !category_root.is_dir() {
900            log::info!(
901                "Category {} is not present in {}; local files are left alone",
902                desc.name,
903                repo_root.display()
904            );
905            plan.tracked_after
906                .extend(tracked_under(&tracked_before, &category_root, repo_root));
907            continue;
908        }
909
910        for gone in tracked_under(&tracked_before, &category_root, repo_root) {
911            if present.contains(&gone) {
912                continue;
913            }
914            let rel = Path::new(&gone)
915                .strip_prefix(repo_relative(repo_root, &category_root).unwrap_or_default())
916                .map(Path::to_path_buf)
917                .unwrap_or_default();
918            // The same refusal `collect_repo_files` applies: a record must
919            // never point a deletion outside its category.
920            if is_unsafe_rel_path(&rel) || is_denied(&rel) {
921                log::warn!("Refusing denied/unsafe tracked path: {gone}");
922                plan.skipped += 1;
923                continue;
924            }
925            let Ok(local_path) = local_destination(desc, claude_dir, &rel, filter) else {
926                continue;
927            };
928            if local_path.is_file() {
929                plan.deletes.push(PlannedDelete {
930                    category: desc.id,
931                    local_path,
932                    category_path: rel,
933                });
934            }
935        }
936        plan.tracked_after.extend(present);
937    }
938
939    Ok(plan)
940}

942/// Apply a pull plan: create missing files, overwrite differing ones (remote
943/// wins), union-merge prompt history and memory indexes, and remove files the
944/// repo no longer has. Under `interactive` in a terminal, each overwrite asks
945/// for per-file confirmation — with the configured merge tool as an option —
946/// and each deletion asks for confirmation; declined files count as skipped.
947pub fn apply_pull(plan: &PullPlan, interactive: bool) -> Result<ArtifactReport> {
948    use std::collections::HashMap;
949
950    let mut by_category: HashMap<CategoryId, CategoryCounts> = HashMap::new();
951    fn counts_for(
952        map: &mut HashMap<CategoryId, CategoryCounts>,
953        id: CategoryId,
954    ) -> &mut CategoryCounts {
955        map.entry(id)
956            .or_insert_with(move || CategoryCounts::new(id))
957    }
958
959    let mut report = ArtifactReport::default();
960    let prompt_overwrites = interactive && crate::interactive_conflict::is_interactive();
961
962    for write in &plan.creates {
963        let bytes = machine_bytes(descriptor(write.category), &plan.tokens, &write.repo_path)?;
964        write_atomic(&write.local_path, &bytes, &write.repo_path)?;
965        counts_for(&mut by_category, write.category).added += 1;
966        report.record(
967            write.category,
968            ArtifactChangeKind::Added,
969            &write.category_path,
970        );
971    }
972
973    for write in &plan.overwrites {
974        let bytes = machine_bytes(descriptor(write.category), &plan.tokens, &write.repo_path)?;
975        let bytes = if prompt_overwrites {
976            match crate::merge_tool::resolve_overwrite(&plan.merge_tool, &write.local_path, &bytes)?
977            {
978                Some(resolved) => resolved,
979                None => {
980                    counts_for(&mut by_category, write.category).skipped += 1;
981                    continue;
982                }
983            }
984        } else {
985            bytes
986        };
987        write_atomic(&write.local_path, &bytes, &write.repo_path)?;
988        counts_for(&mut by_category, write.category).modified += 1;
989        report.record(
990            write.category,
991            ArtifactChangeKind::Modified,
992            &write.category_path,
993        );
994    }
995
996    for write in &plan.unions {
997        let desc = descriptor(write.category);
998        let repo_bytes = machine_bytes(desc, &plan.tokens, &write.repo_path)?;
999        let local_bytes = fs::read(&write.local_path).unwrap_or_default();
1000        let (merged, new_entries) = match desc.merge {
1001            MergeStrategy::UnionMemoryIndex => merge_memory_index(&local_bytes, &repo_bytes),
1002            _ => {
1003                let (text, lines) = merge_history_lines(
1004                    &String::from_utf8_lossy(&local_bytes),
1005                    &String::from_utf8_lossy(&repo_bytes),
1006                );
1007                (text.into_bytes(), lines)
1008            }
1009        };
1010        write_atomic(&write.local_path, &merged, &write.repo_path)?;
1011        let counts = counts_for(&mut by_category, write.category);
1012        counts.modified += 1;
1013        counts.merged_entries += new_entries;
1014        report.record(
1015            write.category,
1016            ArtifactChangeKind::Modified,
1017            &write.category_path,
1018        );
1019    }
1020
1021    for write in &plan.mode_fixes {
1022        if prompt_overwrites && !confirm_executable(&write.local_path) {
1023            counts_for(&mut by_category, write.category).skipped += 1;
1024            continue;
1025        }
1026        let realigned = align_executable_bit(&write.local_path, &write.repo_path);
1027        if realigned {
1028            counts_for(&mut by_category, write.category).modified += 1;
1029            report.record(
1030                write.category,
1031                ArtifactChangeKind::Modified,
1032                &write.category_path,
1033            );
1034        }
1035    }
1036
1037    for delete in &plan.deletes {
1038        if prompt_overwrites && !confirm_deletion(&delete.local_path) {
1039            counts_for(&mut by_category, delete.category).skipped += 1;
1040            continue;
1041        }
1042        if delete.local_path.is_file() {
1043            fs::remove_file(&delete.local_path)
1044                .with_context(|| format!("Failed to remove {}", delete.local_path.display()))?;
1045        }
1046        counts_for(&mut by_category, delete.category).deleted += 1;
1047        report.record(
1048            delete.category,
1049            ArtifactChangeKind::Deleted,
1050            &delete.category_path,
1051        );
1052    }
1053
1054    if plan.tracks_deletions {
1055        tracked::save(
1056            &plan.claude_dir,
1057            &plan.repo_root,
1058            plan.tracked_after.clone(),
1059        )?;
1060    }
1061
1062    report.counts = by_category.into_values().collect();
1063    report.counts.sort_by_key(|c| c.category as usize);
1064    Ok(report)
1065}

1067/// Ask before making a local file runnable, since a hook runs on its own.
1068fn confirm_executable(local_path: &Path) -> bool {
1069    let file_name = local_path
1070        .file_name()
1071        .map(|n| n.to_string_lossy().to_string())
1072        .unwrap_or_else(|| local_path.display().to_string());
1073    inquire::Confirm::new(&format!(
1074        "'{file_name}' is executable on another machine. Make it executable here too?"
1075    ))
1076    .with_default(true)
1077    .with_help_message("Declining leaves the file as it is")
1078    .prompt()
1079    .unwrap_or(false)
1080}
1081
1082/// Ask before removing a local file the sync repo no longer has.
1083fn confirm_deletion(local_path: &Path) -> bool {
1084    let file_name = local_path
1085        .file_name()
1086        .map(|n| n.to_string_lossy().to_string())
1087        .unwrap_or_else(|| local_path.display().to_string());
1088    inquire::Confirm::new(&format!(
1089        "'{file_name}' was deleted on another machine. Delete it here too?"
1090    ))
1091    .with_default(true)
1092    .with_help_message("Declining keeps the local file; it is pushed back on the next push")
1093    .prompt()
1094    .unwrap_or(false)
1095}

1097/// Globs for the managed ignore block: defense-in-depth behind the code-level
1098/// denylist, in case files land in the repo by hand or via other tools.
1099const IGNORE_GLOBS: &[&str] = &[
1100    ".credentials.json",
1101    "settings.local.json",
1102    ".claude.json",
1103    "*.pem",
1104    "*.key",
1105    ".env*",
1106    "daemon*",
1107    "stats-cache.json",
1108    ".last-update-result.json",
1109    "mcp-needs-auth-cache.json",
1110    "shell-snapshots/",
1111    "session-env/",
1112    "file-history/",
1113    "paste-cache/",
1114    "statsig/",
1115    "backups/",
1116    "sessions/",
1117    "**/cache/",
1118    "**/debug/",
1119];
1120
1121const IGNORE_BLOCK_START: &str = "# >>> claude-code-sync managed block — do not edit inside";
1122const IGNORE_BLOCK_END: &str = "# <<< claude-code-sync managed block";
1123
1124/// Build the full managed block for one backend.
1125fn ignore_block(backend: Backend) -> String {
1126    let mut block = String::new();
1127    block.push_str(IGNORE_BLOCK_START);
1128    block.push('\n');
1129    if backend == Backend::Mercurial {
1130        block.push_str("syntax: glob\n");
1131    }
1132    for glob in IGNORE_GLOBS {
1133        block.push_str(glob);
1134        block.push('\n');
1135    }
1136    block.push_str(IGNORE_BLOCK_END);
1137    block.push('\n');
1138    block
1139}

1141/// Write the managed never-sync ignore block into the sync repository's
1142/// ignore file for the given backend. Idempotent; preserves user content
1143/// outside the block. Returns whether the file changed.
1144pub fn ensure_ignore_files(repo_root: &Path, backend: Backend) -> Result<bool> {
1145    let file_name = match backend {
1146        Backend::Git => ".gitignore",
1147        Backend::Mercurial => ".hgignore",
1148    };
1149    let path = repo_root.join(file_name);
1150    let existing = if path.is_file() {
1151        fs::read_to_string(&path)?
1152    } else {
1153        String::new()
1154    };
1155
1156    let block = ignore_block(backend);
1157
1158    let updated = if let (Some(start), Some(end)) = (
1159        existing.find(IGNORE_BLOCK_START),
1160        existing.find(IGNORE_BLOCK_END),
1161    ) {
1162        // Replace the existing block in place.
1163        let end = end + IGNORE_BLOCK_END.len();
1164        // Include the trailing newline of the old block if present.
1165        let end = if existing[end..].starts_with('\n') {
1166            end + 1
1167        } else {
1168            end
1169        };
1170        format!("{}{}{}", &existing[..start], block, &existing[end..])
1171    } else if existing.is_empty() {
1172        block
1173    } else {
1174        let sep = if existing.ends_with('\n') {
1175            "\n"
1176        } else {
1177            "\n\n"
1178        };
1179        format!("{existing}{sep}{block}")
1180    };
1181
1182    if updated == existing {
1183        return Ok(false);
1184    }
1185    fs::write(&path, updated)?;
1186    Ok(true)
1187}
