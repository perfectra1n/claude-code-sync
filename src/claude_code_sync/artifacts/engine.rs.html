42/// Whether a file under this merge strategy records a synced base — the
43/// ONE rule, consulted by push and pull.
44///
45/// All user-editable files need a base: the local-edit-protection
46/// contract (issue #103) requires a recorded base so the next pull
47/// can detect a local edit and route to `kept_local` instead of
48/// silently clobbering the user's changes. For union-merge strategies
49/// the base is the local snapshot at push time, and the apply
50/// re-records after every merge so the next pull sees local == base
51/// and skips the merge when nothing new arrived. Prompt history
52/// (`UnionJsonl`) is append-only and does not need per-line
53/// protection.
54pub(crate) fn records_base(strategy: MergeStrategy, _rel: &Path) -> bool {
55    match strategy {
56        MergeStrategy::RawOverwrite => true,
57        MergeStrategy::UnionJsonl => false,
58        MergeStrategy::UnionMemoryIndex => true,
59    }
60}

62/// The plan's one size-gate triage: outside the limit, or unreadable — a
63/// stat failure must never read as size 0 and wave an oversized (or
64/// broken) file past the guard. Shared by the overwrite arm and the
65/// gone-scan delete arm so the fail-closed semantics cannot drift apart.
66fn outside_size_limit(local_path: &Path, filter: &FilterConfig) -> bool {
67    metadata_outside_limit(fs::metadata(local_path), filter.max_file_size_bytes)
68}

70/// The triage's core, shared with the repo-side scan (which already
71/// holds the metadata): fail closed — a stat failure must never read as
72/// size 0.
73fn metadata_outside_limit<E>(metadata: Result<fs::Metadata, E>, max: u64) -> bool {
74    metadata.map_or(true, |m| m.len() > max)
75}

77/// What a fresh read of a local file says about its edit status relative
78/// to the recorded base — the ONE triage, shared by the plan's delete
79/// arm and both apply arms (overwrites and deletes).
80#[derive(Debug, PartialEq, Eq)]
81enum FreshRead {
82    /// Matches the base, or no base recorded (legacy remote-wins): safe
83    /// to overwrite or delete. No base means no read at all.
84    Clean,
85    /// Edited here since the last sync: must be kept.
86    Dirty,
87    /// Gone from disk since the plan: proceed as if absent.
88    Vanished,
89    /// Present but unreadable: keep and report, never guess and never
90    /// fail the whole sync over one bad file.
91    Unreadable,
92}
93
94fn dirty_status(recorded: Option<&String>, path: &Path) -> FreshRead {
95    let Some(base) = recorded else {
96        return FreshRead::Clean;
97    };
98    match fs::read(path) {
99        Ok(bytes) if bases::is_dirty(Some(base), &bytes) => FreshRead::Dirty,
100        Ok(_) => FreshRead::Clean,
101        Err(e) if e.kind() == std::io::ErrorKind::NotFound => FreshRead::Vanished,
102        Err(e) => {
103            log::warn!(
104                "Keeping {} (unreadable, edited-status unknown): {e}",
105                path.display()
106            );
107            FreshRead::Unreadable
108        }
109    }
110}

112/// F3-class synthetic-Dirty guard: a missing base (None recorded,
113/// concurrent sync pruned it) is treated as "potentially edited"
114/// — silently destroying the file would discard a possibly-edited
115/// user's edit, the exact loss the protect-local-edits contract
116/// exists to prevent. Two arms (kept_local_deletes, deletes)
117/// share this exact predicate; this helper centralises it.
118fn is_potentially_dirty(recorded: Option<&String>, path: &Path) -> bool {
119    recorded.is_none() || dirty_status(recorded, path) == FreshRead::Dirty
120}

122/// Plan-time F3-class signal: a file with no recorded base AND a
123/// tracked entry is "we synced this before, the base got pruned
124/// out from under us" — a potentially-edited file. Fresh installs
125/// (no tracked entry) must still remote-win.
126///
127/// This is the plan-time analogue of `is_potentially_dirty` for
128/// arms that operate on already-read bytes (overwrites): it
129/// substitutes the tracked_before set for the file-read step,
130/// because the file content is in scope at the call site rather
131/// than on disk in a known state.
132fn f3_class_potentially_dirty(
133    base_hashes: &BaseHashes,
134    tracked_before: &TrackedPaths,
135    rel: &str,
136) -> bool {
137    base_hashes.get(rel).is_none() && tracked_before.contains(rel)
138}

140/// Whether the repository still holds the version this machine last synced,
141/// so a local difference is this machine's change alone and the push that
142/// follows publishes it. The protection sentinel records no real version and
143/// never matches, and neither does a held base (see [`bases::held_base`]).
144fn repo_unchanged_since_sync(recorded: Option<&String>, repo_bytes: &[u8]) -> bool {
145    recorded.is_some_and(|base| {
146        bases::is_common_ancestor(base) && *base == bases::hash_bytes(repo_bytes)
147    })
148}

150/// Whether a missing local file is a deletion made here: this machine synced
151/// it, the repository still holds that version, and the category mirrors
152/// deletions. Such a file is left deleted for the push to remove from the
153/// repository, instead of being recreated. A whole missing category folder is
154/// a machine that never had it, not a deletion.
155fn deleted_here_since_sync(
156    desc: &CategoryDescriptor,
157    claude_dir: &Path,
158    tokens: &PathTokens,
159    base_hashes: &BaseHashes,
160    repo_root: &Path,
161    repo_path: &Path,
162) -> bool {
163    if !desc.mirror_deletes || !source_is_present(desc, claude_dir) {
164        return false;
165    }
166    let recorded = repo_relative(repo_root, repo_path).and_then(|rel| base_hashes.get(&rel));
167    recorded.is_some()
168        && machine_bytes(desc, tokens, repo_path)
169            .is_ok_and(|repo_bytes| repo_unchanged_since_sync(recorded, &repo_bytes))
170}

255/// How one artifact file changed during a push or pull.
256#[derive(Debug, Clone, Copy, PartialEq, Eq)]
257pub enum ArtifactChangeKind {
258    Added,
259    Modified,
260    Deleted,
261    /// Left untouched because it was edited locally since the last sync;
262    /// a `push` is what publishes it.
263    KeptLocal,
264    /// The local file is gone (vanished mid-pull, or user-confirmed
265    /// during the prompt) and the repo's deletion is being honored.
266    /// A `push` must NOT recreate the file in the repo: the
267    /// removal has been accepted, and the next pull's
268    /// gone-from-both-sides pass is what clears the record entry.
269    KeptLocalDelete,
270    /// The push refused to re-publish a file the remote lost since the
271    /// last sync (adversarial review 2026-10-09, 5ff1d62 hole). The
272    /// local copy is preserved on disk; the file path is in
273    /// `ArtifactChange.path` (repo-relative). The user-facing escape
274    /// hatch is `claude-code-sync push --resurrect <path>` (see
275    /// `engine::prepare_resurrection`).
276    HeldBackRemoteLost,
277    /// A pull planned to write this file but did not (it appeared,
278    /// vanished or became unreadable while the pull ran, had no backup,
279    /// or a write failed). The local side is then not a decision about
280    /// the repository version, so the push that follows must leave the
281    /// path alone — publishing it, or mirroring its absence as a
282    /// deletion, could overwrite a change another machine made. Retried
283    /// by the next pull.
284    Unfinished,
285}

407/// One collected artifact file: its absolute source under `~/.claude` and its
408/// destination path relative to the category's repo subdirectory.
409struct CollectedFile {
410    abs: PathBuf,
411    rel: PathBuf,
412}
413
414/// Enumerate a category's files on disk. Missing sources yield an empty list;
415/// denied paths and oversized files are skipped (the latter counted).
416///
417/// An oversized file still exists on this machine, so its category-relative
418/// path goes into `held_back`: deletion mirroring must not read "not pushed
419/// this time" as "deleted here" and remove every machine's copy.
420fn collect(
421    desc: &CategoryDescriptor,
422    claude_dir: &Path,
423    filter: &FilterConfig,
424    skipped: &mut usize,
425    held_back: &mut Vec<PathBuf>,
426) -> Result<Vec<CollectedFile>> {
427    let mut files = Vec::new();
428
429    match desc.source {
430        SourceSpec::Files(list) => {
431            for entry in list {
432                let claude_rel = Path::new(entry);
433                if is_denied(claude_rel) {
434                    *skipped += 1;
435                    continue;
436                }
437                let abs = claude_dir.join(entry);
438                if !abs.is_file() {
439                    continue;
440                }
441                // The one shared fail-closed triage: the highest-impact
442                // config files must not be the one place the limit sleeps.
443                if outside_size_limit(&abs, filter) {
444                    log::warn!("Skipping {} (exceeds max_file_size_bytes)", abs.display());
445                    *skipped += 1;
446                    held_back.push(
447                        claude_rel
448                            .file_name()
449                            .map(PathBuf::from)
450                            .unwrap_or_else(|| claude_rel.to_path_buf()),
451                    );
452                    continue;
453                }
454                let rel = claude_rel
455                    .file_name()
456                    .map(PathBuf::from)
457                    .unwrap_or_else(|| claude_rel.to_path_buf());
458                files.push(CollectedFile { abs, rel });
459            }
460        }
461        SourceSpec::Dir(dir) => {
462            let base = claude_dir.join(dir);
463            if !base.is_dir() {
464                return Ok(files);
465            }
466            // Resolved once per project: in name-only mode it reads a transcript.
467            let mut project_names: std::collections::HashMap<String, String> =
468                std::collections::HashMap::new();
469            for entry in walkdir::WalkDir::new(&base)
470                .follow_links(false)
471                .into_iter()
472                .filter_map(|e| e.ok())
473            {
474                if !entry.file_type().is_file() {
475                    continue;
476                }
477                let abs = entry.path();
478                // Deny rules match against the ~/.claude-relative path so a
479                // component like `cache/` is caught wherever it appears.
480                let claude_rel = abs.strip_prefix(claude_dir).unwrap_or(abs);
481                if is_denied(claude_rel) {
482                    *skipped += 1;
483                    continue;
484                }
485                if extension_excluded(desc, abs) {
486                    continue;
487                }
488                let mut rel = abs.strip_prefix(&base).unwrap_or(abs).to_path_buf();
489                // Attachments take the project's repo-side directory name,
490                // mirroring session layout.
491                if desc.dest == DestRoot::SessionTree {
492                    let mut parts = rel.components();
493                    let Some(encoded) = parts.next().and_then(|c| c.as_os_str().to_str()) else {
494                        *skipped += 1;
495                        continue;
496                    };
497                    let project = project_names
498                        .entry(encoded.to_string())
499                        .or_insert_with(|| {
500                            crate::project_map::repo_dir_name_in(filter, &base, encoded)
501                        })
502                        .clone();
503                    rel = Path::new(&project).join(parts.as_path());
504                }
505                // The one shared fail-closed triage, like the Files branch.
506                if outside_size_limit(abs, filter) {
507                    log::warn!("Skipping {} (exceeds max_file_size_bytes)", abs.display());
508                    *skipped += 1;
509                    held_back.push(rel);
510                    continue;
511                }
512                files.push(CollectedFile {
513                    abs: abs.to_path_buf(),
514                    rel,
515                });
516            }
517        }
518    }
519
520    Ok(files)
521}

523/// Write `content` to `path` via a same-directory temp file + atomic rename,
524/// so a reader (or a crash) never sees a half-written file. `mode_source` is
525/// the file whose executable bit the result adopts.
526#[cfg_attr(not(unix), allow(unused_variables))]
527fn write_atomic(path: &Path, content: &[u8], mode_source: &Path) -> Result<()> {
528    let parent = path
529        .parent()
530        .with_context(|| format!("No parent directory for {}", path.display()))?;
531    fs::create_dir_all(parent)?;
532    let tmp = tempfile::NamedTempFile::new_in(parent)?;
533    fs::write(tmp.path(), content)?;
534    // Durability: without fsync, `rename(2)` is not durable across power
535    // loss — the data can be in the page cache but not on disk when the
536    // machine loses power. The file fsync flushes the data; the directory
537    // fsync (Unix only) commits the rename entry itself. Both required
538    // for crash-and-power-loss-safe writes.
539    tmp.as_file().sync_all()?;
540    #[cfg(unix)]
541    {
542        if let Some(mode) = get_mode_for_copy(path, mode_source) {
543            set_mode(tmp.path(), mode);
544        }
545    }
546    tmp.persist(path)
547        .with_context(|| format!("Failed to persist {}", path.display()))?;
548    #[cfg(unix)]
549    {
550        if let Ok(dir) = fs::File::open(parent) {
551            let _ = dir.sync_all();
552        }
553    }
554    Ok(())
555}

557/// The mode a copy of `mode_source` should have at `destination`: what the
558/// destination already has, plus the owner's executable bit when the source is
559/// executable. Granting only: a repository written before this bit was synced
560/// holds every file non-executable, and a pull from it must not disarm the
561/// scripts on this machine.
562#[cfg(unix)]
563fn get_mode_for_copy(destination: &Path, mode_source: &Path) -> Option<u32> {
564    use std::os::unix::fs::PermissionsExt;
565
566    let source_mode = fs::metadata(mode_source).ok()?.permissions().mode();
567    let base = match fs::metadata(destination) {
568        Ok(existing) => existing.permissions().mode() & 0o777,
569        Err(_) => 0o600,
570    };
571    if source_mode & 0o111 == 0 {
572        return Some(base);
573    }
574    Some(base | 0o100)
575}

577/// Apply `mode` and report whether the file actually carries it afterwards. A
578/// filesystem without permission bits (exFAT, some CIFS mounts) either refuses
579/// the call or ignores it; either way the sync continues and the file counts as
580/// unchanged, instead of being offered again on every run.
581#[cfg(unix)]
582fn set_mode(path: &Path, mode: u32) -> bool {
583    use std::os::unix::fs::PermissionsExt;
584
585    if let Err(error) = fs::set_permissions(path, fs::Permissions::from_mode(mode)) {
586        log::warn!("Could not set permissions on {}: {error}", path.display());
587        return false;
588    }
589    let applied = fs::metadata(path).map(|m| m.permissions().mode() & 0o777);
590    applied.is_ok_and(|applied| applied == mode)
591}

593/// Whether `path` is missing an executable bit that `mode_source` has. Content
594/// comparison alone never notices a `chmod +x`, which would leave the bit stuck
595/// at whatever it was when the file was first copied.
596#[cfg(unix)]
597fn executable_bit_differs(path: &Path, mode_source: &Path) -> bool {
598    use std::os::unix::fs::PermissionsExt;
599
600    if is_symlink(path) {
601        return false;
602    }
603    let Some(wanted) = get_mode_for_copy(path, mode_source) else {
604        return false;
605    };
606    let Ok(metadata) = fs::metadata(path) else {
607        return false;
608    };
609    metadata.permissions().mode() & 0o777 != wanted
610}

642/// Copy every enabled artifact category into `<repo_root>/artifacts/`,
643/// classifying each file Added/Modified/Unchanged by byte comparison.
644/// Prompt history and memory indexes are union-merged into the repo copy
645/// instead of overwritten, config files are path-tokenized, and a file this
646/// machine previously synced and has since deleted is removed from the repo.
647pub fn push_artifacts(
648    claude_dir: &Path,
649    repo_root: &Path,
650    filter: &FilterConfig,
651    skip_artifact_paths: &HashSet<(CategoryId, PathBuf)>,
652) -> Result<ArtifactReport> {
653    let mut report = ArtifactReport::default();
654    let tokens = PathTokens::for_claude_dir(claude_dir);
655    let tracked_before = tracked::load(claude_dir, repo_root);
656    // The deltas this push owns. Both records are saved as deltas merged
657    // into a fresh load, never as a start-of-push map: a same-repo pull
658    // can record entries while this push runs, and a whole-entry save
659    // would erase them (the push-side twin of the apply's delta merge).
660    let mut deltas = RecordDeltas::default();
661    let mut tracked_now = TrackedPaths::new();
662
663    for desc in active_categories(filter) {
664        let mut counts = CategoryCounts::new(desc.id);
665
666        let mut held_back = Vec::new();
667        let files = collect(
668            desc,
669            claude_dir,
670            filter,
671            &mut counts.skipped,
672            &mut held_back,
673        )?;
674        let category_root = category_repo_root(desc, repo_root, filter);
675        // The tracked record is the durable "this machine has seen this
676        // file" set. The push consults it to refuse resurrecting a file
677        // the remote lost between syncs (adversarial review 2026-10-09,
678        // 5ff1d62 hole): a fresh-looking "Added" on the push report
679        // would otherwise silently rewrite the remote's deletion. The
680        // local copy is preserved on disk. Re-introducing the file
681        // requires a `pull` after the local `rm` — the pull sees local
682        // and repo both gone and clears the tracked entry, and a
683        // subsequent push on a freshly created local file then lands
684        // as a fresh `Added`. The push itself does not clear tracked
685        // entries; the only way out is the pull. Read once, at the top of
686        // the push (`tracked_before`): the push only saves its record at
687        // the end, so a per-category re-read would see the same set.
688        let mut pushed: TrackedPaths = TrackedPaths::new();
689        // Files kept back for their size are still here: count them as
690        // present so the repo copy (if any) is neither deleted nor forgotten.
691        for rel in held_back {
692            if let Some(rel) = repo_relative(repo_root, &category_root.join(rel)) {
693                pushed.insert(rel);
694            }
695        }
696
697        for file in files {
698            // Per-file skip: a `kept_local` (or `kept_local_delete`)
699            // decision made by the pull must not be reversed by the
700            // next push. Without this, a clean-keep file (local
701            // matches the base) would still differ from the repo
702            // (the repo has the newer version) and a RawOverwrite
703            // write would overwrite the repo's newer bytes with the
704            // local base bytes — the exact data loss the protection
705            // exists to prevent. A Union* file is also skipped, but
706            // the union merge is grow-only and the local lines
707            // survive even if the merge runs.
708            // The is_empty fast path matters: a bare `push` passes an
709            // empty set, and cloning every file's rel just to probe an
710            // empty HashSet is pure waste on large trees.
711            if !skip_artifact_paths.is_empty()
712                && skip_artifact_paths.contains(&(desc.id, file.rel.clone()))
713            {
714                counts.unchanged += 1;
715                continue;
716            }
717            let dest = category_root.join(&file.rel);
718            // The LOCAL bytes of a raw-overwrite file, hashed while they are
719            // still at hand; consumed after the match.
720            let mut base_hash: Option<String> = None;
721
722            match desc.merge {
723                MergeStrategy::UnionJsonl => {
724                    // Never fatal, like the other arms: an unreadable file
725                    // is held back and warned, not read as empty — reading
726                    // a broken local file as empty would promise a merge
727                    // that silently forgets its lines.
728                    let local_text = match fs::read_to_string(&file.abs) {
729                        Ok(text) => text,
730                        Err(e) => {
731                            hold_back(&mut counts, &file.abs, "unreadable", &e);
732                            continue;
733                        }
734                    };
735                    let existed = dest.is_file();
736                    let repo_text = if existed {
737                        match fs::read_to_string(&dest) {
738                            Ok(text) => text,
739                            Err(e) => {
740                                hold_back(&mut counts, &file.abs, "repo copy unreadable", &e);
741                                continue;
742                            }
743                        }
744                    } else {
745                        String::new()
746                    };
747                    let (merged, new_lines) = merge_history_lines(&repo_text, &local_text);
748                    if !existed {
749                        if let Err(e) = write_atomic(&dest, merged.as_bytes(), &file.abs) {
750                            hold_back(&mut counts, &file.abs, "repo write failed", &e);
751                            continue;
752                        }
753                        // Counted only once the write landed: a held-back
754                        // write must not report its lines as merged.
755                        counts.merged_entries += new_lines;
756                        counts.added += 1;
757                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
758                    } else if merged != repo_text {
759                        if let Err(e) = write_atomic(&dest, merged.as_bytes(), &file.abs) {
760                            hold_back(&mut counts, &file.abs, "repo write failed", &e);
761                            continue;
762                        }
763                        // Same landed-write rule as the creation branch
764                        // (and the memory-index twin).
765                        counts.merged_entries += new_lines;
766                        counts.modified += 1;
767                        report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
768                    } else {
769                        counts.unchanged += 1;
770                    }
771                }
772                MergeStrategy::UnionMemoryIndex | MergeStrategy::RawOverwrite => {
773                    // Never fatal, matching the pull side: an unreadable
774                    // local file is held back and warned, so a bad file
775                    // cannot wedge every later sync behind one error.
776                    let read_bytes = match fs::read(&file.abs) {
777                        Ok(bytes) => bytes,
778                        Err(e) => {
779                            hold_back(&mut counts, &file.abs, "unreadable", &e);
780                            continue;
781                        }
782                    };
783                    let existed = dest.is_file();
784                    let unions_index =
785                        desc.merge == MergeStrategy::UnionMemoryIndex && is_memory_index(&file.rel);
786                    // Hashed before `read_bytes` moves into `src_bytes`: the
787                    // base record stores the LOCAL bytes as pushed, per the
788                    // one shared rule (see `records_base`).
789                    base_hash =
790                        records_base(desc.merge, &file.rel).then(|| bases::hash_bytes(&read_bytes));
791                    let mut src_bytes = if desc.tokenize_paths {
792                        tokens.to_repo(&read_bytes)
793                    } else {
794                        read_bytes
795                    };
796                    // Repo-side IO is never fatal either: one bad repo
797                    // file must not wedge every later push.
798                    // Merged lines count only once a write lands (see the
799                    // jsonl arm): `pending_entries` rides to the write.
800                    let mut pending_entries = 0usize;
801                    if existed && unions_index {
802                        match fs::read(&dest) {
803                            Ok(dest_bytes) => {
804                                let (merged, new_entries) =
805                                    merge_memory_index(&dest_bytes, &src_bytes);
806                                pending_entries = new_entries;
807                                src_bytes = merged;
808                            }
809                            Err(e) => {
810                                hold_back(&mut counts, &file.abs, "repo copy unreadable", &e);
811                                continue;
812                            }
813                        }
814                    }
815
816                    if !existed {
817                        // If this machine has a `tracked` entry for the
818                        // file, the remote lost it since the last sync
819                        // (or since this machine last fetched it). The
820                        // local copy is preserved on disk; refusing to
821                        // re-publish it is the safe default — the
822                        // user's intent on the other side was a real
823                        // deletion, and resurrecting it silently
824                        // rewrites the remote history. The
825                        // higher-level `sync` command exposes an
826                        // explicit way to override; here we just
827                        // hold back, count, and warn.
828                        let repo_rel = repo_relative(repo_root, &category_root.join(&file.rel));
829                        if repo_rel
830                            .as_ref()
831                            .map(|r| tracked_before.contains(r))
832                            .unwrap_or(false)
833                        {
834                            log::warn!(
835                                "Not re-publishing {}: the remote lost this file since the last sync; the local copy is preserved",
836                                file.rel.display()
837                            );
838                            counts.held_back_remote_lost += 1;
839                            // Record the file identity (repo-relative) so the
840                            // push summary can list which files were held back
841                            // and `push --resurrect <path>` can target them.
842                            // The escape hatch lives in
843                            // `engine::prepare_resurrection`.
844                            report.record(
845                                desc.id,
846                                ArtifactChangeKind::HeldBackRemoteLost,
847                                &file.rel,
848                            );
849                            continue;
850                        }
851                        if let Err(e) = write_atomic(&dest, &src_bytes, &file.abs) {
852                            hold_back(&mut counts, &file.abs, "repo write failed", &e);
853                            continue;
854                        }
855                        counts.merged_entries += pending_entries;
856                        counts.added += 1;
857                        report.record(desc.id, ArtifactChangeKind::Added, &file.rel);
858                    } else {
859                        let dest_bytes = match fs::read(&dest) {
860                            Ok(bytes) => bytes,
861                            Err(e) => {
862                                hold_back(&mut counts, &file.abs, "repo copy unreadable", &e);
863                                continue;
864                            }
865                        };
866                        if dest_bytes != src_bytes {
867                            if let Err(e) = write_atomic(&dest, &src_bytes, &file.abs) {
868                                hold_back(&mut counts, &file.abs, "repo write failed", &e);
869                                continue;
870                            }
871                            counts.merged_entries += pending_entries;
872                            counts.modified += 1;
873                            report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
874                        } else {
875                            let realigned = align_executable_bit(&dest, &file.abs);
876                            if realigned {
877                                counts.modified += 1;
878                                report.record(desc.id, ArtifactChangeKind::Modified, &file.rel);
879                            } else {
880                                counts.unchanged += 1;
881                            }
882                        }
883                    }
884                }
885            }
886
887            if let Some(rel) = repo_relative(repo_root, &dest) {
888                if let Some(hash) = &base_hash {
889                    deltas.bases_inserts.insert(rel.clone(), hash.clone());
890                }
891                pushed.insert(rel);
892            }
893        }
894
895        if desc.mirror_deletes {
896            if source_is_present(desc, claude_dir) {
897                // The marker only keeps an emptied directory alive; failing
898                // to write it must not abort a push that already wrote
899                // files — warn and let the removals run.
900                if let Err(e) = mark_category_synced(&category_root) {
901                    log::warn!("Failed to mark {} synced: {e}", category_root.display());
902                }
903                let (removed, retained) = remove_from_repo(
904                    desc,
905                    claude_dir,
906                    filter,
907                    &tracked_before,
908                    &pushed,
909                    &category_root,
910                    repo_root,
911                    &mut deltas,
912                    skip_artifact_paths,
913                );
914                counts.deleted += removed.len();
915                for path in &removed {
916                    report.record(desc.id, ArtifactChangeKind::Deleted, path);
917                }
918                tracked_now.extend(pushed);
919                tracked_now.extend(retained);
920            } else {
921                // A category this machine does not have says nothing about
922                // what the others hold: leave the repo copy and the record.
923                log::info!(
924                    "Category {} is not present under {}; its repo copy is left untouched",
925                    desc.name,
926                    claude_dir.display()
927                );
928                tracked_now.extend(tracked_under(&tracked_before, &category_root, repo_root));
929            }
930        } else {
931            // mirror_deletes:false categories still need the pushed paths
932            // recorded in the `tracked` set — the push consults it to
933            // refuse resurrecting a file the remote lost since this
934            // machine last fetched it (adversarial review 2026-10-09,
935            // 5ff1d62 hole). The gone-scan skips these categories
936            // anyway, so recording here cannot trigger a future delete.
937            tracked_now.extend(pushed);
938        }
939
940        report.counts.push(counts);
941    }
942
943    // ONE locked delta merge (see `tracked::save_delta`), unconditional:
944    // a same-repo pull that rewrites the record while this push runs
945    // keeps its entries, and so do the entries of categories this push
946    // did not scan (disabled ones) — their files still exist, and
947    // forgetting them would silently stop their deletion mirroring.
948    // Unconditional also covers mirror_deletes:false categories, whose
949    // pushed paths need to land in the tracked set so the 5ff1d62
950    // guard (refuse to resurrect a remote-lost file) is effective
951    // even when the filter enables only those categories. A
952    // record-save failure must not discard the report and strand a
953    // half-applied push.
954    if let Err(e) = tracked::save_delta(claude_dir, repo_root, tracked_now, deltas.tracked_removals)
955    {
956        log::warn!("Tracked record NOT saved (files are already pushed): {e}");
957    }
958    // Gated on what this push OWNS (its computed inserts and removals),
959    // not on `bases_changed`'s start-of-push movement comparison: a
960    // concurrent writer that pruned an entry between the lock-free load
961    // and this save would make the movement test read false and skip the
962    // save entirely — the pushed file silently losing its base entry
963    // (its next local edit would read as never-synced). The save itself
964    // is a locked delta merge on a FRESH load, and `RepoRecord::write`
965    // skips identical content, so an ownership-gated no-op save is
966    // churn-free.
967    if !deltas.bases_inserts.is_empty() || !deltas.bases_removals.is_empty() {
968        if let Err(e) = bases::save_delta(
969            claude_dir,
970            repo_root,
971            deltas.bases_inserts,
972            deltas.bases_removals,
973        ) {
974            log::warn!("Base record NOT saved (files are already pushed): {e}");
975        }
976    }
977    Ok(report)
978}

997/// Marker file that keeps a category's repo directory present once its last
998/// real file is deleted.
999///
1000/// Git does not track directories, so an emptied category would disappear and
1001/// be read as "this repo has no such category", which pull must not delete
1002/// for. The marker distinguishes an empty category from an absent one.
1003pub const CATEGORY_MARKER: &str = ".synced";

1005/// Keep the category's repo directory alive across an emptying push.
1006fn mark_category_synced(category_root: &Path) -> Result<()> {
1007    let marker = category_root.join(CATEGORY_MARKER);
1008    if marker.is_file() {
1009        return Ok(());
1010    }
1011    fs::create_dir_all(category_root)?;
1012    fs::write(&marker, b"").with_context(|| format!("Failed to write {}", marker.display()))?;
1013    Ok(())
1014}

1031/// The tracked paths that belong to one category's repo directory.
1032fn tracked_under(tracked: &TrackedPaths, category_root: &Path, repo_root: &Path) -> Vec<String> {

1044/// Delete the repo copies of files this machine synced before and no longer
1045/// has, pruning their base entries so the record stops describing files that
1046/// no longer exist on either side.
1047///
1048/// "No longer has" is verified against the LOCAL side, not inferred from the
1049/// push: a file the push held back (unreadable, over the size limit, failed
1050/// write) never entered `pushed`, and treating it as a deletion would delete
1051/// the repo copy — and every other machine's file with it. Such files are
1052/// returned as `retained` so the caller keeps them tracked and the next push
1053/// retries them. Nothing here is fatal: a removal that cannot happen is left
1054/// for the next push, like a held-back file.
1055#[allow(clippy::too_many_arguments)]
1056fn remove_from_repo(
1057    desc: &CategoryDescriptor,
1058    claude_dir: &Path,
1059    filter: &FilterConfig,
1060    tracked_before: &TrackedPaths,
1061    pushed: &TrackedPaths,
1062    category_root: &Path,
1063    repo_root: &Path,
1064    deltas: &mut RecordDeltas,
1065    skip_artifact_paths: &HashSet<(CategoryId, PathBuf)>,
1066) -> (Vec<PathBuf>, TrackedPaths) {
1067    let mut removed = Vec::new();
1068    let mut retained = TrackedPaths::new();
1069    let category_prefix = repo_relative(repo_root, category_root).unwrap_or_default();
1070    for gone in tracked_under(tracked_before, category_root, repo_root) {
1071        if pushed.contains(&gone) {
1072            continue;
1073        }
1074        let category_relative = Path::new(&gone)
1075            .strip_prefix(&category_prefix)
1076            .map(Path::to_path_buf)
1077            .unwrap_or_default();
1078        // The same refusal the pull gone-scan applies: a hand-edited or
1079        // corrupted record entry must never point a repo deletion outside
1080        // its category (".." components resolve straight out of the
1081        // repository otherwise).
1082        if is_unsafe_rel_path(&category_relative) || is_denied(&category_relative) {
1083            // The entry is FORGOTTEN, not retained: every later push would
1084            // refuse it again, and no command could ever clear the loop
1085            // short of hand-editing the tracked record. The repo copy
1086            // stays (the refusal stands), the file simply leaves sync's
1087            // books — loudly, once.
1088            log::warn!(
1089                "Refusing denied/unsafe tracked path {gone} — forgetting the tracked \
1090                 entry (the repo copy is kept; the file is no longer synced)"
1091            );
1092            // The bases entry must go with it: the pull-side gone-scan
1093            // (the only other pruner) iterates tracked entries, so a
1094            // bases key left behind here could never be pruned by any
1095            // command — and a stale hash for a path this tool refuses
1096            // to touch describes nothing.
1097            deltas.bases_removals.push(gone.clone());
1098            deltas.tracked_removals.push(gone);
1099            continue;
1100        }
1101        // The local copy still existing means this is NOT a deletion: the
1102        // push held the file back. Keep the repo copy, the base entry, and
1103        // the tracked entry. A destination that cannot be resolved is
1104        // unverifiable, and an unverifiable file is never delete material.
1105        let local_deleted = match local_destination(desc, claude_dir, &category_relative, filter) {
1106            Ok(local) => !local.is_file(),
1107            Err(_) => {
1108                log::warn!("Keeping repo copy of {gone} (no local destination)");
1109                false
1110            }
1111        };
1112        // A path the pull just held or left unfinished is not this
1113        // machine's deletion to publish: the absence may be a deletion
1114        // made while the pull waited, set against a change the other
1115        // machine made.
1116        if !local_deleted || skip_artifact_paths.contains(&(desc.id, category_relative.clone())) {
1117            retained.insert(gone);
1118            continue;
1119        }
1120        let path = repo_root.join(&gone);
1121        if path.is_file() {
1122            if let Err(e) = fs::remove_file(&path) {
1123                log::warn!("Failed to remove {}: {e}", path.display());
1124                retained.insert(gone);
1125                continue;
1126            }
1127            removed.push(category_relative);
1128        }
1129        // Neither side has this file anymore — the entries must go whether
1130        // or not the repo copy was still there to remove (it may already
1131        // have been deleted outside this tool).
1132        deltas.bases_removals.push(gone.clone());
1133        deltas.tracked_removals.push(gone);
1134    }
1135    (removed, retained)
1136}

1150/// Read-only classification of an artifact pull, computed BEFORE any write so
1151/// the caller can snapshot the exact set of files that will change.
1152#[derive(Debug, Default)]
1153pub struct PullPlan {
1154    /// Local file exists and repo bytes differ: remote wins after snapshot.
1155    pub overwrites: Vec<PlannedWrite>,
1156    /// Local files edited since the last sync: kept as they are, because
1157    /// overwriting them would destroy edits only this machine holds. A
1158    /// `push` is what publishes them.
1159    pub kept_local: Vec<PlannedWrite>,
1160    /// Local files edited here since the last sync while the repository still
1161    /// holds that sync's version: nothing to take. Left as they are and, unlike
1162    /// `kept_local`, not held back — the push that follows publishes them.
1163    pub local_only: Vec<PlannedWrite>,
1164    /// Files this machine synced and has since deleted while the repository
1165    /// still holds that sync's version: left deleted, and the push removes the
1166    /// repository copy.
1167    pub deleted_here: Vec<PlannedWrite>,
1168    /// Files both sides changed since the last sync whose two versions differ
1169    /// only in date-times (Claude Code rewrites timestamps in its plugin and
1170    /// skill manifests on every machine): settled on apply by keeping the later
1171    /// dates, which the push then publishes.
1172    pub date_settles: Vec<PlannedWrite>,
1173    /// No local file yet: created, and recorded for deletion on undo.

1282impl PullPlan {
1283    /// True when applying the plan would write no artifact file by
1284    /// default. Kept-local entries count despite writing nothing
1285    /// non-interactively, because an interactive apply may take the
1286    /// repository copy over them. Records are excluded: recording a base
1287    /// is machine-state change, not an artifact write — see
1288    /// [`Self::changes_machine_state`].
1289    pub fn is_empty(&self) -> bool {
1290        self.overwrites.is_empty()
1291            && self.date_settles.is_empty()
1292            && self.creates.is_empty()
1293            && self.unions.is_empty()
1294            && self.mode_fixes.is_empty()
1295            && self.deletes.is_empty()
1296            && self.kept_local.is_empty()
1297            && self.kept_local_deletes.is_empty()
1298    }
1299
1300    /// True when applying the plan writes anything on this machine:
1301    /// artifact files, or the shared records (which `rewrites_bases` /
1302    /// `rewrites_tracked`, computed at plan time against what is on disk,
1303    /// predict). This is what the snapshot gate consults — a pull whose
1304    /// only effect is creating a record still needs a snapshot for its
1305    /// undo, while a fully no-op pull must not churn one.
1306    ///
1307    /// Kept-local lists count only under `interactive`: a non-interactive
1308    /// apply provably never writes them, and an unresolved edit must not
1309    /// mint a snapshot on every daily sync.
1310    pub fn changes_machine_state(&self, interactive: bool) -> bool {
1311        // Stated directly, not as a nested double negation: some arm writes
1312        // a file, or an interactive apply can (the kept-local lists).
1313        // mode_fixes are deliberately absent: a chmod cannot be restored
1314        // from a bytes snapshot, so a mode-fix-only pull must not mint an
1315        // empty one — the base entry the fix records still gates the
1316        // snapshot through `rewrites_bases`.
1317        let file_writes = !(self.overwrites.is_empty()
1318            && self.date_settles.is_empty()
1319            && self.creates.is_empty()
1320            && self.unions.is_empty()
1321            && self.deletes.is_empty())
1322            || (interactive
1323                && (!self.kept_local.is_empty() || !self.kept_local_deletes.is_empty()));
1324        file_writes || self.rewrites_bases || self.rewrites_tracked
1325    }
1326
1327    /// The repo-relative keys this pull's apply writes or prunes in the
1328    /// BASES record — every arm that records, forgets, or durably declines.
1329    /// `kept_local` included: an interactive take-remote records the repo
1330    /// bytes as the new base, and undo must be able to restore that key
1331    /// per-key instead of falling back to a whole-entry rewind. Undo
1332    /// restores the PRE-PULL value of exactly these keys; entries a push
1333    /// recorded after the pull keep their owner. A superset is safe:
1334    /// restoring a key the apply never moved is a no-op.
1335    pub fn touched_base_keys(&self) -> Vec<String> {
1336        let mut keys: Vec<String> = self
1337            .creates
1338            .iter()
1339            .chain(self.overwrites.iter())
1340            .chain(self.mode_fixes.iter())
1341            .chain(self.kept_local.iter())
1342            // A union-merged memory index records the merged bytes.
1343            .chain(self.unions.iter())
1344            .filter_map(|w| repo_relative(&self.repo_root, &w.repo_path))
1345            .collect();
1346        keys.extend(self.base_hashes.iter().map(|(rel, _)| rel.clone()));
1347        keys.extend(self.deletes.iter().map(|d| d.repo_rel.clone()));
1348        keys.extend(self.kept_local_deletes.iter().map(|d| d.repo_rel.clone()));
1349        keys.extend(self.base_prunes.iter().cloned());
1350        keys
1351    }
1352
1353    /// Whether the snapshot must carry the bases record: it exists on disk
1354    /// and the apply can write it — the non-interactive writes
1355    /// `rewrites_bases` predicts, plus the interactive kept-local
1356    /// resolutions (a take-remote records the repo bytes as the new
1357    /// base) that `rewrites_bases` deliberately excludes; the snapshot
1358    /// gate's own `interactive` term covers their file-write side. The
1359    /// ONE rule — `paths_to_snapshot` and
1360    /// `Snapshot::attach_record_bookkeeping` both call these, so the
1361    /// carrier and the declaration can never drift apart.
1362    pub fn carries_bases_record(&self, interactive: bool) -> bool {
1363        self.bases_record_exists && self.can_write_bases(interactive)
1364    }
1365
1366    /// Whether the snapshot must carry the tracked record: it exists and
1367    /// the apply will actually rewrite it (`rewrites_tracked` — tracked
1368    /// writes are fully predictable without the interactive term, unlike
1369    /// bases: no apply arm touches the tracked record only because a
1370    /// prompt was answered). Carrying it otherwise would declare record
1371    /// surgery — and snapshot the shared file — for a record the apply
1372    /// provably never writes.
1373    pub fn carries_tracked_record(&self) -> bool {
1374        self.tracked_record_exists && self.rewrites_tracked
1375    }
1376
1377    /// Whether the apply may create the bases record (it does not exist
1378    /// yet and this plan can write it — same rule as
1379    /// [`Self::carries_bases_record`]).
1380    pub fn may_create_bases_record(&self, interactive: bool) -> bool {
1381        !self.bases_record_exists && self.can_write_bases(interactive)
1382    }
1383
1384    /// Whether the apply can write the bases record under the given
1385    /// interactivity — see [`Self::carries_bases_record`]. Public for the
1386    /// pull's concurrent-writer warning, which must mirror the snapshot
1387    /// gate exactly instead of recomposing the existence split.
1388    pub fn can_write_bases(&self, interactive: bool) -> bool {
1389        self.rewrites_bases
1390            || (interactive && (!self.kept_local.is_empty() || !self.kept_local_deletes.is_empty()))
1391    }
1392
1393    /// Whether the apply may create the tracked record: it does not exist
1394    /// and there is something to record — the same rule as
1395    /// [`Self::rewrites_tracked`], whose absent-file branch this is.
1396    pub fn may_create_tracked_record(&self) -> bool {
1397        !self.tracked_record_exists && self.rewrites_tracked
1398    }
1399
1400    /// Existing local files the caller must snapshot before applying
1401    /// (overwritten raw files, union-merged files, deletions, and — only
1402    /// under `interactive` — kept-local files: an interactive apply may
1403    /// take the repository copy over the local edit, and a snapshot is
1404    /// what makes `undo pull` able to bring the edit back; the
1405    /// non-interactive apply provably never writes them).
1406    ///
1407    /// The shared records are included as carriers only, per the
1408    /// `carries_*_record` rules above: `undo pull` reads this
1409    /// repository's pre-pull entry from the snapshot bytes and re-applies
1410    /// it surgically — the files hold every sync repository's state and
1411    /// must not be restored wholesale.
1412    pub fn paths_to_snapshot(&self, interactive: bool) -> Vec<PathBuf> {
1413        let mut paths: Vec<PathBuf> = self
1414            .overwrites
1415            .iter()
1416            .chain(self.date_settles.iter())
1417            .chain(self.unions.iter())
1418            .map(|w| w.local_path.clone())
1419            .chain(self.deletes.iter().map(|d| d.local_path.clone()))
1420            .collect();
1421        if interactive {
1422            paths.extend(self.kept_local.iter().map(|w| w.local_path.clone()));
1423            // An interactive confirm can also DELETE a kept-local file —
1424            // same undo requirement as a take-remote.
1425            paths.extend(self.kept_local_deletes.iter().map(|d| d.local_path.clone()));
1426        }
1427        if self.carries_tracked_record() {
1428            paths.push(tracked::record_path(&self.claude_dir));
1429        }
1430        if self.carries_bases_record(interactive) {
1431            paths.push(bases::record_path(&self.claude_dir));
1432        }
1433        paths
1434    }
1435
1436    /// Local paths this pull will create; recording them as a snapshot's
1437    /// `deleted_files` makes undo remove them again.
1438    ///
1439    /// The base record is deliberately absent: it is one file shared by
1440    /// every sync repository, and `undo pull` restores this repository's
1441    /// entry from the snapshot instead of deleting the file.
1442    pub fn created_paths(&self) -> Vec<String> {
1443        self.creates
1444            .iter()
1445            .map(|w| w.local_path.to_string_lossy().to_string())
1446            .collect()
1447    }
1448}

1450/// Enumerate one category's files as stored in the sync repository, returning
1451/// (absolute repo path, path relative to the category subdir). Denied and
1452/// unsafe paths are refused here, so nothing below ever sees them.
1453fn collect_repo_files(
1454    desc: &CategoryDescriptor,
1455    repo_root: &Path,
1456    filter: &FilterConfig,
1457    skipped: &mut usize,
1458    held_back: &mut Vec<PathBuf>,
1459) -> Vec<(PathBuf, PathBuf)> {
1460    let category_root = category_repo_root(desc, repo_root, filter);
1461    if !category_root.is_dir() {
1462        return Vec::new();
1463    }
1464
1465    let mut files = Vec::new();
1466    for entry in walkdir::WalkDir::new(&category_root)
1467        .follow_links(false)
1468        .into_iter()
1469        .filter_map(|e| e.ok())
1470    {
1471        if !entry.file_type().is_file() {
1472            continue;
1473        }
1474        let abs = entry.path();
1475        let rel = abs
1476            .strip_prefix(&category_root)
1477            .unwrap_or(abs)
1478            .to_path_buf();
1479        if extension_excluded(desc, &rel) {
1480            continue;
1481        }
1482        // The marker exists to keep the directory, and belongs to no machine.
1483        if rel == Path::new(CATEGORY_MARKER) {
1484            continue;
1485        }
1486        if is_unsafe_rel_path(&rel) || is_denied(&rel) {
1487            log::warn!(
1488                "Refusing denied/unsafe artifact from sync repo: {}",
1489                abs.display()
1490            );
1491            *skipped += 1;
1492            continue;
1493        }
1494        // The size limit applies to the repo side too — the mirror of the
1495        // push-side guard: another machine with a higher limit may have
1496        // pushed a file this one must not write. Held back, NOT gone:
1497        // deletion mirroring must never read a not-pulled file as
1498        // deleted-here (the pull-side twin of push's held_back).
1499        // The one shared fail-closed triage (metadata form — the entry
1500        // already holds it).
1501        if metadata_outside_limit(entry.metadata(), filter.max_file_size_bytes) {
1502            log::warn!("Skipping {} (exceeds max_file_size_bytes)", abs.display());
1503            *skipped += 1;
1504            held_back.push(abs.to_path_buf());
1505            continue;
1506        }
1507        files.push((abs.to_path_buf(), rel));
1508    }
1509    files
1510}

1512/// Why a repo file has no local destination. The caller reports the two cases
1513/// differently: an unmapped project is one warning for all of its files, an
1514/// unrestorable path is a refusal worth its own line.
1515enum SkipReason {
1516    /// This machine has no directory for the named sync-repo project.
1517    UnmappedProject(String),
1518    /// Nothing in the category can receive this path.
1519    NotRestorable,
1520}

1522/// Map a category-relative repo path back to its absolute local destination
1523/// under `~/.claude`, or the reason it has none.
1524fn local_destination(
1525    desc: &CategoryDescriptor,
1526    claude_dir: &Path,
1527    rel: &Path,
1528    filter: &FilterConfig,
1529) -> Result<PathBuf, SkipReason> {
1530    match desc.source {
1531        // File lists are stored flat in the repo; restore to the listed
1532        // location whose file name matches. An unlisted name has NO valid
1533        // destination — the allowlist must hold on pull as well as push, or a
1534        // poisoned repo could plant arbitrary files at the top of ~/.claude.
1535        SourceSpec::Files(list) => list
1536            .iter()
1537            .find(|entry| Path::new(entry).file_name() == rel.file_name())
1538            .map(|entry| claude_dir.join(entry))
1539            .ok_or(SkipReason::NotRestorable),
1540        SourceSpec::Dir(dir) => {
1541            if desc.dest == DestRoot::SessionTree {
1542                // Repo path is <project-dir>/<rest>; resolve the leading
1543                // component the way session pull does.
1544                let split = crate::project_map::split_project_path(rel);
1545                let (project, inside_project) = split.ok_or(SkipReason::NotRestorable)?;
1546                let projects_dir = claude_dir.join(dir);
1547                let local_project =
1548                    crate::project_map::local_project_dir(filter, &projects_dir, project)
1549                        .ok_or_else(|| SkipReason::UnmappedProject(project.to_string()))?;
1550                return Ok(local_project.join(inside_project));
1551            }
1552            Ok(claude_dir.join(dir).join(rel))
1553        }
1554    }
1555}

1580/// Classify what a pull would write, without writing. Remote (repo) bytes win
1581/// for raw categories; union targets are compared against local ∪ remote; a
1582/// file this machine synced before and the repo no longer has is a deletion.
1583pub fn plan_pull(claude_dir: &Path, repo_root: &Path, filter: &FilterConfig) -> Result<PullPlan> {
1584    let mut plan = PullPlan {
1585        tokens: PathTokens::for_claude_dir(claude_dir),
1586        claude_dir: claude_dir.to_path_buf(),
1587        repo_root: repo_root.to_path_buf(),
1588        merge_tool: filter.merge_tool.clone(),
1589        prefer_merge_tool: filter.prefer_merge_tool,
1590        tracks_deletions: active_categories(filter).any(|desc| desc.mirror_deletes),
1591        ..Default::default()
1592    };
1593    // A PRESENT but broken record is not the fresh-machine state: say
1594    // so, loudly — a corrupt bases record silently turns the protection
1595    // off (every differing file reads as a clean overwrite), a corrupt
1596    // tracked record silently stops deletion mirroring.
1597    if !super::repo_record::intact::<BaseHashes>(&bases::record_path(claude_dir)) {
1598        log::warn!(
1599            "the base record is unreadable or corrupt — local-edit protection is OFF this round; \
1600             fix or remove {}",
1601            bases::record_path(claude_dir).display()
1602        );
1603    }
1604    if !super::repo_record::intact::<TrackedPaths>(&tracked::record_path(claude_dir)) {
1605        log::warn!(
1606            "the tracked record is unreadable or corrupt — deletion mirroring is OFF this round; \
1607             fix or remove {}",
1608            tracked::record_path(claude_dir).display()
1609        );
1610    }
1611    // The records save under an OS file lock; on a filesystem that does
1612    // not support it (old NFS, some SMB/fuse mounts) every save fails and
1613    // both protections are permanently off — say so at plan time, where
1614    // the corrupt-record warnings live, instead of leaving it to per-save
1615    // warns scattered through the apply. An ABSENT Claude directory is
1616    // the fresh-machine state (nothing to protect yet; the saves create
1617    // everything on first write) — probing it would make a read-only
1618    // `status` create directories, so the probe runs only when the
1619    // directory exists.
1620    if claude_dir.is_dir() {
1621        if let Err(e) = super::repo_record::record_lock_probe(claude_dir) {
1622            log::warn!(
1623                "record locking is unavailable — local-edit protection and deletion mirroring \
1624                 cannot be recorded this round ({e})"
1625            );
1626        }
1627    }
1628    let tracked_before = tracked::load(claude_dir, repo_root);
1629    plan.tracked_before = tracked_before.clone();
1630    let base_hashes: BaseHashes = bases::load(claude_dir, repo_root);
1631    // A machine that has synced this repository before but holds no base
1632    // for a differing file is upgrading from a version that kept no bases
1633    // (or lost its record): the difference may be an edit made here, and
1634    // with no base nothing tells it apart from a stale copy. Hold it rather
1635    // than remote-win over it. Only a machine new to this repository —
1636    // nothing tracked, nothing recorded — takes the repository's version.
1637    let synced_this_repo_before = !tracked_before.is_empty() || !base_hashes.is_empty();
1638
1639    for desc in active_categories(filter) {
1640        let category_root = category_repo_root(desc, repo_root, filter);
1641        let mut present: TrackedPaths = TrackedPaths::new();
1642
1643        let mut repo_held_back = Vec::new();
1644        let collected = collect_repo_files(
1645            desc,
1646            repo_root,
1647            filter,
1648            &mut plan.skipped,
1649            &mut repo_held_back,
1650        );
1651        // Size-held-back repo files still exist in the repository: they
1652        // count as present so the deletion-mirror arm never reads them
1653        // as gone and deletes a clean local copy. But they are NOT
1654        // tracked: this machine never received the file, and a tracked
1655        // path with no local copy is exactly what the next push's
1656        // remove_from_repo reads as a local deletion — deleting the repo
1657        // copy for every machine.
1658        let mut repo_held_back_rels: TrackedPaths = TrackedPaths::new();
1659        for abs in &repo_held_back {
1660            if let Some(rel) = repo_relative(repo_root, abs) {
1661                repo_held_back_rels.insert(rel.clone());
1662                present.insert(rel);
1663            }
1664        }
1665        for (repo_path, rel) in collected {
1666            let local_path = match local_destination(desc, claude_dir, &rel, filter) {
1667                Ok(local_path) => local_path,
1668                Err(SkipReason::UnmappedProject(project)) => {
1669                    plan.skipped += 1;
1670                    plan.unmapped_projects
1671                        .entry(project)
1672                        .or_default()
1673                        .push(repo_path);
1674                    continue;
1675                }
1676                Err(SkipReason::NotRestorable) => {
1677                    plan.skipped += 1;
1678                    log::warn!(
1679                        "Skipping {} (not a file the {} category restores)",
1680                        repo_path.display(),
1681                        desc.name
1682                    );
1683                    continue;
1684                }
1685            };
1686            if desc.mirror_deletes {
1687                if let Some(tracked_path) = repo_relative(repo_root, &repo_path) {

1764                    // entry" (memory_index.rs): a user-curated bullet
1765                    // at a target key the incoming (repo) also has is
1766                    // treated as a match and overwritten. That is
1767                    // silent clobber for issue #103. The same F3 + base

1769                    // have no recorded base but the file is in

1771                    // match the local bytes, the user edited this
1772                    // since the last sync and the next pull must not
1773                    // union-merge over it.

1857                            Err(e) => {

1868                            // the last sync, the repository holds nothing this

2012                        "Skipping {} (unreadable or exceeds max_file_size_bytes)",

2059                        // kept_local_deletes and re-arm the tracked

2080                    FreshRead::Unreadable => {

2145        .chain(&plan.unions)

2147        || plan.mode_fixes.iter().any(|w| {

2149                && repo_relative(repo_root, &w.repo_path)
2150                    .is_some_and(|rel| planned_base_hashes.get(&rel) != base_hashes.get(&rel))
2151        });

2154            || !plan.base_prunes.is_empty()
2155            || plan
2156                .deletes
2157                .iter()
2158                .any(|d| base_hashes.contains_key(&d.repo_rel))
2159            || plan
2160                .base_hashes
2161                .iter()
2162                .any(|(rel, hash)| base_hashes.get(rel) != Some(hash))
2163    } else {
2164        moves_base
2165            || plan
2166                .base_hashes
2167                .iter()
2168                .any(|(rel, hash)| base_hashes.get(rel) != Some(hash))
2169    };
2170    // The apply saves a TOUCHED-KEY delta; the prediction uses the same
2171    // semantics, not a whole-entry compare: an entry the pull's scan does
2172    // not own (a disabled category's, say) would otherwise read as a
2173    // pending rewrite forever, minting a no-op snapshot and history entry
2174    // on every pull. Alias retirement still counts — the apply's save
2175    // retires it. The key set is computed once here (see the field).
2176    let mut touched_tracked: Vec<String> = plan
2177        .creates
2178        .iter()
2179        .filter_map(|w| repo_relative(repo_root, &w.repo_path))
2180        .collect();
2181    touched_tracked.extend(plan.deletes.iter().map(|d| d.repo_rel.clone()));
2182    touched_tracked.extend(plan.kept_local_deletes.iter().map(|d| d.repo_rel.clone()));
2183    touched_tracked.extend(plan.base_prunes.iter().cloned());
2184    for key in &plan.tracked_after {
2185        if !plan.tracked_before.contains(key) {
2186            touched_tracked.push(key.clone());
2187        }
2188    }
2189    plan.touched_tracked_keys = touched_tracked;
2190    let tracked_membership_flips = plan
2191        .touched_tracked_keys
2192        .iter()
2193        .any(|key| plan.tracked_before.contains(key) != plan.tracked_after.contains(key));
2194    plan.tracked_alias_present = tracked::has_aliased_entry(claude_dir, repo_root);
2195    plan.rewrites_tracked =
2196        plan.tracks_deletions && (tracked_membership_flips || plan.tracked_alias_present);
2197
2198    Ok(plan)

2201/// Apply a pull plan: create missing files, overwrite differing ones (remote
2202/// wins), union-merge prompt history and memory indexes, and remove files the
2203/// repo no longer has. Under `interactive` in a terminal, each overwrite asks
2204/// for per-file confirmation — with the configured merge tool as an option —
2205/// and each deletion asks for confirmation; declined files count as skipped.
2206pub fn apply_pull(plan: &PullPlan, interactive: bool) -> Result<ArtifactReport> {
2207    use std::collections::HashMap;
2208
2209    let mut by_category: HashMap<CategoryId, CategoryCounts> = HashMap::new();
2210    fn counts_for(
2211        map: &mut HashMap<CategoryId, CategoryCounts>,
2212        id: CategoryId,
2213    ) -> &mut CategoryCounts {
2214        map.entry(id)
2215            .or_insert_with(move || CategoryCounts::new(id))
2216    }
2217
2218    /// Count and name one kept-local file — the shared tail of every arm
2219    /// that decides to keep one. `nothing_to_publish` is decided AT the
2220    /// keep site, where the arm knows WHY it kept: a file the arm knows
2221    /// still matches its base (an overwrite or take whose write failed)
2222    /// holds the gate but has nothing a push would publish, and the
2223    /// "push to publish" hint must not be given for it; a declined
2224    /// deletion is the opposite — the file exists only here, and the
2225    /// next push republishes it.
2226    fn report_kept(
2227        map: &mut HashMap<CategoryId, CategoryCounts>,
2228        report: &mut ArtifactReport,
2229        category: CategoryId,
2230        path: &Path,
2231        nothing_to_publish: bool,
2232    ) {
2233        counts_for(map, category).kept_local += 1;
2234        if nothing_to_publish {
2235            report.kept_local_clean += 1;
2236        }
2237        report.record(category, ArtifactChangeKind::KeptLocal, path);
2238    }

3395        // window is the race); non-interactive reaching here already
3396        // read Clean at the pre-prompt match, so only the None-base
3397        // half can still fire.
3398        let post_prompt_dirty = recorded_now.is_none()
3399            || (prompt_overwrites
3400                && dirty_status(recorded_now, &delete.local_path) == FreshRead::Dirty);
3401        if post_prompt_dirty {
3402            // Edit saved during the prompt, or no recorded base: an
3403            // edited file is kept, never destroyed (the snapshot holds
3404            // only pre-pull bytes). A missing base treats the file as
3405            // potentially edited until the records can be re-classified.
3406            log::warn!(
3407                "Keeping {} (edited while the prompt was open, or its base was cleared mid-pull)",
3408                delete.local_path.display()
3409            );
3410            report_kept(
3411                &mut by_category,
3412                &mut report,
3413                delete.category,
3414                &delete.category_path,
3415                false,
3416            );
3417            // The tracked entry must survive this skip, mirroring the
3418            // pre-prompt Dirty/Unreadable branches: a transient
3419            // keep that the next round will re-classify. Without the
3420            // push, the tracked-delta loop drops the entry, and the
3421            // next pull sees the locally-edited file as never-synced
3422            // and silently overwrites it (the data-loss the original
3423            // F3-class fix was meant to prevent).
3424            deletes_to_retry.push(delete.repo_rel.clone());
3425            continue;
3426        }
3427        if delete.local_path.is_file() {
3428            if let Err(e) = fs::remove_file(&delete.local_path) {
3429                log::warn!(
3430                    "Skipping delete of {} (remove failed): {e}",
3431                    delete.local_path.display()
3432                );
3433                counts_for(&mut by_category, delete.category).skipped += 1;
3434                deletes_to_retry.push(delete.repo_rel.clone());
3435                continue;
3436            }
3437        }
3438        // The removal key counts only when an entry actually existed:
3439        // keys_written feeds the undo, and a no-op removal would let it
3440        // delete a base a LATER push legitimately recorded.
3441        if synced_bases.remove(&delete.repo_rel).is_some() {
3442            bases_changed = true;
3443            bases_removals.push(delete.repo_rel.clone());
3444        }
3445        counts_for(&mut by_category, delete.category).deleted += 1;
3446        report.record(
3447            delete.category,
3448            ArtifactChangeKind::Deleted,
3449            &delete.category_path,
3450        );
3451    }
3452
3453    // Base hashes verified at plan time: recorded as-is, without re-reading
3454    // the files — an edit made while the pull waited must NOT end up
3455    // recorded as the synced state. Only a value that actually moves
3456    // becomes an insert: an unchanged entry is a no-op the apply does not
3457    // own (and the undo must not "restore").
3458    for (rel, hash) in &plan.base_hashes {
3459        // `insert` returns the previous value in one BTreeMap descent.
3460        if synced_bases.insert(rel.clone(), hash.clone()) != Some(hash.clone()) {
3461            bases_changed = true;
3462            bases_inserts.insert(rel.clone(), hash.clone());
3463        }
3464    }
3465    // Files that exist nowhere anymore: their entries go too (no counts —
3466    // nothing visible happened to any file).
3467    for rel in &plan.base_prunes {
3468        if synced_bases.remove(rel).is_some() {
3469            bases_changed = true;
3470            bases_removals.push(rel.clone());
3471        }
3472    }
3473
3474    // Decided at apply time from the delta itself — the same touched-key
3475    // semantics the plan's `rewrites_tracked` predicts (a whole-entry
3476    // compare would see disabled categories' entries as pending
3477    // rewrites forever). The files are already written: a record-save
3478    // failure must not turn the whole pull into an Err — pull_history
3479    // would skip the operation record and strand the snapshot, making
3480    // the overwritten files unrecoverable. Degrade to a warning
3481    // instead: a stale record costs one extra kept-local round, a lost
3482    // undo costs the files.
3483    let mut removals = Vec::new();
3484    let mut inserts = TrackedPaths::new();
3485    // Hash lookups: the retry lists can grow with the pull.
3486    let deletes_to_retry: std::collections::HashSet<&String> = deletes_to_retry.iter().collect();
3487    let creates_to_retry: std::collections::HashSet<&String> = creates_to_retry.iter().collect();
3488    for key in &plan.touched_tracked_keys {
3489        if deletes_to_retry.contains(key) || creates_to_retry.contains(key) {
3490            // The deletion (or creation) did not execute: the entry must
3491            // stay exactly as it was so the decision is retried, exactly
3492            // like the plan-time unreadable path re-arms it.
3493            continue;
3494        }
3495        match (
3496            plan.tracked_before.contains(key),
3497            plan.tracked_after.contains(key),
3498        ) {
3499            (false, true) => {
3500                inserts.insert(key.clone());
3501            }
3502            (true, false) => removals.push(key.clone()),
3503            _ => {}
3504        }
3505    }
3506    let tracked_save_gate = plan.tracks_deletions
3507        && ((!inserts.is_empty() || !removals.is_empty()) || plan.tracked_alias_present);
3508    let mut tracked_keys_written: Vec<String> = Vec::new();
3509    if tracked_save_gate {
3510        // The delta this plan OWNS — exactly the keys its scan decided
3511        // about — merged into whatever is on disk NOW in ONE locked unit
3512        // (see `tracked::save_delta`): a same-repo push that records
3513        // paths while this (possibly minutes-long interactive) apply ran
3514        // keeps them, and so do the entries of categories this pull did
3515        // not scan — their files still exist, and dropping them would
3516        // silently stop their deletion mirroring. The write skips
3517        // identical content, so a delta that changes nothing does not
3518        // churn the file.
3519        tracked_keys_written.extend(inserts.iter().cloned());
3520        tracked_keys_written.extend(removals.iter().cloned());
3521        match tracked::save_delta(&plan.claude_dir, &plan.repo_root, inserts, removals) {
3522            Ok(()) => {}
3523            Err(e) => {
3524                log::warn!("Tracked record NOT saved (files are already applied): {e}");
3525                // Same contract as the bases save below: a failed save moved
3526                // nothing, and reported keys would let a later undo "restore"
3527                // entries a concurrent writer legitimately recorded.
3528                tracked_keys_written.clear();
3529            }
3530        }
3531    }
3532    let mut bases_keys_written: Vec<String> = Vec::new();
3533    if bases_changed {
3534        // One locked load-merge-write (see `bases::save_delta`): a
3535        // concurrent same-repo push's entries are preserved, and the lock
3536        // covers the READ half too. Keys reported only on a successful
3537        // save, like the tracked save above: a failed one moved nothing,
3538        // and the undo would otherwise "restore" keys a later push
3539        // legitimately recorded.
3540        bases_keys_written.extend(bases_removals.iter().cloned());
3541        bases_keys_written.extend(bases_inserts.keys().cloned());
3542        // Protection entries are NOT pull state (see `protection_keys`).
3543        // HashSet lookup is O(n+m) — Vec::contains was O(n*m).
3544        let protection_set: std::collections::HashSet<&str> =
3545            protection_keys.iter().map(String::as_str).collect();
3546        bases_keys_written.retain(|key| !protection_set.contains(key.as_str()));
3547        match bases::save_delta(
3548            &plan.claude_dir,
3549            &plan.repo_root,
3550            bases_inserts,
3551            bases_removals,
3552        ) {
3553            Ok(()) => {}
3554            Err(e) => {
3555                log::warn!("Base record NOT saved (files are already applied): {e}");
3556                bases_keys_written.clear();
3557            }
3558        }
3559    }
3560
3561    // What the caller's operation record needs to know: exactly the keys
3562    // this apply moved in each shared record (see the fields). Collected
3563    // from the deltas, not the plan — the plan's lists include no-ops a
3564    // later push must not have "restored".
3565    report.bases_keys_written = bases_keys_written;
3566    report.tracked_keys_written = tracked_keys_written;
3567
3568    report.counts = by_category.into_values().collect();
3569    report.counts.sort_by_key(|c| c.category as usize);
3570    Ok(report)
3571}

3662/// Build the full managed block for one backend.
3663fn ignore_block(backend: Backend) -> String {

3672        block.push('\n');
3673    }
3674    block.push_str(IGNORE_BLOCK_END);
3675    block.push('\n');

3679/// Write the managed never-sync ignore block into the sync repository's
3680/// ignore file for the given backend. Idempotent; preserves user content
3681/// outside the block. Returns whether the file changed.
3682pub fn ensure_ignore_files(repo_root: &Path, backend: Backend) -> Result<bool> {
3683    let file_name = match backend {
3684        Backend::Git => ".gitignore",
3685        Backend::Mercurial => ".hgignore",
3686    };
3687    let path = repo_root.join(file_name);
3688    let existing = if path.is_file() {
3689        fs::read_to_string(&path)?
3690    } else {
3691        String::new()
3692    };
3693
3694    let block = ignore_block(backend);
3695
3696    let updated = if let (Some(start), Some(end)) = (
3697        existing.find(IGNORE_BLOCK_START),
3698        existing.find(IGNORE_BLOCK_END),
3699    ) {
3700        // Replace the existing block in place.
3701        let end = end + IGNORE_BLOCK_END.len();
3702        // Include the trailing newline of the old block if present.
3703        let end = if existing[end..].starts_with('\n') {
3704            end + 1
3705        } else {
3706            end
3707        };
3708        format!("{}{}{}", &existing[..start], block, &existing[end..])
3709    } else if existing.is_empty() {
3710        block
3711    } else {
3712        let sep = if existing.ends_with('\n') {
3713            "\n"
3714        } else {
3715            "\n\n"
3716        };
3717        format!("{existing}{sep}{block}")
3718    };
3719
3720    if updated == existing {
3721        return Ok(false);
3722    }
3723    fs::write(&path, updated)?;
3724    Ok(true)
3725}

3727/// Remove a list of repo-relative paths from the per-machine tracked record.
3728///
3729/// The 5ff1d62 push guard (engine.rs:711-759) refuses to re-publish a file
3730/// the remote lost when the file is in the tracked record. The user-facing
3731/// escape hatch `claude-code-sync push --resurrect <path>` calls this helper
3732/// to drop the path from the record first; the next push then sees no entry
3733/// and publishes the file as a fresh `Added`.
3734///
3735/// Defense in depth: returns `Err` if any path is NOT in the current tracked
3736/// record. The CLI flag should have validated against the held-back set, but
3737/// if it didn't, refuse rather than silently no-op. The exception is the
3738/// initial install case (record empty): an empty tracked record is exactly
3739/// the state this helper is meant to set, so a no-op is correct there.
3740pub fn prepare_resurrection(claude_dir: &Path, repo_root: &Path, paths: &[String]) -> Result<()> {
3741    let current = tracked::load(claude_dir, repo_root);
3742    let unknown: Vec<&str> = paths
3743        .iter()
3744        .filter(|p| !current.contains(*p))
3745        .map(|p| p.as_str())
3746        .collect();
3747    if !unknown.is_empty() {
3748        anyhow::bail!(
3749            "cannot resurrect paths that are not in the tracked record: {}",
3750            unknown.join(", ")
3751        );
3752    }
3753    if paths.is_empty() {
3754        return Ok(());
3755    }
3756    // `save_delta` is a locked load-merge-write: a concurrent same-repo
3757    // pull that records entries while this runs keeps them. The empty
3758    // insert set is the no-op case for the inserts half; only the removals
3759    // half matters.
3760    let empty: TrackedPaths = TrackedPaths::new();
3761    tracked::save_delta(claude_dir, repo_root, empty, paths.to_vec())
3762}

3764/// Plan which files the 5ff1d62 push guard would refuse for a given filter,
3765/// without performing any writes. Drives `status --held-back` without
3766/// running a full push (which is the only place the guard normally fires).
3767///
3768/// Returns `(category, repo-relative-path)` pairs in push-iteration order.
3769/// The category is the one whose push would have refused the file.
3770pub fn plan_held_back_remote_lost(
3771    claude_dir: &Path,
3772    repo_root: &Path,
3773    filter: &FilterConfig,
3774) -> Result<Vec<(CategoryId, PathBuf)>> {
3775    let mut held_back: Vec<(CategoryId, PathBuf)> = Vec::new();
3776    let tracked_seen = tracked::load(claude_dir, repo_root);
3777    for desc in active_categories(filter) {
3778        let category_root = category_repo_root(desc, repo_root, filter);
3779        let mut counts = CategoryCounts::new(desc.id);
3780        let mut held_back_local: Vec<PathBuf> = Vec::new();
3781        let files = collect(
3782            desc,
3783            claude_dir,
3784            filter,
3785            &mut counts.skipped,
3786            &mut held_back_local,
3787        )?;
3788        for file in files {
3789            let dest = category_root.join(&file.rel);
3790            // Mirror the push guard's "is the remote missing this file" test.
3791            if !dest.is_file() {
3792                if let Some(rel) = repo_relative(repo_root, &dest) {
3793                    if tracked_seen.contains(&rel) {
3794                        held_back.push((desc.id, PathBuf::from(rel)));
3795                    }
3796                }
3797            }
3798        }
3799    }
3800    Ok(held_back)
