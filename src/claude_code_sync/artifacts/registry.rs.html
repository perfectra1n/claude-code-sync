1//! The category registry: one row per syncable artifact kind. Everything the
2//! engine does — collecting sources, choosing repo destinations, picking a
3//! merge strategy — is driven by this data.
4
5use serde::{Deserialize, Serialize};
6
7/// Identifies one syncable artifact category.
8#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
9#[serde(rename_all = "kebab-case")]
10pub enum CategoryId {
11    Settings,
12    Memory,
13    Skills,
14    Agents,
15    Commands,
16    Rules,
17    Hooks,
18    Plugins,
19    Plans,
20    Todos,
21    PromptHistory,
22    /// Non-JSONL files inside `~/.claude/projects/` (images, PDFs, per-project
23    /// memory). Gated by `!exclude_attachments` instead of a toggle, honoring
24    /// the long-documented attachments contract.
25    ProjectAttachments,
26}
27
28/// What to copy, relative to `~/.claude`. Missing sources are treated as
29/// empty, never as an error — a machine may simply not have e.g. `todos/`.
30#[derive(Debug, Clone, Copy, PartialEq, Eq)]
31pub enum SourceSpec {
32    /// Exact files. Anything not listed is structurally unreachable — this is
33    /// how `plugins/` syncs its two manifests but never `cache/`.
34    Files(&'static [&'static str]),
35    /// A directory walked recursively (symlinks not followed).
36    Dir(&'static str),
37}
38
39/// How pull reconciles a repo copy with the local file.
40#[derive(Debug, Clone, Copy, PartialEq, Eq)]
41pub enum MergeStrategy {
42    /// Remote bytes win (only written when they differ; snapshot first).
43    RawOverwrite,
44    /// Line-wise union of both sides (append-only JSONL like history.jsonl);
45    /// applied on push and pull so machines converge grow-only.
46    UnionJsonl,
47    /// Union of `MEMORY.md` indexes by link target, applied on push and pull;
48    /// every other file in the category raw-overwrites. Overwriting an index
49    /// with a sparser one orphans the other machine's memory files.
50    UnionMemoryIndex,
51}
52
53/// Where a category's files live inside the sync repository.
54#[derive(Debug, Clone, Copy, PartialEq, Eq)]
55pub enum DestRoot {
56    /// `<repo>/artifacts/<repo_subdir>/...` — the default for all categories.
57    Artifacts,
58    /// `<repo>/<sync_subdirectory>/...` — shares the session tree (used only
59    /// by project attachments, which live beside their transcripts).
60    SessionTree,
61}
62
63/// One registry row.
64#[derive(Debug)]
65pub struct CategoryDescriptor {
66    pub id: CategoryId,
67    /// Config key / CLI name (kebab-case).
68    pub name: &'static str,
69    /// Subdirectory under the sync repo's `artifacts/` root (unused for
70    /// `DestRoot::SessionTree`).
71    pub repo_subdir: &'static str,
72    pub source: SourceSpec,
73    pub merge: MergeStrategy,
74    pub dest: DestRoot,
75    /// File extensions this category never copies (e.g. attachments skip
76    /// `.jsonl`, which belongs to the session pipeline).
77    pub exclude_extensions: &'static [&'static str],
78    /// Rewrite this machine's absolute paths to tokens into the repo and back
79    /// out. For config files that embed paths; user content is stored verbatim.
80    pub tokenize_paths: bool,
81    /// Propagate deletions: a file this machine synced before and no longer has
82    /// is removed from the other side. Only for curated directories.
83    pub mirror_deletes: bool,
84    /// Short human description for wizard / config --show.
85    pub description: &'static str,
86}

88/// Root directory inside the sync repository for all artifact categories.
89/// Deliberately a constant: a fixed category→path mapping keeps pull
90/// deterministic and the deny-guarantee auditable.
91pub const ARTIFACTS_SUBDIR: &str = "artifacts";

243/// Per-category enable switches, stored in the `[sync_artifacts]` table of
244/// config.toml. Every field defaults to false so configs written by older
245/// versions parse unchanged and existing users opt in explicitly.
246#[derive(Debug, Clone, Default, Serialize, Deserialize, PartialEq, Eq)]
247pub struct ArtifactToggles {
248    #[serde(default)]
249    pub settings: bool,
250    #[serde(default)]
251    pub memory: bool,
252    #[serde(default)]
253    pub skills: bool,
254    #[serde(default)]
255    pub agents: bool,
256    #[serde(default)]
257    pub commands: bool,
258    #[serde(default)]
259    pub rules: bool,
260    #[serde(default)]
261    pub hooks: bool,
262    #[serde(default)]
263    pub plugins: bool,
264    #[serde(default)]
265    pub plans: bool,
266    #[serde(default)]
267    pub todos: bool,
268    #[serde(default)]
269    pub prompt_history: bool,
270}
271
272impl ArtifactToggles {
273    /// A toggles value with every category enabled, `hooks` included.
274    pub fn all_enabled() -> Self {
275        Self {
276            settings: true,
277            memory: true,
278            skills: true,
279            agents: true,
280            commands: true,
281            rules: true,
282            hooks: true,
283            plugins: true,
284            plans: true,
285            todos: true,
286            prompt_history: true,
287        }
288    }
289
290    /// Every category except `hooks` (used by onboarding defaults and the
291    /// `all` CLI shorthand). Hooks run whatever the sync repository hands
292    /// them, so they are only ever switched on by name.
293    pub fn all_but_hooks() -> Self {
294        Self {
295            hooks: false,
296            ..Self::all_enabled()
297        }
298    }
299
300    /// Whether any category is enabled at all.
301    pub fn any_enabled(&self) -> bool {
302        REGISTRY.iter().any(|d| self.is_enabled(d.id))
303    }
304
305    /// Read the switch for one category.
306    pub fn is_enabled(&self, id: CategoryId) -> bool {
307        match id {
308            CategoryId::Settings => self.settings,
309            CategoryId::Memory => self.memory,
310            CategoryId::Skills => self.skills,
311            CategoryId::Agents => self.agents,
312            CategoryId::Commands => self.commands,
313            CategoryId::Rules => self.rules,
314            CategoryId::Hooks => self.hooks,
315            CategoryId::Plugins => self.plugins,
316            CategoryId::Plans => self.plans,
317            CategoryId::Todos => self.todos,
318            CategoryId::PromptHistory => self.prompt_history,
319            // Attachments are gated by FilterConfig::exclude_attachments, not
320            // by a toggle; as a toggle they always read as off.
321            CategoryId::ProjectAttachments => false,
322        }
323    }
324
325    /// Flip the switch for one category.
326    pub fn set_enabled(&mut self, id: CategoryId, value: bool) {
327        match id {
328            CategoryId::Settings => self.settings = value,
329            CategoryId::Memory => self.memory = value,
330            CategoryId::Skills => self.skills = value,
331            CategoryId::Agents => self.agents = value,
332            CategoryId::Commands => self.commands = value,
333            CategoryId::Rules => self.rules = value,
334            CategoryId::Hooks => self.hooks = value,
335            CategoryId::Plugins => self.plugins = value,
336            CategoryId::Plans => self.plans = value,
337            CategoryId::Todos => self.todos = value,
338            CategoryId::PromptHistory => self.prompt_history = value,
339            CategoryId::ProjectAttachments => {}
340        }
341    }
342}

344/// Look up a descriptor by its CLI/config name (kebab-case).
345pub fn find_by_name(name: &str) -> Option<&'static CategoryDescriptor> {
346    REGISTRY.iter().find(|d| d.name == name)
347}

349/// Registry rows driven by ArtifactToggles — everything except attachments,
350/// which the long-standing exclude_attachments flag governs.
351pub fn toggleable() -> impl Iterator<Item = &'static CategoryDescriptor> {
352    REGISTRY
353        .iter()
354        .filter(|d| d.id != CategoryId::ProjectAttachments)
355}
356
357/// All registry rows whose toggle is on.
358#[allow(dead_code)] // used via the library target; the bin compiles this module separately
359pub fn enabled_categories(toggles: &ArtifactToggles) -> Vec<&'static CategoryDescriptor> {
360    REGISTRY
361        .iter()
362        .filter(|d| toggles.is_enabled(d.id))
363        .collect()
364}

408        let ph = REGISTRY
409            .iter()
410            .find(|d| d.id == CategoryId::PromptHistory)
411            .unwrap();
412        assert_eq!(ph.merge, MergeStrategy::UnionJsonl);
413        assert_eq!(ph.source, SourceSpec::Files(&["history.jsonl"]));
414        // Only project attachments, which carry per-project MEMORY.md indexes,
415        // merge as well; everything else raw-overwrites.
416        for d in REGISTRY
417            .iter()
418            .filter(|d| d.id != CategoryId::PromptHistory && d.id != CategoryId::ProjectAttachments)
419        {
420            assert_eq!(d.merge, MergeStrategy::RawOverwrite, "{}", d.name);
421        }
422        let attachments = REGISTRY
423            .iter()
424            .find(|d| d.id == CategoryId::ProjectAttachments)
425            .unwrap();
426        assert_eq!(attachments.merge, MergeStrategy::UnionMemoryIndex);
427    }

482    #[test]
483    fn test_toggles_map_every_category() {
484        let mut toggles = ArtifactToggles::default();
485        assert!(!toggles.any_enabled());
486        for d in toggleable() {
487            assert!(!toggles.is_enabled(d.id), "{} defaults off", d.name);
488            toggles.set_enabled(d.id, true);
489            assert!(toggles.is_enabled(d.id), "{} can be enabled", d.name);
490        }
491        assert_eq!(toggles, ArtifactToggles::all_enabled());
492        // Attachments are flag-governed, not toggleable.
493        assert!(!toggles.is_enabled(CategoryId::ProjectAttachments));
494        toggles.set_enabled(CategoryId::ProjectAttachments, true);
495        assert!(!toggles.is_enabled(CategoryId::ProjectAttachments));
496    }
