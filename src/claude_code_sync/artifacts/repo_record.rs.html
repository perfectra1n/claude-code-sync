1//! Repo-scoped records under `~/.claude`: one JSON file, one entry per sync
2//! repository. Shared by the tracked-paths and base-hashes records, which
3//! differ only in what they store per repository.
4//!
5//! Entries are keyed by the repository's *canonical* path, so spelling
6//! changes of the same checkout (`~/repo`, a symlink to it, a trailing
7//! slash) keep sharing one entry instead of silently forking the record.
8//! Lookups also honor the raw spelling, because a record written by an
9//! older version may still be filed under it.
10
11use anyhow::{Context, Result};
12use serde::{Deserialize, Serialize};
13use std::collections::{BTreeMap, BTreeSet};
14use std::fs;
15use std::path::{Path, PathBuf};
16use std::time::{Duration, Instant};

18/// Serializes read-modify-write cycles on the shared record files across
19/// processes — without it, two syncs against different repositories would
20/// each write the whole file and silently erase each other's entries
21/// (last writer wins, protection included).
22///
23/// An OS file lock (`flock` / `LockFileEx`) on a marker file, held for
24/// the duration of one load-merge-write: the kernel releases it when the
25/// holder exits or crashes, so there is no staleness to judge, no pid to
26/// inspect, and no break to race — the entire class of "is this lock
27/// dead?" bugs cannot exist. A lock that cannot be taken within
28/// `LOCK_GIVE_UP_AFTER` is an ERROR (see `acquire`): the callers degrade
29/// their record save to a warning rather than write unprotected.
30pub(super) struct RecordLock {
31    file: Option<fs::File>,
32}
33
34/// A legitimate record write is sub-second; a holder still running after
35/// this window is either wedged or contended, and waiting longer cannot
36/// help — every failure is an error, never a proceed-unprotected.
37const LOCK_GIVE_UP_AFTER: Duration = Duration::from_secs(15);
38
39impl RecordLock {
40    /// Acquire the lock, or explain why not. Every failure is an ERROR:
41    /// writing the shared records unprotected would silently
42    /// last-writer-win the holder's entries away — a record save that
43    /// degrades to a warning for one round costs far less.
44    fn acquire(claude_dir: &Path) -> Result<RecordLock> {
45        // The marker file is never deleted: an OS lock does not care
46        // about the file existing, only about who holds it.
47        let path = claude_dir.join(".claude-code-sync-records.lock");
48        let open = || -> Result<fs::File> {
49            fs::OpenOptions::new()
50                .create(true)
51                .read(true)
52                .write(true)
53                .truncate(false)
54                .open(&path)
55                .with_context(|| format!("cannot open record lock {}", path.display()))
56        };
57        let file = match open() {
58            Ok(file) => file,
59            // A missing Claude directory is one create away — and if the
60            // create fails, no amount of retrying will fix it (the same
61            // goes for every other permanent open error: a component that
62            // is a regular file, a read-only filesystem). Error in
63            // milliseconds, never spin the give-up window.
64            Err(_) => {
65                fs::create_dir_all(claude_dir)
66                    .with_context(|| format!("cannot create {}", claude_dir.display()))?;
67                open().inspect_err(|_| {
68                    log::warn!(
69                        "the directory exists now, but the lock file {} still cannot be \
70                         opened — check its own permissions",
71                        path.display()
72                    )
73                })?
74            }
75        };
76        let start = Instant::now();
77        loop {
78            match file.try_lock() {
79                Ok(()) => return Ok(RecordLock { file: Some(file) }),
80                // Held by someone else: wait, within the give-up window.
81                Err(std::fs::TryLockError::WouldBlock) => {
82                    if start.elapsed() > LOCK_GIVE_UP_AFTER {
83                        return Err(anyhow::anyhow!(
84                            "record lock {} not acquired in {:?} (holder still running); \
85                             refusing to write the shared records unprotected",
86                            path.display(),
87                            LOCK_GIVE_UP_AFTER
88                        ));
89                    }
90                    std::thread::sleep(Duration::from_millis(20));
91                }
92                // An I/O error is not contention (a filesystem without
93                // flock support, say): retrying cannot help, and spinning
94                // the give-up window on every record save would add 15
95                // seconds per save on exactly the machines already
96                // having trouble.
97                Err(std::fs::TryLockError::Error(e)) => {
98                    return Err(anyhow::anyhow!(
99                        "record lock {} cannot be taken on this filesystem: {e}",
100                        path.display()
101                    ));
102                }
103            }
104        }
105    }
106}

108/// Take the inter-process record lock for the duration of the returned
109/// guard's scope. Errors are part of the contract (see `acquire`): the
110/// callers degrade their record save to a warning instead of writing
111/// unprotected.
112pub(super) fn record_lock(claude_dir: &Path) -> Result<RecordLock> {
113    RecordLock::acquire(claude_dir)
114}
115
116/// Whether the record lock is SUPPORTED here — a plan-time probe that
117/// never waits: one try_lock, and contention (WouldBlock) counts as
118/// SUCCESS (the lock works; another sync just holds it). Only a hard
119/// error — a filesystem without flock support, which fails every save —
120/// comes back as Err, so the plan can announce the permanently-off
121/// protections once, where its other record-health warnings live.
122pub fn record_lock_probe(claude_dir: &Path) -> Result<()> {
123    let path = claude_dir.join(".claude-code-sync-records.lock");
124    // No `create`: the probe must not plant the marker file as a side
125    // effect of a read-only command — an absent lock file means locking
126    // trivially works (nothing to contend with, and the first save
127    // creates it).
128    let file = match std::fs::OpenOptions::new()
129        .truncate(false)
130        .write(true)
131        .open(&path)
132    {
133        Ok(file) => file,
134        Err(e) if e.kind() == std::io::ErrorKind::NotFound => return Ok(()),
135        Err(e) => return Err(e.into()),
136    };
137    match file.try_lock() {
138        Ok(_guard) => Ok(()),
139        Err(std::fs::TryLockError::WouldBlock) => Ok(()),
140        Err(std::fs::TryLockError::Error(e)) => Err(anyhow::anyhow!(
141            "record lock {} cannot be taken on this filesystem: {e}",
142            path.display()
143        )),
144    }
145}

147/// Whether `path` carries `file_name` under ANY directory, with no
148/// Claude-dir anchoring. Only meaningful on paths already known to be
149/// records (the snapshot's explicit declarations): a synced artifact can
150/// share a record's name, which is precisely why undo never uses this to
151/// DISCOVER records, only to pick a parser for a declared key. One shared
152/// helper so the per-record copies cannot drift.
153pub(super) fn is_record_file(path: &Path, file_name: &str) -> bool {
154    path.file_name().is_some_and(|n| n == file_name)
155}

157impl Drop for RecordLock {
158    fn drop(&mut self) {
159        if let Some(file) = self.file.take() {
160            let _ = file.unlock();
161        }
162        // A crashed holder needs no cleanup: the kernel released the lock
163        // when the process died.
164    }
165}

167/// A record file mapping each sync repository to the value this machine
168/// last recorded for it.
169#[derive(Debug, Serialize, Deserialize)]
170pub struct RepoRecord<V> {
171    // An explicit path (not `#[serde(default)]`): the derive's bound
172    // inference would then demand `V: Default`, which the value types have
173    // no reason to implement.
174    #[serde(default = "empty_repos")]
175    repos: BTreeMap<String, V>,
176}
177
178fn empty_repos<V>() -> BTreeMap<String, V> {
179    BTreeMap::new()
180}
181
182/// The per-entry shape `merge_under_lock` needs: a default (an absent
183/// entry) and emptiness (an emptied entry is forgotten, not planted as
184/// `"<repo>": {}`). Implemented by both record value types.
185pub trait EntryValue: Default {
186    fn is_empty(&self) -> bool;
187    /// Restore one touched key to its PRE-PULL state from `pre_pull`:
188    /// present there = set (to that value), absent = remove. One method
189    /// so the shared restore cannot drift between the map- and set-shaped
190    /// records.
191    fn restore_key(&mut self, key: &str, pre_pull: &Self);
192}
193
194impl EntryValue for BTreeMap<String, String> {
195    fn is_empty(&self) -> bool {
196        BTreeMap::is_empty(self)
197    }
198    fn restore_key(&mut self, key: &str, pre_pull: &Self) {
199        match pre_pull.get(key) {
200            Some(hash) => {
201                self.insert(key.to_string(), hash.clone());
202            }
203            None => {
204                self.remove(key);
205            }
206        }
207    }
208}
209
210impl EntryValue for BTreeSet<String> {
211    fn is_empty(&self) -> bool {
212        BTreeSet::is_empty(self)
213    }
214    fn restore_key(&mut self, key: &str, pre_pull: &Self) {
215        if pre_pull.contains(key) {
216            self.insert(key.to_string());
217        } else {
218            self.remove(key);
219        }
220    }
221}

223// Manual so an empty record needs nothing from `V` — a record read for a
224// repo that has none must default even when `V` has no meaningful default.
225impl<V> Default for RepoRecord<V> {
226    fn default() -> Self {
227        Self {
228            repos: BTreeMap::new(),
229        }
230    }
231}

233/// The canonical spelling of a repository root. Falls back to the given
234/// path when it cannot be resolved (a record may outlive its repository).
235///
236/// Known limit: a repository that is *moved* (canonicalize resolves, but
237/// to a new path no old spelling matches) orphans its entries — they sit
238/// in the shared file, protect nothing, and only a `forget` under an old
239/// spelling (or hand removal) clears them. Accepted: the failure is
240/// inert, and keying by anything but the path (a repo id file, say)
241/// would put machine state inside the synced repository.
242fn repo_key(repo_root: &Path) -> String {
243    fs::canonicalize(repo_root)
244        .map(|p| p.to_string_lossy().into_owned())
245        .unwrap_or_else(|_| repo_root.to_string_lossy().into_owned())
246}
247
248/// Every spelling the repository may be filed under: the canonical key
249/// first, then the raw path (an older version wrote that one).
250fn repo_keys(repo_root: &Path) -> Vec<String> {
251    let mut keys = vec![repo_key(repo_root)];
252    let raw = repo_root.to_string_lossy().into_owned();
253    if keys[0] != raw {
254        keys.push(raw);
255    }
256    keys
257}

259/// Whether the record file on disk parses — with the CONCRETE value
260/// type, not `serde_json::Value`: a record whose values have the wrong
261/// shape (a hash where a set belongs, a bare string where a map does)
262/// is corrupt for its loader even though it is valid JSON, and a
263/// Value-typed check would wave it through while `load` silently reads
264/// an empty record. An ABSENT record legitimately reads as empty (the
265/// fresh-machine and upgrade path); a PRESENT but unreadable or corrupt
266/// one is not that — callers warn, because silently reading it as empty
267/// turns off everything the record protects.
268pub fn intact<V: for<'de> serde::Deserialize<'de>>(record_path: &Path) -> bool {
269    match fs::read_to_string(record_path) {
270        Ok(text) => serde_json::from_str::<RepoRecord<V>>(&text).is_ok(),
271        // ABSENT is the legitimate fresh-machine state — only a PRESENT
272        // but unreadable file is corruption.
273        Err(e) if e.kind() == std::io::ErrorKind::NotFound => true,
274        Err(_) => false,
275    }
276}

278/// The mechanical per-record wrappers every record module needs, so the
279/// two cannot drift: path resolution, file-name recognition, entry
280/// extraction from raw bytes, load, forget, and save — everything that
281/// differs only in the file-name constant and the value type.
282#[macro_export]
283macro_rules! repo_record_wrappers {
284    ($file_name:expr, $value:ty) => {
285        /// Where the record lives for a given Claude directory.
286        pub fn record_path(claude_dir: &Path) -> PathBuf {
287            claude_dir.join($file_name)
288        }
289
290        /// Whether `path` carries this record's file name under ANY
291        /// directory — only meaningful on paths already known to be
292        /// records (the snapshot's explicit declarations): a synced
293        /// artifact can share the name, which is precisely why undo
294        /// never uses this to DISCOVER records, only to pick a parser
295        /// for a declared key.
296        pub fn is_record_path(path: &Path) -> bool {
297            $crate::artifacts::repo_record::is_record_file(path, $file_name)
298        }
299
300        /// The entry `repo_root` had in a whole-record file's raw bytes
301        /// (a pre-pull snapshot of the shared record, say). `None` when
302        /// the bytes do not PARSE — a caller must treat that as corrupt
303        /// (the plan-time `intact()` rule: a present-but-broken record
304        /// turns off everything it protects and must be loud, never
305        /// silently read as an empty entry); an entry-less file for a
306        /// repo that had none still reads as `Some(empty)`.
307        pub fn entry_from_record_bytes(bytes: &[u8], repo_root: &Path) -> Option<$value> {
308            match RepoRecord::<$value>::from_bytes_strict(bytes) {
309                Some(record) => Some(record.get(repo_root).unwrap_or_default()),
310                None => None,
311            }
312        }
313
314        /// What this machine last recorded for `repo_root`.
315        pub fn load(claude_dir: &Path, repo_root: &Path) -> $value {
316            RepoRecord::<$value>::read(claude_dir, $file_name)
317                .get(repo_root)
318                .unwrap_or_default()
319        }
320
321        /// Forget everything this machine recorded about `repo_root`.
322        pub fn forget(claude_dir: &Path, repo_root: &Path) -> Result<()> {
323            let _lock = $crate::artifacts::repo_record::record_lock(claude_dir)?;
324            let mut record =
325                RepoRecord::<$value>::try_read(claude_dir, $file_name)?.unwrap_or_default();
326            if record.remove(repo_root) {
327                record.write(claude_dir, $file_name)?;
328            }
329            Ok(())
330        }
331
332        /// Record the entry for `repo_root`, replacing the previous one.
333        pub fn save(claude_dir: &Path, repo_root: &Path, entry: $value) -> Result<()> {
334            let _lock = $crate::artifacts::repo_record::record_lock(claude_dir)?;
335            let mut record =
336                RepoRecord::<$value>::try_read(claude_dir, $file_name)?.unwrap_or_default();
337            record.insert(repo_root, entry);
338            record.write(claude_dir, $file_name)
339        }
340    };
341}

343impl<V: Clone + for<'de> Deserialize<'de> + Serialize> RepoRecord<V> {
344    /// The record parsed from raw file bytes (a pre-pull snapshot of the
345    /// shared file, say). Unparseable bytes read as empty — same contract
346    /// as `read`.
347    pub fn from_bytes(bytes: &[u8]) -> Self {
348        serde_json::from_slice(bytes).unwrap_or_default()
349    }
350
351    /// The record parsed from raw bytes, or `None` when they do not
352    /// parse: callers that must DISTINGUISH corrupt from empty (the
353    /// undo's record surgery) use this, so a truncated snapshot blob
354    /// warns and pins instead of silently reading as an empty pre-pull
355    /// entry and deleting live keys.
356    pub fn from_bytes_strict(bytes: &[u8]) -> Option<Self> {
357        serde_json::from_slice(bytes).ok()
358    }
359
360    /// The record as stored under `claude_dir/file_name`. An absent or
361    /// unreadable file reads as empty: a fresh machine, or one upgrading
362    /// from a version without the record, loses nothing it had before.
363    pub fn read(claude_dir: &Path, file_name: &str) -> Self {
364        let Ok(text) = fs::read_to_string(claude_dir.join(file_name)) else {
365            return Self::default();
366        };
367        serde_json::from_str(&text).unwrap_or_default()
368    }
369
370    /// The record parsed from disk OR `None` when the bytes do not
371    /// parse — every saving path (`save_delta`, `forget`) checks it
372    /// before its locked write: writing anything over a corrupt document
373    /// would erase every other repository's entries (the unwrap_to_default
374    /// path makes the lock think the file is empty). The check is a
375    /// fail-closed companion to `intact` (which only probes in plan_pull).
376    pub fn try_read(claude_dir: &Path, file_name: &str) -> Result<Option<Self>> {
377        let path = claude_dir.join(file_name);
378        match fs::read_to_string(&path) {
379            Ok(text) => match serde_json::from_str::<Self>(&text) {
380                Ok(record) => Ok(Some(record)),
381                Err(e) => Err(anyhow::anyhow!(
382                    "record {} is corrupt and cannot be merged into safely ({e}); \
383                     fix or remove it before syncing again",
384                    path.display()
385                )),
386            },
387            Err(e) if e.kind() == std::io::ErrorKind::NotFound => Ok(None),
388            Err(e) => Err(anyhow::anyhow!(
389                "cannot read record {}: {e}",
390                path.display()
391            )),
392        }
393    }
394
395    /// The value recorded for `repo_root`, under any spelling of it.
396    pub fn get(&self, repo_root: &Path) -> Option<V> {
397        repo_keys(repo_root)
398            .iter()
399            .find_map(|key| self.repos.get(key).cloned())
400    }
401
402    /// Record `value` for `repo_root` under its canonical key, retiring any
403    /// raw-spelling alias an older version left behind.
404    pub fn insert(&mut self, repo_root: &Path, value: V) {
405        let mut keys = repo_keys(repo_root);
406        // The canonical key comes first; the rest are legacy spellings.
407        let canonical = keys.remove(0);
408        for key in keys {
409            self.repos.remove(&key);
410        }
411        self.repos.insert(canonical, value);
412    }
413
414    /// Forget everything recorded about `repo_root`, under any spelling.
415    pub fn remove(&mut self, repo_root: &Path) -> bool {
416        let mut removed = false;
417        for key in repo_keys(repo_root) {
418            removed |= self.repos.remove(&key).is_some();
419        }
420        removed
421    }
422
423    /// Whether `repo_root` is filed under a non-canonical spelling — an
424    /// older version's raw key. The next `insert` retires it, which
425    /// rewrites the file; writers that predict their writes need to know.
426    pub fn has_aliased_entry(&self, repo_root: &Path) -> bool {
427        let keys = repo_keys(repo_root);
428        keys[1..].iter().any(|key| self.repos.contains_key(key))
429    }
430
431    /// The ONE locked read-merge-write every record mutation goes through:
432    /// the lock covers the LOAD half too, so a same-repo writer landing
433    /// between a caller's fresh read and its save cannot be silently
434    /// overwritten (a load outside this function races every other
435    /// writer — `save`'s own lock covers only its write). `merge` mutates
436    /// this repository's entry in place; an entry that empties out is
437    /// forgotten rather than planted as `"<repo>": {}`.
438    pub fn merge_under_lock(
439        claude_dir: &Path,
440        file_name: &str,
441        repo_root: &Path,
442        merge: impl FnOnce(&mut V),
443    ) -> Result<()>
444    where
445        V: EntryValue,
446    {
447        let _lock = record_lock(claude_dir)?;
448        // Fail-closed on a corrupt record: see `try_read`'s doc — a
449        // corrupt file reading as empty would erase every other
450        // repository's entries on the first save.
451        let mut record = Self::try_read(claude_dir, file_name)?.unwrap_or_default();
452        let mut current: V = record.get(repo_root).unwrap_or_default();
453        merge(&mut current);
454        if current.is_empty() {
455            if record.remove(repo_root) {
456                return record.write(claude_dir, file_name);
457            }
458            return Ok(());
459        }
460        record.insert(repo_root, current);
461        record.write(claude_dir, file_name)
462    }
463
464    /// The undo's per-key restore, shared by BOTH records: set each
465    /// touched key back to its pre-pull value (absent pre-pull = remove),
466    /// in one locked merge — a later push's entries keep their owner.
467    /// One source, so the two wrappers cannot drift (one restoring
468    /// per-key while the other goes wholesale would silently strand half
469    /// the undo).
470    pub fn restore_keys_under_lock(
471        claude_dir: &Path,
472        file_name: &str,
473        repo_root: &Path,
474        pre_pull: &V,
475        touched: &[String],
476    ) -> Result<()>
477    where
478        V: EntryValue,
479    {
480        Self::merge_under_lock(claude_dir, file_name, repo_root, |current| {
481            for key in touched {
482                current.restore_key(key, pre_pull);
483            }
484        })
485    }
486
487    /// Write the record back atomically (temp file + rename), skipping the
488    /// write when the file already holds exactly this content — an unchanged
489    /// sync must not churn the file (or its mtime) for nothing. A crash
490    /// mid-write must not truncate the record: both files reading as empty
491    /// is one unprotected, remote-wins cycle for every repository.
492    pub fn write(&self, claude_dir: &Path, file_name: &str) -> Result<()> {
493        let path: PathBuf = claude_dir.join(file_name);
494        let text = serde_json::to_string_pretty(self)?;
495        if let Ok(existing) = fs::read_to_string(&path) {
496            if existing == text {
497                return Ok(());
498            }
499        }
500        fs::create_dir_all(claude_dir)?;
501        let tmp = tempfile::NamedTempFile::new_in(claude_dir)
502            .with_context(|| format!("Failed to stage {}", path.display()))?;
503        fs::write(tmp.path(), &text)?;
504        // Flush to the disk BEFORE the rename: with delayed allocation
505        // (ext4/xfs), a rename landing before the data blocks turns a
506        // power loss right after this write into a zero-length or
507        // truncated record — the exact corruption the temp+rename exists
508        // to prevent. A record reading as empty is one unprotected,
509        // remote-wins cycle for every repository, so the fsync cost is
510        // the cheap half of the invariant.
511        tmp.as_file()
512            .sync_all()
513            .with_context(|| format!("Failed to flush {}", path.display()))?;
514        tmp.persist(&path)
515            .with_context(|| format!("Failed to write {}", path.display()))?;
516        Ok(())
517    }
518}

635        let claude = tempfile::tempdir().unwrap();
636        let held = record_lock(claude.path()).unwrap();
637        // The kernel owns the arbitration now: a second taker — even in
638        // the same process, on its own handle — fails while held.
639        let second = fs::OpenOptions::new()
640            .create(true)
641            .read(true)
642            .write(true)
643            .truncate(false)
644            .open(claude.path().join(".claude-code-sync-records.lock"))
645            .unwrap();
646        assert!(second.try_lock().is_err(), "the held lock blocks");
647        drop(held);
648        assert!(second.try_lock().is_ok(), "release reopens it");
649        let _ = second.unlock();
650    }
